def _price_history(periods: int = 160) -> pd.DataFrame:
    # A sawtooth: the sustained drops carry price under its lower Bollinger band
    # while RSI is oversold, so bollinger-rsi enters, and the climbs back through
    # the middle band close the trade -- giving the gate tests real trades.
    price = 100.0
    closes = []
    for number in range(periods):
        price += 2.0 if number % 40 < 25 else -3.0
        closes.append(price)
    return pd.DataFrame(
        {
            "Date": pd.bdate_range("2024-01-01", periods=periods),
            "Symbol": "000001", "Name": "A", "Market": "KOSPI",
            "Open": closes, "High": [value + 1 for value in closes],
            "Low": [value - 1 for value in closes], "Close": closes,
            "Volume": 10_000, "TradingValue": 6_000_000_000, "MarketCap": 1_000_000_000_000,
        }
    )


def _trades() -> pd.DataFrame:
    return pd.DataFrame(
        {
            "Segment": ["out_of_sample"] * 130,
            "Market": ["KOSPI"] * 100 + ["KOSDAQ"] * 30,
            "Strategy": ["bollinger-rsi"] * 100 + ["rsi-obv"] * 30,
            "NetReturnPct": [0.35] * 70 + [-0.02] * 30 + [0.30] * 21 + [-0.02] * 9,
        }
    )


def _windowed_trades(window_losses: tuple[int, ...]) -> pd.DataFrame:
    """One bollinger-rsi scope of ten-trade windows, given losses per window.

    Callers pass the same number of losses in total, so win rate, median and
    profit factor come out identical; only how the losses cluster changes.
    """
    returns: list[float] = []
    windows: list[int] = []
    for window, losses in enumerate(window_losses, start=1):
        returns.extend([0.35] * (10 - losses) + [-0.50] * losses)
        windows.extend([window] * 10)
    return pd.DataFrame(
        {
            "Segment": "out_of_sample", "Market": "KOSPI", "Strategy": "bollinger-rsi",
            "Window": windows, "NetReturnPct": returns,
        }
    )


def test_goal_summary_uses_costed_trade_return_and_strategy_sample_sizes():
    summary = summarize_reversion_trades(_trades(), ReversionGoal())
    swing = summary[(summary["Market"] == "KOSPI") & (summary["Strategy"] == "bollinger-rsi")].iloc[0]
    long_term = summary[(summary["Market"] == "KOSDAQ") & (summary["Strategy"] == "rsi-obv")].iloc[0]
    assert swing["Closed Trade Count"] == 100
    assert swing["Win Rate"] == pytest.approx(0.70)
    # 70 wins of 0.35 against 30 losses of 0.02.
    assert swing["Profit Factor"] == pytest.approx(24.5 / 0.6)
    assert swing["Payoff Ratio"] == pytest.approx(0.35 / 0.02)
    assert swing["Median Net Return"] > 0
    assert bool(swing["Passes Goal"])
    assert long_term["Closed Trade Count"] == 30
    assert bool(long_term["Passes Goal"])


def test_profit_factor_alone_cannot_tell_a_steady_rule_from_a_lucky_one():
    # The measured case this guards: the broken rsi-obv exit scored a *higher*
    # profit factor than the fixed one (1.600 vs 1.562) while only 48.5% of its
    # out-of-sample windows closed profitably against 71.4%. Replacing the old
    # mean-net-return bar with profit factor was necessary but not sufficient --
    # window share is the criterion that actually separates the two.
    hidden = summarize_reversion_trades(
        _windowed_trades((0, 0, 0, 0, 5, 5, 5, 5, 5, 5)), ReversionGoal()
    ).iloc[0]
    steady = summarize_reversion_trades(_windowed_trades((3,) * 10), ReversionGoal()).iloc[0]

    assert hidden["Profit Factor"] == pytest.approx(steady["Profit Factor"])
    assert hidden["Win Rate"] == pytest.approx(steady["Win Rate"])
    assert hidden["Profitable Window Share"] == pytest.approx(0.40)
    assert steady["Profitable Window Share"] == pytest.approx(1.00)
    assert not bool(hidden["Passes Goal"]), "profit concentrated in four windows is not evidence"
    assert bool(steady["Passes Goal"])


def test_a_scope_without_window_labels_is_not_failed_on_a_measure_it_cannot_take():
    # Unlabelled trades mean the metric is undefined, not that the rule was
    # unstable. Window Count is published beside the share so a reader can see
    # which of the two it is.
    summary = summarize_reversion_trades(_trades(), ReversionGoal())
    row = summary[(summary["Market"] == "KOSPI") & (summary["Strategy"] == "bollinger-rsi")].iloc[0]
    assert row["Window Count"] == 0
    assert row["Profitable Window Share"] == pytest.approx(0.0)
    assert bool(row["Passes Goal"])


def test_open_positions_are_counted_rather_than_deleted_from_the_losing_tail():
    # The bug this replaces: the old harness dropped every non-closed position
    # before summarising. These strategies exit only on recovery, so an open
    # position is by construction one that had not recovered -- deleting them
    # deleted only losers, and a 96%-censored sample read as a 96% win rate.
    trades = _trades()
    losing = trades.iloc[:100].assign(Status="open", NetReturnPct=-0.40)
    mixed = pd.concat([trades.assign(Status="closed"), losing], ignore_index=True)
    row = summarize_reversion_trades(mixed, ReversionGoal())
    swing = row[(row["Market"] == "KOSPI") & (row["Strategy"] == "bollinger-rsi")].iloc[0]
    assert swing["Open Trade Count"] == 100, "unclosed positions must stay visible"
    assert swing["Closed Trade Count"] == 100, "and must not inflate the realised sample"
    assert swing["Win Rate"] == pytest.approx(0.70), "goal metrics still use realised returns only"


def test_window_labels_follow_the_entry_date_and_never_drop_a_boundary_trade():
    # A trade entered inside window 1 belongs to window 1 even when it exits
    # deep into window 2 -- that is the window a live trader was deciding in.
    # Slicing per window instead would have deleted it as "still open".
    dates = pd.DatetimeIndex(pd.bdate_range("2024-01-01", periods=300))
    config = ReversionValidationConfig(training_sessions=100, testing_sessions=50)
    trades = pd.DataFrame(
        {
            "Entry Timestamp": [dates[50], dates[100], dates[149], dates[150]],
            "Exit Timestamp": [dates[120], dates[280], dates[299], dates[299]],
            "NetReturnPct": [0.1, -0.5, -0.5, 0.1],
        }
    )
    labelled = _label_windows(trades, dates, config)
    assert len(labelled) == len(trades), "labelling must never remove a trade"
    assert pd.isna(labelled["Window"].iloc[0]), "a training-period entry is not out-of-sample"
    # Entry at 100 and 149 both sit in the first 50-session window; 150 opens the next.
    assert list(labelled["Window"].iloc[1:]) == [1, 1, 2]


def test_goal_fails_when_closed_trade_sample_is_too_small():
    summary = summarize_reversion_trades(_trades().iloc[:99], ReversionGoal())
    row = summary[(summary["Market"] == "KOSPI") & (summary["Strategy"] == "bollinger-rsi")].iloc[0]
    assert not bool(row["Passes Goal"])


def test_validation_requires_expected_history_and_out_of_sample_sessions():
    history = pd.DataFrame({"Date": pd.date_range("2024-01-01", periods=3)})
    with pytest.raises(ValueError, match="missing"):
        run_reversion_validation(history)

    complete = pd.DataFrame(
        {
            "Date": pd.date_range("2024-01-01", periods=3), "Symbol": ["000001"] * 3,
            "Name": ["A"] * 3, "Market": ["KOSPI"] * 3, "Open": [100] * 3,
            "High": [100] * 3, "Low": [100] * 3, "Close": [100] * 3,
            "Volume": [1] * 3, "TradingValue": [100] * 3, "MarketCap": [100] * 3,
        }
    )
    with pytest.raises(ValueError, match="after the training"):
        run_reversion_validation(complete, ReversionValidationConfig(training_sessions=3))


def test_validation_report_writes_summary_and_trade_records(tmp_path):
    trades = _trades()
    result = type("Result", (), {"trades": trades, "summary": summarize_reversion_trades(trades)})()
    # A stale checkpoint must not bring the retired presentation column back.
    result.summary["Cumulative Return (KRW)"] = 123.0
    result.trades["Cumulative Return (KRW)"] = 123.0
    html, csv = write_reversion_validation_reports(
        result, tmp_path / "validation.html", tmp_path / "validation.csv"
    )
    assert "out-of-sample" in html.read_text(encoding="utf-8")
    assert {"summary", "trade"} <= set(pd.read_csv(csv)["RecordType"])
    assert "Cumulative Return (KRW)" not in html.read_text(encoding="utf-8")
    assert "Cumulative Return (KRW)" not in pd.read_csv(csv).columns


def test_checkpoint_manifest_reuses_matching_inputs_and_rejects_mismatch(tmp_path):
    history = pd.DataFrame({"Date": pd.to_datetime(["2024-01-02", "2024-01-03"])})
    checkpoint = tmp_path / "checkpoints"
    config = ReversionValidationConfig(checkpoint_dir=checkpoint)

    assert _prepare_checkpoint_dir(history, config) == checkpoint
    assert _prepare_checkpoint_dir(history, config) == checkpoint
    changed = ReversionValidationConfig(
        checkpoint_dir=checkpoint, market_cap_tiers=(("KOSPI", 50), ("KOSDAQ", 50))
    )
    with pytest.raises(ValueError, match="different validation inputs"):
        _prepare_checkpoint_dir(history, changed)


def test_regime_gate_defaults_to_the_inverted_orientation():
    # This family buys weakness, so its gate must read the benchmark the
    # opposite way round from the trend harness: below its own average, not
    # above. Getting this backwards would silently invert the whole experiment.
    assert ReversionValidationConfig().market_filter_inverted is True


def test_market_filter_window_must_be_disabled_or_usable():
    history = _price_history()
    with pytest.raises(ValueError, match="zero .disabled. or at least two"):
        run_reversion_validation(history, ReversionValidationConfig(market_filter_window=1))


def test_market_regime_gate_blocks_entries_without_touching_the_baseline():
    history = _price_history()
    config = ReversionValidationConfig(
        training_sessions=40, testing_sessions=20, liquidity_window=5,
        minimum_history_sessions=10, minimum_average_trading_value=0,
        market_cap_tiers=(("KOSPI", 1),), strategies=("bollinger-rsi",),
    )
    baseline = run_reversion_validation(history, config)
    assert not baseline.trades.empty, "fixture must trade, or the gate test proves nothing"

    dates = pd.DatetimeIndex(history["Date"].drop_duplicates())
    # A benchmark that only ever falls never rises above its own average, so the
    # inverted default admits every entry and the plain orientation blocks them.
    falling = {"KOSPI": pd.Series(range(len(dates), 0, -1), index=dates, dtype=float)}
    gated = replace(config, market_filter_window=5)
    assert not run_reversion_validation(history, gated, falling).trades.empty
    upright = replace(gated, market_filter_inverted=False)
    assert run_reversion_validation(history, upright, falling).trades.empty


def _dated_trades() -> pd.DataFrame:
    """A passing bollinger-rsi scope whose signals all arrive on one session.

    Per-trade metrics do not care that a hundred names fired at once; a
    portfolio does, because it can only buy a few of them.
    """
    trades = _trades()
    trades = trades[trades["Strategy"] == "bollinger-rsi"].reset_index(drop=True)
    return trades.assign(
        Symbol=[f"{number:06d}" for number in range(len(trades))],
        Status="closed",
        **{
            "Entry Timestamp": pd.Timestamp("2024-01-02"),
            "Exit Timestamp": pd.Timestamp("2024-03-01"),
        },
    )


def test_combined_row_dedupes_a_symbol_held_by_the_other_strategy():
    # daily_picks_cli.py pools swing and long_term into one tracked_symbols set
    # before either can enter a name the other already holds. A per-strategy
    # capital row can never see that collision -- this proves the pooled
    # "combined" row does, by overlapping one symbol's bollinger-rsi trade with
    # a same-day rsi-obv trade on the same symbol.
    trades = pd.DataFrame(
        {
            "Segment": ["out_of_sample"] * 2,
            "Market": ["KOSPI"] * 2,
            "Strategy": ["bollinger-rsi", "rsi-obv"],
            "Symbol": ["000001", "000001"],
            "Status": ["closed", "closed"],
            "NetReturnPct": [0.10, 0.10],
            "Entry Timestamp": [pd.Timestamp("2024-01-02")] * 2,
            "Exit Timestamp": [pd.Timestamp("2024-03-01")] * 2,
        }
    )
    summary = summarize_reversion_trades(trades)
    kospi = summary[summary["Market"] == "KOSPI"]
    per_strategy = kospi[kospi["Strategy"].isin(["bollinger-rsi", "rsi-obv"])]
    combined = kospi[kospi["Strategy"] == "combined"].iloc[0]

    # Neither per-strategy row sees the other strategy's trade, so both count
    # their one entry as deployed.
    assert per_strategy["Deployed Trade Count"].tolist() == [1, 1]
    # The pooled row sees both and must skip the second as already held.
    assert combined["Deployed Trade Count"] == 1
    assert combined["Skipped Held Already"] == 1
    assert pd.isna(combined["Passes Goal"])
    assert pd.isna(combined["Minimum Trade Count"])


def test_no_combined_row_when_a_scope_has_only_one_strategy():
    # A market where only one strategy fired candidates has nothing to pool --
    # emitting a "combined" row identical to the single-strategy row would be
    # noise, not information.
    trades = _dated_trades()
    summary = summarize_reversion_trades(trades)
    assert "combined" not in summary.loc[summary["Market"] == "KOSPI", "Strategy"].tolist()


def test_the_deployment_rule_changes_the_capital_columns_and_never_the_verdict():
    # Capital metrics are reported, not enforced. Which CAGR a strategy must
    # clear is a DECISIONS.md decision, so a change to the deployment rule must
    # move every capital number and leave PASS exactly where it was.
    trades = _dated_trades()
    narrow = summarize_reversion_trades(trades, None, DeploymentConfig(daily_entry_cap=1))
    wide = summarize_reversion_trades(trades, None, DeploymentConfig(daily_entry_cap=100))

    assert narrow["Deployed Trade Count"].sum() < wide["Deployed Trade Count"].sum()
    assert narrow["Skipped Daily Cap"].sum() > wide["Skipped Daily Cap"].sum()
    assert narrow["Passes Goal"].tolist() == wide["Passes Goal"].tolist()


def test_a_summary_without_entry_dates_reports_blanks_rather_than_failing():
    # Callers that only ever wanted per-trade metrics never had to carry entry
    # and exit dates. This layer arriving is no reason to start rejecting them,
    # and a zero CAGR would be a claim where n/a is the truth.
    summary = summarize_reversion_trades(_trades())
    assert summary["CAGR"].isna().all()
    assert summary["Greedy Hindsight High CAGR"].isna().all()
    # The pooled "combined" row has no per-strategy goal to check against, so
    # it alone is allowed to leave Passes Goal unevaluated.
    single_strategy = summary[summary["Strategy"] != "combined"]
    assert single_strategy["Passes Goal"].notna().all()

def test_deployment_holding_cap_skips_locked_limit_down_and_uses_next_open():
    dates = pd.to_datetime(["2024-01-02", "2024-01-03", "2024-01-04"])
    history = pd.DataFrame(
        {
            "Date": dates,
            "Symbol": "000001",
            "Name": "A",
            "Market": "KOSPI",
            "Open": [100.0, 70.5, 80.0],
            "High": [101.0, 70.5, 81.0],
            "Low": [99.0, 70.5, 79.0],
            "Close": [100.0, 70.5, 80.0],
            "Volume": [1_000, 1_000, 1_000],
            "TradingValue": [100_000, 70_500, 80_000],
            "MarketCap": [1_000_000_000] * 3,
        }
    )
    trades = pd.DataFrame(
        {
            "Symbol": ["000001"],
            "Status": ["closed"],
            "Entry Timestamp": [dates[0]],
            "Exit Timestamp": [pd.Timestamp("2024-02-01")],
            "NetReturnPct": [0.50],
            "Size": [1.0],
            "Avg Entry Price": [100.05],
            "Entry Fees": [0.0150075],
        }
    )
    config = ReversionValidationConfig(deployment=DeploymentConfig(max_holding_days=1))
    valued = _attach_deployment_valuations(trades, history, config)

    assert valued.loc[0, "Deployment Exit Timestamp"] == dates[2]
    entry_cash = 100.05 + 0.0150075
    exit_cash = 80.0 * (1 - config.slippage_rate) * (
        1 - config.fee_rate - config.sell_tax_rate
    )
    assert valued.loc[0, "Deployment NetReturnPct"] == pytest.approx(
        exit_cash / entry_cash - 1
    )


def test_live_selection_key_uses_previous_close_rank_not_entry_day_rank():
    dates = pd.to_datetime(["2024-01-02", "2024-01-03"])
    ranked = pd.DataFrame(
        {
            "Date": [dates[0], dates[0], dates[1], dates[1]],
            "Market": ["KOSPI"] * 4,
            "Symbol": ["A", "B", "A", "B"],
            "MarketCapRank": [2, 1, 1, 2],
        }
    )
    trades = pd.DataFrame(
        {
            "Entry Timestamp": [dates[1]] * 2,
            "Market": ["KOSPI"] * 2,
            "Symbol": ["A", "B"],
            "Strategy": ["bollinger-rsi", "rsi-obv"],
        }
    )
    attached = _attach_live_selection_keys(trades, ranked)
    key = attached.set_index("Symbol")["Live Selection Key"]
    assert key["A"] == "000002:A:00"
    assert key["B"] == "000001:B:01"
