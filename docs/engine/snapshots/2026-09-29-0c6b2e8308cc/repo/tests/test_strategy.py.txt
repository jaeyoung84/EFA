def test_signal_is_shifted_to_next_session():
    close = pd.Series([1.0, 1.0, 1.0, 3.0, 4.0], index=pd.date_range("2024-01-01", periods=5))

    result = moving_average_signals(close, fast_window=2, slow_window=3)

    regime = result["fast"] > result["slow"]
    first_regime_date = regime[regime].index[0]
    first_entry_date = result.index[result["entries"]][0]
    assert first_entry_date > first_regime_date

def test_bollinger_rsi_orders_are_shifted_and_do_not_overlap():
    index = pd.bdate_range("2024-01-01", periods=12)
    close = pd.Series([10, 10, 10, 10, 10, 6, 7, 8, 10, 11, 10, 9], index=index)

    result = bollinger_rsi_signals(
        close, window=3, standard_deviations=1, rsi_window=2, oversold=40, exit_rsi=55
    )

    observed_entry = (close <= result["bb_lower"]) & (result["rsi"] <= 40)
    assert result.loc[observed_entry, "entries"].sum() == 0
    assert result["entries"].any()
    assert result.index[result["entries"]][0] > result.index[observed_entry][0]
    assert not (result["entries"] & result["exits"]).any()

def test_volatility_squeeze_is_causal_and_returns_required_columns():
    index = pd.bdate_range("2024-01-01", periods=145)
    close = pd.Series(
        [100 + (position % 7) for position in range(110)]
        + [106 + (position % 2) * 0.1 for position in range(34)]
        + [110],
        index=index,
    )
    prices = pd.DataFrame(
        {"High": close + 1, "Low": close - 1, "Close": close, "Volume": 1_000}, index=index
    )
    prices.loc[index[-1], "Volume"] = 3_000

    short = volatility_squeeze_signals(
        prices.iloc[:-1], ema_window=5, band_window=5, atr_window=3,
        compression_lookback=10, breakout_window=5, volume_window=5,
    )
    full = volatility_squeeze_signals(
        prices, ema_window=5, band_window=5, atr_window=3,
        compression_lookback=10, breakout_window=5, volume_window=5,
    )

    assert {
        "ema30", "squeeze_bandwidth", "squeeze_atr_percent", "squeeze",
        "breakout_level", "volume_average", "volume_confirmed", "breakout", "entries", "exits",
    } == set(full)
    pd.testing.assert_frame_equal(short, full.loc[short.index])

def test_donchian_uses_prior_channel_and_shifts_entry():
    index = pd.bdate_range("2024-01-01", periods=10)
    close = pd.Series([10, 10, 10, 11, 12, 13, 14, 15, 14, 13], index=index)
    prices = pd.DataFrame(
        {"High": close + 0.5, "Low": close - 0.5, "Close": close}, index=index
    )

    result = donchian_atr_signals(
        prices, entry_window=3, exit_window=2, atr_window=2, atr_multiplier=2
    )

    breakout = close > result["donchian_upper"]
    first_breakout = breakout[breakout].index[0]
    first_order = result.index[result["entries"]][0]
    assert not bool(result.loc[first_breakout, "entries"])
    assert first_order > first_breakout
    assert result.loc[index[3], "donchian_upper"] == prices["High"].iloc[:3].max()


def test_donchian_atr_stop_exits_on_next_session():
    index = pd.bdate_range("2024-01-01", periods=11)
    close = pd.Series([10, 10, 10, 12, 14, 15, 16, 15, 13, 12, 11], index=index)
    prices = pd.DataFrame(
        {"High": close + 0.5, "Low": close - 0.5, "Close": close}, index=index
    )
    result = donchian_atr_signals(
        prices, entry_window=3, exit_window=2, atr_window=2, atr_multiplier=1
    )

    assert result["entries"].sum() == 1
    assert result["exits"].sum() == 1
    exit_order_date = result.index[result["exits"]][0]
    prior_date = result.index[result.index.get_loc(exit_order_date) - 1]
    assert close.loc[prior_date] <= result.loc[prior_date, "atr_stop"] or (
        close.loc[prior_date] < result.loc[prior_date, "donchian_lower"]
    )

def test_macd_trend_entry_is_shifted_to_next_session():
    index = pd.bdate_range("2024-01-01", periods=20)
    close = pd.Series([10, 9, 8, 7, 6, 7, 8, 9, 10, 11, 12, 13, 12, 11, 10, 9, 8, 9, 10, 11], index=index)

    result = macd_trend_signals(
        close, fast_span=2, slow_span=4, signal_span=2, trend_window=3
    )

    above = result["macd"] > result["macd_signal"]
    bullish_cross = above & ~above.shift(1, fill_value=False) & (close > result["trend_sma"])
    first_observation = bullish_cross[bullish_cross].index[0]
    first_order = result.index[result["entries"]][0]
    assert not bool(result.loc[first_observation, "entries"])
    assert first_order > first_observation


def test_macd_trend_filter_blocks_cross_below_long_sma():
    index = pd.bdate_range("2024-01-01", periods=20)
    close = pd.Series([20, 19, 18, 17, 16, 17, 18, 19, 18, 17, 16, 15, 14, 15, 16, 17, 16, 15, 14, 13], index=index)
    result = macd_trend_signals(
        close, fast_span=2, slow_span=4, signal_span=2, trend_window=10
    )
    above = result["macd"] > result["macd_signal"]
    crosses = above & ~above.shift(1, fill_value=False)

    blocked = crosses & (close <= result["trend_sma"])
    assert blocked.any()
    assert not result["entries"].shift(-1, fill_value=False).loc[blocked].any()

def test_rsi_obv_calculation_and_next_session_entry():
    index = pd.bdate_range("2024-01-01", periods=12)
    close = pd.Series([10, 9, 8, 7, 6, 7, 8, 9, 10, 11, 10, 9], index=index)
    volume = pd.Series([100, 100, 100, 100, 100, 300, 300, 300, 200, 200, 400, 400], index=index)
    prices = pd.DataFrame({"Close": close, "Volume": volume})

    result = rsi_obv_signals(
        prices, rsi_window=2, oversold=40, exit_rsi=70, obv_signal_window=2
    )

    expected_obv = (volume * close.diff().apply(
        lambda value: 1 if value > 0 else (-1 if value < 0 else 0)
    )).cumsum()
    pd.testing.assert_series_equal(result["obv"], expected_obv, check_names=False)
    observations = result["rsi_recovery"] & result["obv_rising"]
    first_observation = observations[observations].index[0]
    first_order = result.index[result["entries"]][0]
    assert not bool(result.loc[first_observation, "entries"])
    assert first_order > first_observation

def test_rsi_obv_exits_on_neutral_rsi_without_waiting_for_obv():
    # The old rule was `(rsi >= 70) & obv_weakening`. A recovery that stalls
    # short of the opposite extreme never satisfied it, so the position was
    # held indefinitely -- 9.4% of a ten-year sample never closed. The exit now
    # targets the completion of the reversion, the same level bollinger-rsi
    # uses, and the OBV term no longer gates it.
    index = pd.bdate_range("2024-01-01", periods=40)
    # Fall hard enough to drive RSI under 30, then recover past neutral and
    # stall there -- the shape that used to trap a position forever.
    recovery = [43.0, 46.0, 49.0, 52.0] + [51.0 if step % 2 else 52.0 for step in range(21)]
    close = pd.Series([100.0] * 5 + [100.0 - 6 * step for step in range(1, 11)] + recovery, index=index)
    prices = pd.DataFrame({"Close": close, "Volume": 1_000.0}, index=index)

    signals = rsi_obv_signals(prices, rsi_window=5, obv_signal_window=3)
    assert signals["entries"].any(), "the recovery has to produce an entry to test the exit"
    assert signals["exits"].any(), "a stalled recovery must still close"
    assert signals["rsi"].max() < 70.0, "and it must close without ever reaching overbought"

    observed = signals.index.get_loc(signals.index[signals["exits"]][0]) - 1
    assert signals["rsi"].iloc[observed] >= 50.0, "the exit is observed at neutral RSI"

    # The same prices under the level the rule used to require: nothing closes.
    assert not rsi_obv_signals(
        prices, rsi_window=5, exit_rsi=70.0, obv_signal_window=3
    )["exits"].any(), "this is the trade the old rule held indefinitely"


def test_ema_resistance_confirms_swings_causally_and_shifts_entry():
    index = pd.bdate_range("2024-01-01", periods=12)
    high = pd.Series([10, 12, 10, 11, 12.02, 10, 11, 11.5, 13, 14, 13, 12], index=index)
    close = pd.Series([9, 10, 9, 10, 11, 9, 10, 11, 12.5, 13, 12, 11], index=index)
    open_ = close - 0.3
    open_.iloc[8] = 11
    low = pd.Series([8, 9, 8, 9, 9, 8.5, 9, 10, 10, 11, 10, 9], index=index)
    prices = pd.DataFrame({"Open": open_, "High": high, "Low": low, "Close": close})

    result = ema_resistance_signals(
        prices, ema_window=3, resistance_threshold=0.005, minimum_touches=2,
        strong_body_lookback=2, strong_body_multiplier=1.5,
    )

    assert pd.isna(result.loc[index[1], "confirmed_swing_high"])
    assert result.loc[index[2], "confirmed_swing_high"] == 12
    assert result.loc[index[5], "confirmed_swing_high"] == pytest.approx(12.02)
    observation_dates = result.index[result["breakout"]]
    assert len(observation_dates) == 1
    order_date = result.index[result["entries"]][0]
    assert order_date > observation_dates[0]


def test_ema_resistance_past_does_not_change_when_future_is_appended():
    index = pd.bdate_range("2024-01-01", periods=14)
    high = pd.Series([10, 12, 10, 11, 12.02, 10, 11, 13, 10, 15, 10, 16, 10, 17], index=index)
    close = high - 1
    open_ = close - 0.5
    low = close - 1
    short = pd.DataFrame(
        {"Open": open_.iloc[:10], "High": high.iloc[:10], "Low": low.iloc[:10],
         "Close": close.iloc[:10]}
    )
    full = pd.DataFrame({"Open": open_, "High": high, "Low": low, "Close": close})

    short_result = ema_resistance_signals(short, ema_window=3)
    full_result = ema_resistance_signals(full, ema_window=3)

    pd.testing.assert_frame_equal(short_result, full_result.loc[short_result.index])

def test_darvas_box_entry_shifts_and_stop_only_ratchets_up():
    index = pd.bdate_range("2024-01-01", periods=10)
    high = pd.Series([11, 11, 11, 13.5, 14, 12.5, 13, 13.2, 13, 12], index=index)
    low = pd.Series([9, 9, 9, 12.5, 12, 11.5, 11, 11.2, 10.5, 10], index=index)
    close = pd.Series([10, 10, 10, 13, 13.5, 12, 12.5, 12.8, 11.0, 11.2], index=index)
    prices = pd.DataFrame({"High": high, "Low": low, "Close": close}, index=index)

    result = darvas_box_signals(prices, box_window=3)

    assert result["entries"].sum() == 1
    assert result.index[result["entries"]][0] == index[4]
    assert result["exits"].sum() == 1
    assert result.index[result["exits"]][0] == index[9]
    # The box floor briefly dips (11) below the ratcheted stop (11.5); the
    # stop must not follow it back down.
    assert result.loc[index[7], "box_floor"] == 11
    assert result.loc[index[6], "box_stop"] == 11.5
    assert result.loc[index[7], "box_stop"] == 11.5

def test_turtle_entry_shifts_pyramids_and_ratchets_stop():
    index = pd.bdate_range("2024-01-01", periods=16)
    close = pd.Series(
        [100, 100, 100, 100, 100, 100, 110, 112, 115, 119, 124, 95, 90, 88, 87, 86],
        index=index,
    )
    prices = pd.DataFrame({"High": close + 1, "Low": close - 1, "Close": close}, index=index)

    result = turtle_signals(
        prices, entry_window=5, exit_window=3, atr_window=3,
        atr_stop_multiplier=2, pyramid_atr_fraction=0.5, max_units=3,
    )

    entry_dates = list(result.index[result["entries"]])
    assert entry_dates == list(pd.to_datetime(["2024-01-10", "2024-01-12", "2024-01-15"]))
    assert list(result.index[result["exits"]]) == list(pd.to_datetime(["2024-01-17"]))
    assert result["units"].max() == 3

    holding_stops = result["turtle_stop"].dropna()
    assert (holding_stops.diff().dropna() >= 0).all()
