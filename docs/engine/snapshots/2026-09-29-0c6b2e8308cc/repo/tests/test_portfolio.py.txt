import pandas as pd

from korea_quant.portfolio import equal_weight_targets


def test_equal_weight_targets_limits_candidates():
    i=pd.bdate_range("2024-01-01", periods=2); c=["A","B","C"]
    e=pd.DataFrame([[True,True,True],[False,False,False]],index=i,columns=c)
    s=pd.DataFrame([[1,3,2],[1,2,3]],index=i,columns=c)
    w=equal_weight_targets(e,s,2)
    assert w.loc[i[0]].tolist()==[0.0,0.5,0.5]
