class FakePortfolio:
    def __init__(self):
        self._value = pd.Series(
            [100.0, 110.0, 99.0, 120.0],
            index=pd.to_datetime(["2024-01-30", "2024-01-31", "2024-02-01", "2024-02-29"]),
        )
        self.trades = SimpleNamespace(
            records_readable=pd.DataFrame(
                {"PnL": [10.0, -5.0, 7.0], "Status": ["Closed", "Closed", "Open"]}
            )
        )
        self.orders = SimpleNamespace(records_readable=pd.DataFrame({"Fees": [1.5, 2.5]}))

    def value(self):
        return self._value


def test_build_report_calculates_required_metrics_and_months():
    benchmark = pd.Series([100, 102, 101, 108], index=FakePortfolio().value().index)
    report = build_report(FakePortfolio(), benchmark, BacktestConfig())

    assert set(report.metrics) == {
        "CAGR",
        "MDD",
        "Sharpe Ratio",
        "Total Return",
        "Benchmark Return",
        "Trade Count",
        "Win Rate",
        "Fee Rate",
        "Total Fees",
        "Slippage Rate",
    }
    assert report.metrics["Total Return"] == pytest.approx(0.2)
    assert report.metrics["Benchmark Return"] == pytest.approx(0.08)
    assert report.metrics["MDD"] == pytest.approx(-0.1)
    assert report.metrics["Trade Count"] == 2
    assert report.metrics["Win Rate"] == pytest.approx(0.5)
    assert report.metrics["Total Fees"] == 4.0
    assert len(report.monthly_returns) == 2


def test_write_reports_contains_metrics_and_all_csv_sections(tmp_path):
    benchmark = pd.Series([100, 102, 101, 108], index=FakePortfolio().value().index)
    report = build_report(FakePortfolio(), benchmark, BacktestConfig())
    html_path, csv_path = write_reports(report, tmp_path / "report.html", tmp_path / "report.csv")

    html = html_path.read_text(encoding="utf-8")
    csv = pd.read_csv(csv_path)
    assert "CAGR" in html and "Asset Curve" in html and "Monthly Returns" in html
    assert set(csv["section"]) == {"summary", "monthly_return", "equity_curve"}
    assert {"Strategy", "Benchmark"} <= set(csv.loc[csv.section == "monthly_return", "metric"])

def test_run_baseline_applies_fees_and_slippage():
    index = pd.bdate_range("2024-01-01", periods=30)
    close = pd.Series([100 + index % 8 for index in range(30)], index=index)
    prices = pd.DataFrame({"Open": close, "Close": close, "Volume": 1000}, index=index)
    free, _ = run_baseline(
        prices,
        BacktestConfig(
            fast_window=2,
            slow_window=5,
            fee_rate=0,
            slippage_rate=0,
            minimum_average_trading_value=0,
            minimum_signal_trading_value=0,
        ),
    )
    costly, _ = run_baseline(
        prices,
        BacktestConfig(
            fast_window=2,
            slow_window=5,
            fee_rate=0.01,
            slippage_rate=0.01,
            minimum_average_trading_value=0,
            minimum_signal_trading_value=0,
        ),
    )

    assert costly.value().iloc[-1] < free.value().iloc[-1]


def test_liquidity_filter_uses_prior_close_observation_for_open_entry():
    index = pd.bdate_range("2024-01-01", periods=4)
    signals = pd.DataFrame({"entries": True, "exits": False}, index=index)
    prices = pd.DataFrame(
        {
            "Close": 100,
            "Volume": 1,
            "TradingValue": [10, 10, 1_000, 1_000],
        },
        index=index,
    )
    config = BacktestConfig(
        liquidity_window=2,
        minimum_average_trading_value=100,
        minimum_signal_trading_value=500,
    )

    filtered = apply_liquidity_filter(signals, prices, config)

    assert bool(filtered.loc[index[2], "liquidity_observed"])
    assert not bool(filtered.loc[index[2], "entries"])
    assert bool(filtered.loc[index[3], "entries"])


def test_relative_strength_filter_uses_prior_close_for_open_entry():
    index = pd.bdate_range("2024-01-01", periods=4)
    signals = pd.DataFrame({"entries": True, "exits": False}, index=index)
    close = pd.Series([100, 100, 120, 120], index=index)
    market = pd.Series(100, index=index)

    filtered = apply_rs_filter(signals, close, market, window=1, threshold=0.10)

    assert bool(filtered.loc[index[2], "relative_strength_observed"])
    assert not bool(filtered.loc[index[2], "entries"])
    assert bool(filtered.loc[index[3], "entries"])


def test_outlier_filter_includes_exact_threshold_and_blocks_following_sessions():
    index = pd.bdate_range("2024-01-01", periods=6)
    signals = pd.DataFrame({"entries": True, "exits": False}, index=index)
    close = pd.Series([100, 100, 50, 50, 50, 50], index=index)

    filtered = apply_outlier_filter(signals, close, threshold=0.50, cooldown=2)

    assert bool(filtered.loc[index[2], "outlier_observed"])
    assert bool(filtered.loc[index[2], "entries"])
    assert not bool(filtered.loc[index[3], "entries"])
    assert not bool(filtered.loc[index[4], "entries"])
    assert bool(filtered.loc[index[5], "entries"])


def test_market_trend_filter_uses_prior_close_for_entries_and_keeps_exits():
    index = pd.bdate_range("2024-01-01", periods=5)
    signals = pd.DataFrame(
        {"entries": [False, False, True, False, True], "exits": [False, False, False, True, False]},
        index=index,
    )
    market_close = pd.Series([100, 99, 100, 90, 100], index=index)

    filtered = apply_market_trend_filter(signals, market_close, window=2)

    assert not filtered.loc[index[2], "entries"]
    assert not filtered.loc[index[4], "entries"]
    assert filtered.loc[index[3], "exits"]
    assert not filtered.loc[index[0], "market_entry_allowed"]

def test_golden_cross_filter_uses_prior_close_and_keeps_exits():
    index = pd.bdate_range("2024-01-01", periods=6)
    signals = pd.DataFrame(
        {
            "entries": [False, False, False, True, True, False],
            "exits": [False, False, False, False, False, True],
        },
        index=index,
    )
    close = pd.Series([10, 9, 10, 11, 12, 13], index=index)

    filtered = apply_golden_cross_filter(signals, close, fast_window=2, slow_window=3)

    assert not filtered.loc[index[3], "entries"]
    assert filtered.loc[index[4], "entries"]
    assert filtered.loc[index[5], "exits"]
    assert filtered.loc[index[4], "golden_entry_allowed"]


def test_golden_cross_filter_can_exit_only_after_regime_loss():
    index = pd.bdate_range("2024-01-01", periods=7)
    signals = pd.DataFrame({"entries": False, "exits": False}, index=index)
    close = pd.Series([10, 11, 12, 13, 8, 7, 6], index=index)

    filtered = apply_golden_cross_filter(
        signals, close, fast_window=2, slow_window=3, exit_on_regime_loss=True
    )

    lost_regime = filtered.index[
        ~filtered["golden_cross_observed"] & filtered["golden_slow"].notna()
    ][0]
    assert not bool(filtered.loc[lost_regime, "exits"])
    assert filtered.loc[filtered.index[filtered.index.get_loc(lost_regime) + 1], "exits"]

def test_trade_blocks_require_the_price_limit_to_remain_locked_all_session():
    index = pd.bdate_range("2024-01-01", periods=5)
    prices = pd.DataFrame(
        {
            "Open": [100, 130, 130, 91, 91],
            "High": [100, 130, 135, 91, 100],
            "Low": [100, 130, 120, 91, 85],
            "Close": [100, 100, 130, 130, 100],
            "Volume": [100, 100, 100, 100, 100],
        },
        index=index,
    )
    cant_buy, cant_sell = build_trade_block_masks(prices, 0.30)
    assert cant_buy.tolist() == [False, True, False, False, False]
    assert cant_sell.tolist() == [False, False, False, True, False]


def test_trade_blocks_cancel_entries_and_delay_exits_until_trading_resumes():
    index = pd.bdate_range("2024-01-01", periods=5)
    entries = pd.Series([False, True, False, False, False], index=index)
    exits = pd.Series([False, False, True, False, False], index=index)
    cant_buy = pd.Series([False, True, False, False, False], index=index)
    cant_sell = pd.Series([False, False, True, True, False], index=index)
    new_entries, new_exits = apply_trade_blocks(entries, exits, cant_buy, cant_sell)
    assert not new_entries.any()
    assert new_exits.tolist() == [False, False, False, False, True]
    assert new_exits.sum() == 1


def test_run_signals_caps_entries_but_fully_closes_positions():
    index = pd.bdate_range("2024-01-01", periods=4)
    prices = pd.DataFrame(
        {
            "Open": [10.0, 10.0, 10.0, 10.0],
            "Close": [10.0, 10.0, 10.0, 10.0],
            "Volume": [100, 100, 100, 100],
            "TradingValue": [1_000.0, 1_000.0, 1_000.0, 1_000.0],
        },
        index=index,
    )
    signals = pd.DataFrame(
        {"entries": [False, True, False, False], "exits": [False, False, True, False]},
        index=index,
    )
    config = BacktestConfig(
        initial_cash=1_000, fee_rate=0, sell_tax_rate=0, slippage_rate=0,
        maximum_participation_rate=0.10, filter_halts=False,
    )
    portfolio = run_signals(prices, signals, config)
