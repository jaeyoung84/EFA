import os
import sqlite3
import warnings
from pathlib import Path
from data_loader import (
    load_ohlcv_from_sqlite,
    load_market_from_sqlite,

    load_universe_markets,            # V2-3
    check_data_coverage,
    SQLITE_MARKET_TICKER,
    SQLITE_KOSDAQ_MARKET_TICKER,      # V2-2
)
# F-5: 파일명과 일치하는 모듈명 사용 (v2: strategy_validation_v2)

# W-2 (D-38): research/daily_trend_pullback_t0/run.py → parents[2] = 저장소 루트 (D-37 방식, cwd 무관)
#   v0.4-rc1 원본은 "./data/krx_daily.sqlite3"(cwd 상대, 저장소 KrxDailyStore 기본값과 같은 규약)였음.
_REPO_ROOT    = Path(__file__).resolve().parents[2]
DB_PATH       = _REPO_ROOT / "data" / "krx_daily.sqlite3"
START_DATE    = "2022-01-01"
END_DATE      = "2026-09-08"
DB_START      = "2016-08-29"
CAPITAL       = 100_000_000
MARKET_TICKER = SQLITE_MARKET_TICKER   # "069500" (H-17)
KOSDAQ_MARKET_TICKER = SQLITE_KOSDAQ_MARKET_TICKER   # "229200" (V2-2)
UNIVERSE_EXCLUDE = (MARKET_TICKER, KOSDAQ_MARKET_TICKER)   # V2-2: 두 벤치마크 종목 모두 유니버스에서 제외

# W-2 (D-38): 읽기 전용 강제 — 이 프로세스가 여는 sqlite3 연결을 file:…?mode=ro URI로 바꾼다.
#   pullback_realdata_run.py(2026-09-11)·pullback_v04_repro.py(2026-09-12) 래퍼와 같은 방식(실행 2회로 검증).
#   data_loader는 str(db_path)로 연결하므로 Path/str 모두 처리하고, 이미 URI(file:)이면 그대로 둔다.
#   main() 진입 시에만 설치 — import만으로는 부작용 없음. 연구 스크립트 전용(라이브 코드는 이 모듈을 import하지 않음).
_orig_sqlite_connect = sqlite3.connect


def _readonly_sqlite_connect(database, *args, **kwargs):
    if isinstance(database, (str, os.PathLike)) and not str(database).startswith("file:"):
        kwargs["uri"] = True
        return _orig_sqlite_connect(f"file:{os.fspath(database)}?mode=ro", *args, **kwargs)
    return _orig_sqlite_connect(database, *args, **kwargs)


# V2-2·V2-3: 벤치마크 2개·유니버스 시장 매핑 — run.py 와 run_forward.py 가 공유 (import 부작용 없음)
def load_benchmarks(end_date: str = END_DATE) -> dict:
    """{"KOSPI": 069500 프레임, "KOSDAQ": 229200 프레임} — 각각 H-17 방식(부재 시 RuntimeError)."""
    return {
        "KOSPI":  load_market_from_sqlite(DB_PATH, index_ticker=MARKET_TICKER,
                                          start_date=DB_START, end_date=end_date),
        "KOSDAQ": load_market_from_sqlite(DB_PATH, index_ticker=KOSDAQ_MARKET_TICKER,
                                          start_date=DB_START, end_date=end_date),
    }


def load_universe_and_markets() -> tuple:
    """H-16 유니버스 + V2-3 시장 매핑. 두 집합이 다르면 RuntimeError(같은 모집단 규칙이어야 함)."""
    tickers    = load_universe_symbols(DB_PATH, exclude=UNIVERSE_EXCLUDE)
    market_map = load_universe_markets(DB_PATH, exclude=UNIVERSE_EXCLUDE)
    if set(tickers) != set(market_map):
        only_t = sorted(set(tickers) - set(market_map))[:5]
        only_m = sorted(set(market_map) - set(tickers))[:5]
        raise RuntimeError(f"[V2-3] 유니버스/시장 매핑 불일치: 유니버스만 {only_t} / 매핑만 {only_m}")
    return tickers, market_map


def main():
    sqlite3.connect = _readonly_sqlite_connect   # W-2: 읽기 전용 강제 (위 주석)

    # F-6: 폴백 없음 — 시장 시리즈 부재 시 명시적 RuntimeError (H-17: 로더가 직접 raise)
    # FDR 자동 폴백은 D-3(데이터 축 불일치) 재현 → 삭제
    print("STEP 1. 시장 시리즈 로딩 (날짜축의 단일 출처) — V2-2: 벤치마크 2개(069500·229200)")
    try:
        benchmarks = load_benchmarks(END_DATE)
        market_df  = benchmarks["KOSPI"]   # 날짜축 정렬(H-13)·g3 진단은 v1 과 같은 069500
    except RuntimeError as e:
        raise RuntimeError(
            f"시장 시리즈 로드 실패:\n{e}\n\n"
            f"해결 방법:\n"
            f"  1. DB 스키마 확인 후 index_ticker/table/ticker_col/date_col 인자 조정\n"
            f"  2. FDR 사용 시 D-3 데이터 축 불일치 위험 — 별도 승인 필요"
        ) from e

    print("STEP 2. 유니버스 (H-16) + 시장 매핑 (V2-3)")
    tickers, market_map = load_universe_and_markets()

    print("STEP 3. SQLite DB 로딩 (H-13: 시장 날짜축 정렬)")
    with warnings.catch_warnings():
        warnings.simplefilter("ignore")   # 종목별 비거래일 제거·검증 경고 다량 → 요약만
        ohlcv = load_ohlcv_from_sqlite(
            DB_PATH, tickers=tickers,
            start_date=DB_START, end_date=END_DATE,
            market_df=market_df
        )

    print("STEP 4. 데이터 정합성 점검")
    check_data_coverage(ohlcv, market_df, START_DATE, END_DATE)

    print("STEP 5. 전략 실행")
    equity, trades, fold_report = run_full_system(
        ohlcv, market_df,
        start_date=START_DATE,
        end_date=END_DATE,
        initial_capital=CAPITAL,
        benchmarks=benchmarks,        # V2-2
        market_map=market_map         # V2-3
    )

    return equity, trades, fold_report


if __name__ == "__main__":
    main()
