def make_ohlcv(
    n: int = 400,
    start: str = "2025-01-02",
    close_start: float = 10000.0,
    trend: float = 0.0003,
    seed: int = 42
) -> pd.DataFrame:
    rng   = np.random.default_rng(seed)
    dates = pd.bdate_range(start=start, periods=n, freq="B")
    close = close_start * np.cumprod(1 + trend + rng.normal(0, 0.015, n))
    # E-9a: 거래량 충분히 높여 min_turnover 확정 통과
    volume = rng.integers(500_000, 2_000_000, n).astype(int)

    df = pd.DataFrame({
        "Open":   close * rng.uniform(0.99, 1.01, n),
        "High":   close * rng.uniform(1.00, 1.03, n),
        "Low":    close * rng.uniform(0.97, 1.00, n),
        "Close":  close,
        "Volume": volume,
    }, index=dates)
    df["High"]     = df[["Open","High","Close"]].max(axis=1)
    df["Low"]      = df[["Open","Low","Close"]].min(axis=1)
    df["Turnover"] = df["Close"] * df["Volume"]
    return df


def make_trade(
    ticker: str = "TEST",
    entry_price: float = 10000.0,
    exit_price: float  = 10500.0,
    stop_price: float  = 9500.0,
    shares: int = 100,
    holding_days: int = 5,
    exit_reason: str = "trailing",        # V1-4: ema_cross 청산 없음
    entry_date: str = "2025-03-01",
    exit_date: str  = "2025-03-08"
) -> pd.Series:
    return pd.Series({
        "ticker":       ticker,
        "sector":       "Unknown",
        "entry_date":   pd.Timestamp(entry_date),
        "exit_date":    pd.Timestamp(exit_date),
        "entry_price":  entry_price,
        "exit_price":   exit_price,
        "stop_price":   stop_price,
        "shares":       shares,
        "entry_cost":   np.nan,
        "exit_cost":    np.nan,
        "pnl_net":      np.nan,
        "r_multiple":   np.nan,
        "exit_reason":  exit_reason,
        "holding_days": holding_days,
        "score":        0.5
    })


def make_trades_with_dates(
    n_win: int = 6,
    n_loss: int = 4,
    eq_index: pd.DatetimeIndex = None
) -> pd.DataFrame:
    if eq_index is None:
        eq_index = pd.bdate_range("2024-01-01", periods=400)

    n_total = n_win + n_loss
    step    = max(len(eq_index) // (n_total + 1), 1)
    dates   = [
        eq_index[min(i * step, len(eq_index) - 1)]
        for i in range(1, n_total + 1)
    ]

    rows = []
    for i in range(n_win):
        ep = 10000.0
        rows.append(make_trade(
            ticker=f"WIN{i:02d}",
            entry_price=ep, exit_price=ep * 1.08,
            stop_price=ep * 0.93, shares=100,
            holding_days=5, exit_reason="trailing",      # V1-4
            entry_date=str((dates[i] - pd.Timedelta(days=5)).date()),
            exit_date=str(dates[i].date())
        ))
    for i in range(n_loss):
        j  = n_win + i
        ep = 10000.0
        rows.append(make_trade(
            ticker=f"LOSS{i:02d}",
            entry_price=ep, exit_price=ep * 0.95,
            stop_price=ep * 0.93, shares=100,
            holding_days=3, exit_reason="stop",
            entry_date=str((dates[j] - pd.Timedelta(days=3)).date()),
            exit_date=str(dates[j].date())
        ))

    df = pd.DataFrame(rows, columns=TRADE_COLUMNS).reset_index(drop=True)
    return attach_costs(df)


def make_equity(
    n: int = 200,
    start: str = "2025-01-02",
    initial: float = 100_000_000.0,
    drift: float = 0.0005,
    seed: int = 42
) -> pd.DataFrame:
    rng   = np.random.default_rng(seed)
    dates = pd.bdate_range(start=start, periods=n)
    eq    = initial * np.cumprod(1 + drift + rng.normal(0, 0.008, n))
    return pd.DataFrame({
        "equity":    eq,
        "cash":      eq * 0.3,
        "positions": rng.integers(0, CONFIG["max_positions"] + 1, n)
    }, index=dates)


def make_system_ohlcv(n: int = 350, seed: int = 0) -> dict:
    """n=350: SMA50·ADX14 정상 계산 보장"""
    return {
        "AAA": make_ohlcv(n, seed=seed,   trend=0.003),
        "BBB": make_ohlcv(n, seed=seed+1, trend=0.002),
        "CCC": make_ohlcv(n, seed=seed+2, trend=0.001),
    }


def make_market(n: int = 350) -> pd.DataFrame:
    return make_ohlcv(n, close_start=2800.0, trend=0.0004, seed=99)


def _run_system(n=350, capital=100_000_000):
    ohlcv     = make_system_ohlcv(n)
    market_df = make_market(n)
    start     = list(ohlcv["AAA"].index)[0]
    end       = list(ohlcv["AAA"].index)[-1]
    return run_period(
        ohlcv, market_df, start, end, FIXED_CONFIG,
        initial_capital=capital,
        compute_pb_effectiveness=False
    )

class TestF1DelPositions:
    """
    F-1: pending_exits 적용 시 del positions[ticker] 없으면
         같은 포지션이 매일 재청산 → 현금 증식 → R3 거짓 통과.
    """

    def test_no_duplicate_entry_date_per_ticker(self):
        """
        F-1 직접 단언: (ticker, entry_date) 중복 = 같은 포지션 중복 청산.
        exit_date가 매일 달라 종전 중복 검사(entry+exit+shares)는 잡지 못함.
        validate_trade_frame에 추가된 검사가 이를 잡아야 함.
        """
        equity, trades, _ = _run_system()
        if trades.empty:
            pytest.skip("거래 없음")

        dup = trades.duplicated(subset=["ticker", "entry_date"])
        assert not dup.any(), \
            f"(ticker, entry_date) 중복 {dup.sum()}건 — F-1 del positions[ticker] 미적용"

    def test_cash_not_growing_unboundedly(self):
        """
        F-1 버그: 중복 청산 → 현금 무한 증식.
        equity 최대값이 initial_capital × 10 미만이어야 함.
        """
        capital = 100_000_000
        equity, _, _ = _run_system(capital=capital)
        max_eq = equity["equity"].max()
        assert max_eq < capital * 10, \
            f"equity 최대 {max_eq:,.0f} > {capital*10:,.0f} — 현금 증식 의심 (F-1)"

    def test_positions_not_stuck_at_max(self):
        """
        F-1 버그: positions가 제거되지 않아 slots=0 고착 → 신규 진입 불가.
        positions가 항상 max_positions에 있지 않아야 함.
        """
        equity, trades, _ = _run_system()
        if len(equity) < 20:
            pytest.skip("데이터 부족")

        pos_series = equity["positions"].iloc[:-1]
        max_pos    = CONFIG["max_positions"]
        stuck_rate = (pos_series == max_pos).mean()
        assert stuck_rate < 0.95, \
            f"positions가 {stuck_rate:.1%} 시간 max 고착 — F-1 del 미적용"

    def test_validate_catches_duplicate_entry_date(self):
        """validate_trade_frame이 (ticker, entry_date) 중복을 잡아야 함"""
        df = make_trades_with_dates(2, 0)
        # 의도적으로 같은 entry_date로 두 번째 행 변조
        df2 = df.copy()
        df2.loc[1, "entry_date"] = df2.loc[0, "entry_date"]
        df2.loc[1, "ticker"]     = df2.loc[0, "ticker"]
        # exit_date만 다르게 (기존 중복 검사 우회)
        df2.loc[1, "exit_date"]  = df2.loc[0, "exit_date"] + pd.Timedelta(days=1)

        ok, errs = validate_trade_frame(df2, raise_on_error=False)
        assert not ok
        assert any("entry_date" in e or "중복" in e for e in errs)

    def test_exit_proceeds_applied_next_day(self):
        """
        E-1·F-1: 청산 대금은 다음 반복 시작 시 반영.
        청산 결정일 equity에 t+1 시가 정보 없어야 함.
        """
        equity, trades, _ = _run_system()
        if trades.empty or len(equity) < 5:
            pytest.skip("데이터 부족")

        eq  = equity["equity"].iloc[:-1]
        pct = eq.pct_change().dropna()

        # 단일일 5% 초과 상승이 10% 이상이면 이상 (중복 청산 = 현금 급증)
        large_up = (pct > 0.05).mean()
        assert large_up < 0.10, \
            f"단일일 +5% 초과 비율 {large_up:.1%} — 청산 현금 선반영 의심"

# 2. F-3: forced_close holding_days 봉거리 검증
# ══════════════════════════════════════════════

class TestF3HoldingDays:
    """
    F-3: forced_close는 당일 종가 체결.
    봉거리 = exit_date_index - entry_date_index (익일 시가와 달리 -1).
    정상 청산: holding_days = i - entry_idx ✓ (불변)
    forced_close: holding_days = i - entry_idx - 1 ✓ (F-3)
    기간종료: holding_days = last_idx - entry_idx - 1 ✓ (F-3)
    """

    def test_forced_close_holding_days_gte_1(self):
        """
        G-3 수정: all_dates 합집합 축을 외부에서 재현할 수 없으므로
        봉거리 정확값 대신 >= 1 범위만 확인.
        (F-3 + E-3 + A-1 보장)
        """
        ohlcv     = make_system_ohlcv(350)
        market_df = make_market(350)
        start     = list(ohlcv["AAA"].index)[0]
        end       = list(ohlcv["AAA"].index)[-1]

        _, trades, _ = run_period(
            ohlcv, market_df, start, end, FIXED_CONFIG,
            compute_pb_effectiveness=False
        )

        forced = trades[trades["exit_reason"] == "forced_close"]
        if forced.empty:
            pytest.skip("forced_close 거래 없음")

        assert (forced["holding_days"] >= 1).all(), \
            f"forced_close holding_days < 1:\n{forced[['ticker','holding_days','entry_date','exit_date']]}"

    def test_normal_exit_holding_days_matches_union_index(self):
        """
        G-3: 정상 청산은 all_dates 합집합 봉거리와 일치.
        종목별 df.index 대신 전체 bdate_range 근사치 사용.
        """
        ohlcv     = make_system_ohlcv(350)
        market_df = make_market(350)
        start     = list(ohlcv["AAA"].index)[0]
        end       = list(ohlcv["AAA"].index)[-1]

        _, trades, _ = run_period(
            ohlcv, market_df, start, end, FIXED_CONFIG,
            compute_pb_effectiveness=False
        )

        normal = trades[trades["exit_reason"] != "forced_close"]
        if normal.empty:
            pytest.skip("정상 청산 없음")

        # holding_days >= 1 및 holding_limit 이하 (정지로 초과 가능)
        assert (normal["holding_days"] >= 1).all()
        # exit_date > entry_date (익일 체결)
        assert (normal["exit_date"] > normal["entry_date"]).all()

    def test_all_holding_days_gte_1(self):
        """E-3·A-1 보장 하에 모든 경로에서 holding_days >= 1"""
        _, trades, _ = _run_system()
        if not trades.empty:
            assert (trades["holding_days"] >= 1).all(), \
                f"holding_days < 1:\n{trades[trades['holding_days'] < 1]}"

    def test_holding_limit_gte_15(self):
        """E-9c: 보유한도 청산은 holding_days >= 15 (정지로 초과 가능)"""
        _, trades, _ = _run_system()
        limit = trades[trades["exit_reason"] == "holding_limit"]
        if not limit.empty:
            assert (limit["holding_days"] >= CONFIG["holding_limit"]).all()

class TestF4PositionsCount:
    """
    F-4: equity["positions"] = len(positions) (오늘 실보유).
    pending_exits도 오늘 보유 중 → 포함이 맞음.
    slots 계산은 내일 기준 (max - (len-pending_exits) - pending_entries).
    """

    def test_positions_within_max(self):
        """positions는 항상 [0, max_positions] 범위"""
        equity, _, _ = _run_system()
        assert (equity["positions"] >= 0).all()

        mismatches = 0
        for date in df.index:
            row    = df.loc[date]
            scalar = _entry_signal(row, FIXED_CONFIG, apply_pullback=False)
            vec    = bool(base.get(date, False))
            if scalar != vec:
                mismatches += 1

        assert mismatches == 0, \
            f"base 신호 벡터/스칼라 불일치 {mismatches}건"

class TestF7bReverseIndex:
    """F-7b: date→tickers 역인덱스 정합성"""

    def test_reverse_index_complete(self):
        """역인덱스의 (ticker, date) 집합이 entry_cache의 full 신호와 일치"""
        ohlcv    = make_system_ohlcv(350)
        prepared = {t: _compute_indicators(df) for t, df in ohlcv.items()}
        cache    = _compute_entry_signals(prepared, FIXED_CONFIG)

        # 역인덱스 직접 구성
        date_to_tickers = {}
        for ticker, sig in cache.items():
            for d in sig["full"].index[sig["full"]]:
                date_to_tickers.setdefault(d, []).append(ticker)

        # entry_cache의 (ticker, date) 집합
        expected = set()
        for ticker, sig in cache.items():
            for d in sig["full"].index[sig["full"]]:
                expected.add((ticker, d))

        # 역인덱스에서 재구성한 집합
        actual = set()
        for d, tickers in date_to_tickers.items():
            for ticker in tickers:
                actual.add((ticker, d))

        assert actual == expected, \
            f"역인덱스 누락 {len(expected - actual)}건, 초과 {len(actual - expected)}건"

class TestD1HoldingDays:

    def test_stop_fires_on_day_one_gap_scenario(self):
        """
        F-9c: `if holding_days < 1: continue` 삭제 확인.
        진입 익일 갭다운에서 손절(holding_days=1)이 가능해야 함.
        """
        n     = 350
        dates = pd.bdate_range("2025-01-02", periods=n)
        rng   = np.random.default_rng(7)
        close = 10000.0 * np.cumprod(1 + 0.002 + rng.normal(0, 0.01, n))
        vol   = np.full(n, 1_000_000)

        df = pd.DataFrame({
            "Open":    close * 0.99,
            "High":    close * 1.02,
            "Low":     close * 0.98,
            "Close":   close,
            "Volume":  vol,
            "Turnover": close * vol
        }, index=dates)
        df["High"] = df[["Open","High","Close"]].max(axis=1)
        df["Low"]  = df[["Open","Low","Close"]].min(axis=1)

        # 201봉부터 대형 갭다운 (stop_price 돌파)
        for j in range(200, 210):
            df.iloc[j, df.columns.get_loc("Open")]  = close[j] * 0.80
            df.iloc[j, df.columns.get_loc("Close")] = close[j] * 0.82
            df.iloc[j, df.columns.get_loc("Low")]   = close[j] * 0.79
            df.iloc[j, df.columns.get_loc("High")]  = close[j] * 0.83

        ohlcv     = {"GAPPER": df}
        market_df = make_market(n)
        start     = dates[0]
        end       = dates[-1]

        # RuntimeError 없이 실행 완료
        equity, trades, _ = run_period(
            ohlcv, market_df, start, end, FIXED_CONFIG,
            compute_pb_effectiveness=False
        )
        assert isinstance(trades, pd.DataFrame)
        if not trades.empty:
            assert (trades["holding_days"] >= 1).all()

            stop_day1 = trades[
                (trades["exit_reason"] == "stop")
                & (trades["holding_days"] == 1)
            ]
            # 생성됐다면 검증기를 통과해야 함
            if not stop_day1.empty:
                ok, _ = validate_trade_frame(stop_day1.reset_index(drop=True), raise_on_error=False)
                assert ok

def make_monotone_ohlcv(
    n: int = 300,
    start: str = "2025-01-02",
    close_start: float = 10000.0,
    step: float = 0.001,
    drop_idx=(),
) -> pd.DataFrame:
    """
    H-11: 결정적 시리즈. Close 가 매 봉 +0.1% 단조 증가, Open=0.999C, High=1.01C, Low=0.99C.
      - stop:      next_open ≥ Open[i+1] > entry_price > stop_price  → 발동 불가
      - ema_cross: 단조 증가 → Close > EMA10 항상               → 발동 불가
      - trailing:  next_open ≈ 0.999999·C_i > C_i − 2·ATR(≈0.04·C_i) → 발동 불가
      → 정상 청산은 holding_limit(15) 만 가능. 신호는 _inject_signals 로 주입한다.
    drop_idx: 삭제할 봉의 위치 인덱스(거래정지·상장폐지 시뮬레이션).
    """
    dates = pd.bdate_range(start=start, periods=n, freq="B")
    close = close_start * (1.0 + step) ** np.arange(n)
    df = pd.DataFrame({
        "Open":   close * 0.999,
        "High":   close * 1.01,
        "Low":    close * 0.99,
        "Close":  close,
        "Volume": np.full(n, 1_000_000, dtype=int),
    }, index=dates)
    df["Turnover"] = df["Close"] * df["Volume"]
    drop_idx = list(drop_idx)
    if drop_idx:
        df = df.drop(index=dates[drop_idx])
    return df


def make_monotone_market(n: int = 300) -> pd.DataFrame:
    """단조 증가 시장 시리즈 → SMA50 계산 이후 항상 Close > SMA50 (시장 필터 ON)"""
    return make_monotone_ohlcv(n, close_start=2800.0)


def _inject_signals(monkeypatch, signal_dates: dict):
    """
    H-11: run_period 가 모듈 전역으로 조회하는 _compute_entry_signals 를 결정적 가짜로 교체.
    signal_dates = {ticker: [Timestamp, ...]} 에 있는 날짜만 full/base True.
    """
    def fake(prepared, config):
        out = {}
        for t, df in prepared.items():
            s = pd.Series(False, index=df.index, dtype=bool)
            for d in signal_dates.get(t, []):
                if d in s.index:
                    s.loc[d] = True
            out[t] = {"base": s.copy(), "full": s}
        return out
    monkeypatch.setattr(strat, "_compute_entry_signals", fake)


def _union_axis(ohlcv: dict, start, end) -> list:
    """run_period 내부 all_dates(합집합 축) 재현 — _attach_warmup 은 기간 내 봉을 바꾸지 않는다"""
    return sorted(set(
        d for df in ohlcv.values() for d in df.index if start <= d <= end
    ))


def _assert_union_identity(trades: pd.DataFrame, axis: list):
    """
    H-1: 모든 청산 경로에서 holding_days == pos[exit_date] − pos[entry_date]
      정상 청산:  i − entry_idx,           exit=all_dates[i+1], entry=all_dates[entry_idx+1]
      forced_close(루프 내): i − entry_idx − 1, exit=all_dates[i]
      forced_close(기간종료): last_idx − entry_idx − 1, exit=all_dates[last_idx]
    """
    pos = {d: k for k, d in enumerate(axis)}
    for _, tr in trades.iterrows():
        assert tr["entry_date"] in pos and tr["exit_date"] in pos, \
            f"{tr['ticker']}: 진입/청산일이 합집합 축에 없음"
        expected = pos[tr["exit_date"]] - pos[tr["entry_date"]]
        assert tr["holding_days"] == expected, (
            f"{tr['ticker']} {tr['exit_reason']}: holding_days={tr['holding_days']} "
            f"≠ 축 위치차 {expected} ({tr['entry_date'].date()}→{tr['exit_date'].date()})"
        )


def _run_injected(monkeypatch, ohlcv: dict, signals: dict, n: int = 300):
    """단조 픽스처 + 주입 신호로 run_period 실행. 반환: equity, trades, axis"""
    market_df = make_monotone_market(n)
    axis_src  = pd.bdate_range(start="2025-01-02", periods=n, freq="B")
    start, end = axis_src[0], axis_src[-1]
    _inject_signals(monkeypatch, signals)
    equity, trades, _ = run_period(
        ohlcv, market_df, start, end, FIXED_CONFIG,
        compute_pb_effectiveness=False
    )
    return equity, trades, _union_axis(ohlcv, start, end)

              (b) 시장 Close > SMA50 이어야 한다 (신호일 t → 체결 t+1).

class TestH11InjectedExitPaths:
    """
    ANCHOR(전 봉 보유)로 합집합 축을 고정하고, 대상 종목의 봉을 삭제해 정지·상장폐지를 만든다.
    신호일 = axis[95] → 체결 axis[96] (entry_idx=95). 단조 시리즈라 정상 청산은 holding_limit 뿐.
    """
    SIG = 95

    def _axis(self, n=300):
        return pd.bdate_range(start="2025-01-02", periods=n, freq="B")

    def test_holding_limit_exact_15(self, monkeypatch):
        """기준 경로: 정지 없음 → holding_days == 15, exit = axis[111] 시가"""
        n, ax = 300, self._axis()
        ohlcv = {"ANCHOR": make_monotone_ohlcv(n), "T": make_monotone_ohlcv(n)}
        equity, trades, axis = _run_injected(monkeypatch, ohlcv, {"T": [ax[self.SIG]]})

        assert len(trades) == 1
        tr = trades.iloc[0]
        assert tr["ticker"] == "T"
        assert tr["entry_date"] == ax[self.SIG + 1]
        assert tr["exit_reason"] == "holding_limit"
        assert tr["holding_days"] == CONFIG["holding_limit"]
        assert tr["exit_date"] == ax[self.SIG + 1 + CONFIG["holding_limit"]]
        assert tr["exit_price"] == pytest.approx(ohlcv["T"].loc[tr["exit_date"], "Open"])
        _assert_union_identity(trades, axis)

    def test_stop_on_day_one_gap(self, monkeypatch):
        """H-11: 진입 익일 갭다운 → stop, holding_days == 1 (F-9c: '< 1 continue' 제거 확인)"""
        n, ax = 300, self._axis()
        df = make_monotone_ohlcv(n)
        c96 = df.loc[ax[self.SIG + 1], "Close"]
        gap = ax[self.SIG + 2]
        df.loc[gap, ["Open", "High", "Low", "Close"]] = [0.80 * c96, 0.83 * c96, 0.79 * c96, 0.82 * c96]
        ohlcv = {"ANCHOR": make_monotone_ohlcv(n), "GAPPER": df}

        assert tr["exit_reason"] == "stop"
        assert tr["holding_days"] == 1
        assert tr["exit_date"] == gap
        assert tr["exit_price"] == pytest.approx(0.80 * c96)
        assert tr["exit_price"] <= tr["stop_price"]
        ok, errs = validate_trade_frame(trades, raise_on_error=False)
        assert ok, errs
        _assert_union_identity(trades, axis)

    def test_h12_halt_while_holding_uses_last_close(self, monkeypatch):
        """
        H-12/E-2: 보유 중 axis[100..104] 정지 → 정지일 equity == cash + shares × Close[99],
        positions == 1 유지. 정지 후 정상 재개 → holding_limit 15 로 axis[111] 청산.
        """
        n, ax = 300, self._axis()
        halted = range(100, 105)
        ohlcv = {
            "ANCHOR": make_monotone_ohlcv(n),
            "HALTER": make_monotone_ohlcv(n, drop_idx=halted),
        }
        equity, trades, axis = _run_injected(monkeypatch, ohlcv, {"HALTER": [ax[self.SIG]]})

        assert len(trades) == 1

            assert d in equity.index
            assert equity.loc[d, "positions"] == 1

def _cand(ticker, dist_ema10, adx, rsi, **extra):
    """run_period 가 만드는 후보 dict 와 같은 키"""
    base = {
        "ticker": ticker, "dist_ema10": dist_ema10, "adx": adx, "rsi": rsi,
        "return20": 0.0, "dist_sma50": 0.0, "vol_ratio": 1.0,
        "atr": 100.0, "next_open": 10000.0,
    }
    base.update(extra)
    return base


def _goal_trades(n_win, n_loss, win_mult, loss_mult, eq_index):
    """승·패를 날짜 축에 교대로 배치(폴드마다 섞이도록). 비용 부착 완료 프레임 반환."""
    n, step = n_win + n_loss, max(len(eq_index) // (n_win + n_loss + 1), 1)
    kinds, w, l = [], 0, 0
    while w < n_win or l < n_loss:
        if w < n_win and (l >= n_loss or w * n_loss <= l * n_win):
            kinds.append("W"); w += 1
        else:
            kinds.append("L"); l += 1
    rows = []
    for i, k in enumerate(kinds):
        d  = eq_index[min((i + 1) * step, len(eq_index) - 1)]
        ep = 10000.0
        rows.append(make_trade(
            ticker=f"{k}{i:02d}",
            entry_price=ep, exit_price=ep * (win_mult if k == "W" else loss_mult),
            stop_price=ep * 0.93, shares=100,
            holding_days=5, exit_reason="trailing" if k == "W" else "stop",
            entry_date=str((d - pd.Timedelta(days=5)).date()),
            exit_date=str(d.date()),
        ))
    return attach_costs(pd.DataFrame(rows, columns=TRADE_COLUMNS).reset_index(drop=True))


class TestV1RankingDirection:
    """V1-3: 눌림목 방향 랭킹 — 앵커 근접(dist_ema10↓)·추세 강도(adx↑)·과열 회피(rsi↓). C-3 교정."""

    def test_weights_declared(self):
        assert set(RANK_WEIGHTS) == {"dist_ema10", "adx", "rsi"}
        assert sum(RANK_WEIGHTS.values()) == pytest.approx(1.0)
        assert all(w > 0 for w in RANK_WEIGHTS.values())

    def test_closest_to_ema10_ranks_first(self):
        cands = [_cand("FAR", 0.06, 30, 60), _cand("NEAR", 0.005, 30, 60), _cand("MID", 0.03, 30, 60)]
        assert list(_rank_candidates(cands)["ticker"]) == ["NEAR", "MID", "FAR"]

    def test_lower_rsi_preferred(self):
        cands = [_cand("HOT", 0.02, 30, 69), _cand("COOL", 0.02, 30, 51)]
        assert list(_rank_candidates(cands)["ticker"]) == ["COOL", "HOT"]

    def test_higher_adx_preferred(self):
        cands = [_cand("WEAK", 0.02, 21, 60), _cand("STRONG", 0.02, 45, 60)]
        assert list(_rank_candidates(cands)["ticker"]) == ["STRONG", "WEAK"]

    def test_expansion_metrics_do_not_affect_score(self):
        """v0.4-rc1 랭킹 입력(return20·dist_sma50·vol_ratio)은 v1 점수에 관여하지 않는다"""
        a = [_cand("A", 0.01, 30, 55), _cand("B", 0.03, 25, 65)]
        b = [_cand("A", 0.01, 30, 55, return20=0.9,  dist_sma50=0.5,  vol_ratio=9.0),
             _cand("B", 0.03, 25, 65, return20=-0.5, dist_sma50=-0.2, vol_ratio=0.1)]
        ra, rb = _rank_candidates(a), _rank_candidates(b)
        assert list(ra["ticker"]) == list(rb["ticker"])
        assert np.allclose(ra["score"].to_numpy(), rb["score"].to_numpy())

    def test_v04_top_expansion_is_not_bought_first(self):
        """v0.4-rc1 이면 1위였을 확장 종목(return20·vol_ratio 최상)이 v1 에서는 앵커 근접 종목에 밀린다"""
        cands = [_cand("EXPAND", 0.07, 35, 68, return20=0.40, dist_sma50=0.30, vol_ratio=4.0),
                 _cand("PULL",   0.004, 35, 53, return20=0.05, dist_sma50=0.05, vol_ratio=1.0)]
        assert _rank_candidates(cands)["ticker"].iloc[0] == "PULL"

    def test_score_formula_matches_declared_weights(self):
        cands  = [_cand("A", 0.01, 40, 52), _cand("B", 0.02, 30, 60), _cand("C", 0.03, 20, 68)]
        ranked = _rank_candidates(cands)
        df     = pd.DataFrame(cands)
        expected = (
            RANK_WEIGHTS["dist_ema10"] * (1 - df["dist_ema10"].rank(pct=True))
            + RANK_WEIGHTS["adx"]      * df["adx"].rank(pct=True)
            + RANK_WEIGHTS["rsi"]      * (1 - df["rsi"].rank(pct=True))
        ).sort_values(ascending=False)
        assert list(ranked["ticker"]) == [df.loc[i, "ticker"] for i in expected.index]
        assert np.allclose(ranked["score"].to_numpy(), expected.to_numpy())

    def test_empty_and_columns(self):
        assert _rank_candidates([]).empty
        ranked = _rank_candidates([_cand("A", 0.01, 30, 55)])
        for col in ["dist_ema10_rank", "adx_rank", "rsi_rank", "score", "ticker", "atr", "next_open"]:
            assert col in ranked.columns

    def test_dist_ema10_indicator_present(self):
        """V1-2: _compute_indicators 가 DIST_EMA10 = Close/EMA10 − 1 을 만든다"""
        df = _compute_indicators(make_ohlcv(120, seed=3))
        assert "DIST_EMA10" in df.columns
        ok = df["EMA10"].notna()
        assert np.allclose(df.loc[ok, "DIST_EMA10"], df.loc[ok, "Close"] / df.loc[ok, "EMA10"] - 1)

        hit, reason = _exit_signal(self._row(), self._pos(), CONFIG["holding_limit"], 8999.0, FIXED_CONFIG)

def _mk_on(n=300):
    """단조 증가 벤치마크 → SMA50 유효 이후 항상 ON"""
    return make_monotone_ohlcv(n, close_start=2800.0)


def _mk_off(n=300):
    """단조 감소 벤치마크 → Close < SMA50 항상 → OFF (SMA NaN 구간도 OFF)"""
    return make_monotone_ohlcv(n, close_start=2800.0, step=-0.001)


def _run_gated(monkeypatch, ohlcv, signals, benchmarks, market_map, n=300):
    """주입 신호 + 시장별 게이트로 run_period 실행. market_df 위치 인자는 benchmarks["KOSPI"]."""
    axis_src = pd.bdate_range(start="2025-01-02", periods=n, freq="B")
    start, end = axis_src[0], axis_src[-1]
    _inject_signals(monkeypatch, signals)
    equity, trades, _ = run_period(
        ohlcv, benchmarks["KOSPI"], start, end, FIXED_CONFIG,
        compute_pb_effectiveness=False, benchmarks=benchmarks, market_map=market_map,
    )
    return equity, trades, _union_axis(ohlcv, start, end)

def _fx_insert(path, symbol, market, dates):
    conn = sqlite3.connect(path)
    rows = []
    for k, d in enumerate(dates):
        close, volume = 10000.0 + 10.0 * k, 1000 + k
        rows.append((
            d.strftime("%Y-%m-%d"), symbol, f"NAME_{symbol}", market, "",
            close - 5.0, close + 20.0, close - 20.0, close, volume,
            close * volume * 3.0, 1.0e12, 1_000_000, "2026-09-12T00:00:00",
        ))
    conn.executemany("INSERT INTO daily_prices VALUES (" + ",".join("?" * 14) + ")", rows)
    conn.commit()
    conn.close()

_V2_REF_SHA16 = "73f6af2f238d2c7d"     # 기반 v2 strategy_validation_v2.py — 커밋 2bfed1d 해시 잠금(PREREGISTRATION_v2.md §7)


def _load_v2_ref():
    """
    기반 v2 모듈을 경로로 적재 — sys.modules 미등록·바이트코드 미기록(v2 디렉터리 무접촉).
    v2 는 로컬 import 가 없어 이름 충돌 없음. 파일 부재·해시 불일치면 skip(잠금 표와 다른 v2 로는 동치 검사 무의미).
    """
    import hashlib, importlib.util, sys
    from pathlib import Path
    path = Path(__file__).resolve().parent.parent / "daily_trend_pullback_v2" / "strategy_validation_v2.py"
    if not path.exists():
        pytest.skip("v2 기준 모듈 없음")
    if not hashlib.sha256(path.read_bytes()).hexdigest().startswith(_V2_REF_SHA16):
        pytest.skip("v2 기준 모듈 해시가 잠금 표와 다름")
    prev = sys.dont_write_bytecode
    sys.dont_write_bytecode = True
    try:
        spec = importlib.util.spec_from_file_location("_v2_ref_strategy", path)
        mod  = importlib.util.module_from_spec(spec)
        spec.loader.exec_module(mod)
    finally:
        sys.dont_write_bytecode = prev
    return mod


def _inject_both(monkeypatch, v2mod, signal_dates: dict):
    """H-11 가짜 신호를 v4(strat)·v2 기준 모듈 양쪽에 주입"""
    def fake(prepared, config):
        out = {}
        for t, df in prepared.items():
            s = pd.Series(False, index=df.index, dtype=bool)
            for d in signal_dates.get(t, []):
                if d in s.index:
                    s.loc[d] = True
            out[t] = {"base": s.copy(), "full": s}
        return out
    monkeypatch.setattr(strat, "_compute_entry_signals", fake)
    monkeypatch.setattr(v2mod, "_compute_entry_signals", fake)


def _mk_dip(n=300, dip_idx=98, factor=0.5):
    """
    단조 증가 벤치마크에 하루 급락(Close×factor) 주입 → 그날만 Close ≤ SMA50(국면 이탈), 다음 날 Close 가 SMA50 위로 복귀.
    진입 게이트(_gate_ok)도 그날만 OFF 이지만 신호일(SIG=95)·진입일(96) 과 겹치지 않는다(dip_idx ≥ 96).
    """
    df = _mk_on(n).copy()
    df.iloc[dip_idx, df.columns.get_loc("Close")] *= factor
    return df


def _run_gated_cfg(monkeypatch, ohlcv, signals, benchmarks, market_map, config, n=300):
    """_run_gated 와 동일하되 config 를 지정(regime_exit=False 대조용)"""
    axis_src = pd.bdate_range(start="2025-01-02", periods=n, freq="B")
    start, end = axis_src[0], axis_src[-1]
    _inject_signals(monkeypatch, signals)
    equity, trades, _ = run_period(
        ohlcv, benchmarks["KOSPI"], start, end, config,
        compute_pb_effectiveness=False, benchmarks=benchmarks, market_map=market_map,
    )
    return equity, trades, _union_axis(ohlcv, start, end)

class TestV4RegimeOff:
    """V4-3: _regime_off 의미 — 행 존재 ∧ SMA 유효 ∧ Close ≤ SMA 만 True; 결측/NaN 은 False(보유)."""

    @staticmethod
    def _gf(close, sma):
        idx = pd.bdate_range("2025-01-02", periods=len(close), freq="B")
        return pd.DataFrame({"Close": close, "SMA50": sma}, index=idx), idx

    def test_below_and_equal_are_off(self):
        gf, idx = self._gf([99.0, 100.0], [100.0, 100.0])
        assert _regime_off(gf, idx[0], "SMA50") is True      # Close < SMA
        assert _regime_off(gf, idx[1], "SMA50") is True      # Close == SMA (≤)

    def test_above_is_hold(self):
        gf, idx = self._gf([100.01], [100.0])
        assert _regime_off(gf, idx[0], "SMA50") is False

    def test_nan_sma_is_hold(self):
        gf, idx = self._gf([50.0], [np.nan])
        assert _regime_off(gf, idx[0], "SMA50") is False

    def test_missing_date_is_hold(self):
        gf, idx = self._gf([50.0], [100.0])
        assert _regime_off(gf, idx[0] + pd.Timedelta(days=1), "SMA50") is False

    def test_asymmetry_with_gate_ok_on_nan(self):
        """결측/NaN: 진입 게이트는 차단(False), 국면 이탈 청산은 미발동(False) — 같은 False 지만 뜻이 다름을 고정"""
        gf, idx = self._gf([50.0], [np.nan])
        assert _gate_ok(gf, idx[0], "SMA50") is False and _regime_off(gf, idx[0], "SMA50") is False

    def test_reads_same_frame_as_gate(self):
        """_gate_frame 산출 프레임에서 _gate_ok 와 _regime_off 는 SMA 유효 구간에서 정확히 상보(ON ⇔ not OFF)"""
        n = 120
        bench = _mk_dip(n, 100)
        ax = pd.bdate_range(start="2025-01-02", periods=n, freq="B")
        gf = _gate_frame(bench, ax[0], ax[-1], list(ax), "SMA50", "KOSPI")
        for d in ax:
            if d in gf.index and not np.isnan(gf.loc[d, "SMA50"]):
                assert _gate_ok(gf, d, "SMA50") == (not _regime_off(gf, d, "SMA50"))
        assert _regime_off(gf, ax[100], "SMA50") is True and _regime_off(gf, ax[101], "SMA50") is False


class TestV4ExitSignalPriority:
    """V4-3: stop > holding_limit > trailing > market_regime. regime_exit=False 면 v2 와 동일."""

    @staticmethod
    def _row(close=9800.0, ema10=9900.0, atr=100.0):
        return pd.Series({"Close": close, "EMA10": ema10, "ATR": atr})

    @staticmethod
    def _pos(stop_price=9000.0, highest_close=10000.0):
        return {"stop_price": stop_price, "highest_close": highest_close}

    def test_regime_alone_fires_with_own_label(self):
        # stop(9000)·limit(15)·trailing(9800) 미충족, next_open 9850 → 국면 이탈만
        hit, reason = _exit_signal(self._row(), self._pos(), 3, 9850.0, FIXED_CONFIG, regime_off=True)
        assert (hit, reason) == (True, "market_regime")

    def test_default_is_off(self):
        assert _exit_signal(self._row(), self._pos(), 3, 9850.0, FIXED_CONFIG) == (False, "")

    def test_stop_beats_regime(self):
        assert _exit_signal(self._row(), self._pos(), 3, 8999.0, FIXED_CONFIG, regime_off=True) == (True, "stop")

    def test_holding_limit_beats_regime(self):
        assert _exit_signal(self._row(), self._pos(), CONFIG["holding_limit"], 9850.0, FIXED_CONFIG,
                            regime_off=True) == (True, "holding_limit")

    def test_trailing_beats_regime(self):
        assert _exit_signal(self._row(), self._pos(), 3, 9799.0, FIXED_CONFIG, regime_off=True) == (True, "trailing")

    def test_disabled_config_ignores_flag(self):
        cfg = {**FIXED_CONFIG, "regime_exit": False}
        assert _exit_signal(self._row(), self._pos(), 3, 9850.0, cfg, regime_off=True) == (False, "")

    def test_numpy_bool_flag_accepted(self):
        hit, reason = _exit_signal(self._row(), self._pos(), 3, 9850.0, FIXED_CONFIG, regime_off=np.bool_(True))
        assert hit is True and reason == "market_regime"


class TestV4RegimeExitPerMarket:
    """주입 신호(H-11)·급락 벤치마크: ANCHOR + KOSPI 종목 K + KOSDAQ 종목 Q, 같은 날(SIG=95) 신호 → 96 진입."""
    SIG = 95

    def _ohlcv(self, n=300):
        return {"ANCHOR": make_monotone_ohlcv(n), "K": make_monotone_ohlcv(n), "Q": make_monotone_ohlcv(n)}

    def _map(self, q="KOSDAQ"):
        return {"ANCHOR": "KOSPI", "K": "KOSPI", "Q": q}

    def _signals(self, ax):
        return {"K": [ax[self.SIG]], "Q": [ax[self.SIG]]}

    def test_kospi_dip_exits_only_kospi_ticker(self, monkeypatch):
        n = 300
        ax = pd.bdate_range(start="2025-01-02", periods=n, freq="B")
        ohlcv = self._ohlcv(n)
        _, trades, axis = _run_gated(monkeypatch, ohlcv, self._signals(ax),
                                     {"KOSPI": _mk_dip(n, 98), "KOSDAQ": _mk_on(n)}, self._map())
        by = trades.set_index("ticker")
        assert sorted(by.index) == ["K", "Q"]
        assert (trades["entry_date"] == ax[self.SIG + 1]).all()
        assert by.loc["K", "exit_reason"] == "market_regime"
        assert by.loc["K", "exit_date"] == ax[99] and by.loc["K", "holding_days"] == 3       # 98 종가 판정 → 99 시가
        assert by.loc["K", "exit_price"] == ohlcv["K"].loc[ax[99], "Open"]                    # 체결가 = t+1 시가
        assert by.loc["Q", "exit_reason"] == "holding_limit"
        _assert_union_identity(trades, axis)

    def test_kosdaq_dip_exits_only_kosdaq_ticker(self, monkeypatch):
        n = 300
        ax = pd.bdate_range(start="2025-01-02", periods=n, freq="B")
        _, trades, axis = _run_gated(monkeypatch, self._ohlcv(n), self._signals(ax),
                                     {"KOSPI": _mk_on(n), "KOSDAQ": _mk_dip(n, 98)}, self._map())
        by = trades.set_index("ticker")
        assert by.loc["Q", "exit_reason"] == "market_regime" and by.loc["Q", "exit_date"] == ax[99]
        assert by.loc["K", "exit_reason"] == "holding_limit"
        _assert_union_identity(trades, axis)

    def test_dip_on_entry_day_gives_holding_days_one(self, monkeypatch):
        """최소 보유일 = 1: 진입일(96) 종가에 이탈 판정 → 97 시가 청산 (regime_exit_counts 의 hd1 이 세는 경우)"""
        n = 300
        ax = pd.bdate_range(start="2025-01-02", periods=n, freq="B")
        _, trades, axis = _run_gated(monkeypatch, self._ohlcv(n), self._signals(ax),
                                     {"KOSPI": _mk_dip(n, 96), "KOSDAQ": _mk_on(n)}, self._map())
        by = trades.set_index("ticker")
        assert by.loc["K", "exit_reason"] == "market_regime"
        assert by.loc["K", "entry_date"] == ax[96] and by.loc["K", "exit_date"] == ax[97]
        assert by.loc["K", "holding_days"] == 1
        _assert_union_identity(trades, axis)

    def test_missing_benchmark_row_holds(self, monkeypatch):
        """벤치마크 행 결측(커버리지 299/300 ≥ 0.95) → 그날 판정 없음(보유) → 정상 holding_limit"""
        n = 300
        ax = pd.bdate_range(start="2025-01-02", periods=n, freq="B")

                                  {"KOSPI": _mk_on(n).drop(index=ax[98]), "KOSDAQ": _mk_on(n)}, self._map())

    def test_regime_exit_false_restores_holding_limit(self, monkeypatch):
        """같은 급락 픽스처에서 regime_exit=False 면 market_regime 0건(v2 동작)"""

        ax = pd.bdate_range(start="2025-01-02", periods=n, freq="B")
        _, trades, _ = _run_gated_cfg(monkeypatch, self._ohlcv(n), self._signals(ax),
                                      {"KOSPI": _mk_dip(n, 98), "KOSDAQ": _mk_on(n)}, self._map(),
                                      {**FIXED_CONFIG, "regime_exit": False})
        assert set(trades["exit_reason"]) == {"holding_limit"} and len(trades) == 2

    def test_transfer_symbol_uses_market_at_decision_date(self, monkeypatch):
        """V2-3 규칙 재사용: Q 가 KOSDAQ→KOSPI 이전. 전환일 ≤ 판정일(98)이면 KOSPI 급락으로 청산, 이후면 KOSDAQ(무급락) 보유"""

        seg0 = (ax[0].strftime("%Y-%m-%d"), "KOSDAQ")
        before = {**self._map(), "Q": [seg0, (ax[97].strftime("%Y-%m-%d"), "KOSPI")]}
        _, t_before, _ = _run_gated(monkeypatch, self._ohlcv(n), self._signals(ax), bench, before)
        assert set(t_before["exit_reason"]) == {"market_regime"} and sorted(t_before["ticker"]) == ["K", "Q"]
        after = {**self._map(), "Q": [seg0, (ax[99].strftime("%Y-%m-%d"), "KOSPI")]}

        _inject_signals(monkeypatch, self._signals(ax))
        _, trades, _ = run_period(ohlcv, _mk_dip(n, 98), ax[0], ax[-1], FIXED_CONFIG, compute_pb_effectiveness=False)

        assert set(trades["exit_reason"]) == {"market_regime"} and (trades["exit_date"] == ax[99]).all()
        _assert_union_identity(trades, _union_axis(ohlcv, ax[0], ax[-1]))

    def test_regime_exit_counts_diagnostic(self, monkeypatch):
        """V4-4: 건수만 — total / 시장별(청산일 기준 시장) / hd1 / other_exits"""
        n = 300
        ax = pd.bdate_range(start="2025-01-02", periods=n, freq="B")
        mmap = self._map()
        _, t98, _ = _run_gated(monkeypatch, self._ohlcv(n), self._signals(ax),
                               {"KOSPI": _mk_dip(n, 98), "KOSDAQ": _mk_on(n)}, mmap)
        assert regime_exit_counts(t98, mmap) == {"market_regime_total": 1, "market_regime_KOSPI": 1,
                                                 "market_regime_hd1": 0, "other_exits": 1}
        assert "market_regime_KOSPI" not in regime_exit_counts(t98)                      # market_map 없으면 시장별 생략
        _, t96, _ = _run_gated(monkeypatch, self._ohlcv(n), self._signals(ax),
                               {"KOSPI": _mk_dip(n, 96), "KOSDAQ": _mk_dip(n, 96)}, mmap)
        assert regime_exit_counts(t96, mmap) == {"market_regime_total": 2, "market_regime_KOSDAQ": 1,
                                                 "market_regime_KOSPI": 1, "market_regime_hd1": 2, "other_exits": 0}
        assert regime_exit_counts(t98.iloc[0:0], mmap) == {"market_regime_total": 0, "market_regime_hd1": 0, "other_exits": 0}
        assert regime_exit_counts(None) == {"market_regime_total": 0, "market_regime_hd1": 0, "other_exits": 0}


class TestV4EquivalenceWithV2:
    """regime_exit=False 인 v4 ≡ 해시 잠금 v2 모듈(경로 적재) — 진입·게이트·랭킹·비용·자금관리 무변경의 직접 증거 + 양성 대조."""
    SIG = 95

    def _fixture(self, n=300):
        ax = pd.bdate_range(start="2025-01-02", periods=n, freq="B")
        ohlcv = {"ANCHOR": make_monotone_ohlcv(n), "K": make_monotone_ohlcv(n), "Q": make_monotone_ohlcv(n)}
        bench = {"KOSPI": _mk_dip(n, 98), "KOSDAQ": _mk_on(n)}
        mmap  = {"ANCHOR": "KOSPI", "K": "KOSPI", "Q": "KOSDAQ"}
        return ax, ohlcv, bench, mmap, {"K": [ax[self.SIG]], "Q": [ax[self.SIG]]}

    def test_config_differs_only_by_regime_exit_and_observations(self):
        v2 = _load_v2_ref()

        assert set(CONFIG) - set(v2.CONFIG) == {"regime_exit"}
        diff = {k for k in v2.CONFIG if CONFIG[k] != v2.CONFIG[k]}
        assert diff == {"declared_observations"} and v2.CONFIG["declared_observations"] == 3
        assert set(VALID_EXIT_REASONS) - set(v2.VALID_EXIT_REASONS) == {"market_regime"}

    def test_injected_signals_equal_when_disabled(self, monkeypatch):
        v2 = _load_v2_ref()

        pd.testing.assert_frame_equal(t4, t2)

    def test_injected_signals_differ_when_enabled(self, monkeypatch):
        """양성 대조: 같은 픽스처에서 regime_exit=True 면 v2 와 달라야 한다(검사 자체가 살아 있음)"""

        _inject_both(monkeypatch, v2, sig)
        _, t4, _ = run_period(ohlcv, bench["KOSPI"], ax[0], ax[-1], FIXED_CONFIG, compute_pb_effectiveness=False,
                              benchmarks=bench, market_map=mmap)
        _, t2, _ = v2.run_period(ohlcv, bench["KOSPI"], ax[0], ax[-1], v2.FIXED_CONFIG, compute_pb_effectiveness=False,
                                 benchmarks=bench, market_map=mmap)
