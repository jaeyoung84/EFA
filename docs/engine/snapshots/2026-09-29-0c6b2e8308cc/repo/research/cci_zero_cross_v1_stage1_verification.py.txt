from __future__ import annotations

import sqlite3
import sys
from pathlib import Path

import numpy as np
import pandas as pd
from numpy.lib.stride_tricks import sliding_window_view

REPO = Path(__file__).resolve().parent.parent
sys.path.insert(0, str(REPO / "src"))

from korea_quant.eligibility import entry_eligibility  # noqa: E402
from korea_quant.krx import _fill_halted_trading_days  # noqa: E402
from korea_quant.paper_trading import (  # noqa: E402
    PaperCostConfig,
    _first_observed_row_after,
    _first_row_after,
    _prepared_history,
)
from korea_quant.reversion_validation import (  # noqa: E402
    ReversionValidationConfig,
    _label_windows,
)
from korea_quant.strategy_goals import ReversionGoal, TrendGoal, trade_metrics  # noqa: E402

DB = REPO / "data" / "krx_daily.sqlite3"


def _load_data() -> pd.DataFrame:
    con = sqlite3.connect(DB)
    try:
        raw = pd.read_sql_query(
            "SELECT trade_date,symbol,name,market,open,high,low,close,volume,"
            "trading_value,market_cap FROM daily_prices WHERE market IN ('KOSPI','KOSDAQ') "
            "ORDER BY symbol,trade_date",
            con,
        )
    finally:
        con.close()
    raw = raw.rename(columns={
        "trade_date": "Date", "symbol": "Symbol", "name": "Name", "market": "Market",
        "open": "Open", "high": "High", "low": "Low", "close": "Close",
        "volume": "Volume", "trading_value": "TradingValue", "market_cap": "MarketCap",
    })
    raw["Date"] = pd.to_datetime(raw["Date"])
    raw["Symbol"] = raw["Symbol"].astype(str)
    for c in ("Open", "High", "Low", "Close"):
        raw[c] = pd.to_numeric(raw[c], errors="coerce")

    filled = _fill_halted_trading_days(raw.copy())
    data = entry_eligibility(
        filled, market_cap_tiers=(("KOSPI", 200), ("KOSDAQ", 150)), liquidity_window=20,
        minimum_average_trading_value=5_000_000_000, minimum_history_sessions=260,
        outlier_gap_threshold=0.50, outlier_cooldown_sessions=60,
    )
    return data.sort_values(["Symbol", "Date"]).reset_index(drop=True)


def _add_cci(data: pd.DataFrame, window: int = 20) -> pd.DataFrame:
    tp = ((data["High"] + data["Low"] + data["Close"]) / 3.0).to_numpy(float)
    win = sliding_window_view(tp, window)
    ma_w = win.mean(axis=1)
    mad_w = np.abs(win - ma_w[:, None]).mean(axis=1)
    cci = np.full(len(tp), np.nan)
    with np.errstate(divide="ignore", invalid="ignore"):
        cci[window - 1:] = (tp[window - 1:] - ma_w) / (0.015 * mad_w)
    cci[data.groupby("Symbol").cumcount().to_numpy() < window - 1] = np.nan
    data = data.copy()
    data["CCI"] = cci
    prev = data.groupby("Symbol")["CCI"].shift(1)
    data["CrossUp"] = ((prev <= 0) & (data["CCI"] > 0)).fillna(False)
    data["CrossDown"] = ((prev >= 0) & (data["CCI"] < 0)).fillna(False)
    return data


def _build_signal_trades(data: pd.DataFrame, sim_start: pd.Timestamp) -> pd.DataFrame:
    trades = []
    sig = data[(data["CrossUp"] | data["CrossDown"]) & (data["Date"] >= sim_start)]
    for sym, g in sig.groupby("Symbol", sort=False):
        g = g.sort_values("Date")
        awaiting_exit = False
        entry_date = None
        for _, row in g.iterrows():
            if row["CrossUp"]:
                if awaiting_exit or not bool(row["EntryAllowed"]):
                    continue
                entry_date = row["Date"]
                awaiting_exit = True
            elif row["CrossDown"] and awaiting_exit:
                trades.append({
                    "Symbol": sym, "EntrySignalDate": entry_date, "ExitSignalDate": row["Date"],
                })
                awaiting_exit = False
                entry_date = None
    return pd.DataFrame(trades)


def _fill_trades(data: pd.DataFrame, trades_df: pd.DataFrame, costs: PaperCostConfig) -> pd.DataFrame:
    prepared = _prepared_history(data.copy(), costs.price_limit_ratio)
    prepared["Idx"] = prepared.groupby("Symbol").cumcount()
    group_dict = {sym: g for sym, g in prepared.groupby("Symbol", sort=False)}

    results = []
    for t in trades_df.itertuples(index=False):
        rows = group_dict.get(t.Symbol)
        if rows is None:
            continue
        next_session = _first_observed_row_after(rows, t.EntrySignalDate)
        if next_session is None:
            continue
        fillable = (
            pd.notna(next_session["Open"]) and float(next_session["Open"]) > 0
            and not bool(next_session["cant_buy"])
        )
        if not fillable:
            continue
        entry_date = next_session["Date"]
        entry_price = float(next_session["Open"])

        exit_fill = _first_row_after(rows, t.ExitSignalDate, "cant_sell")
        if exit_fill is None:
            continue

        results.append({
            "Symbol": t.Symbol, "Entry Timestamp": entry_date, "EntryPrice": entry_price,
            "ExitDate": exit_fill["Date"], "ExitPrice": float(exit_fill["Open"]),
        })

    res = pd.DataFrame(results)
    entry_cash = res["EntryPrice"] * (1 + costs.slippage_rate) * (1 + costs.fee_rate)
    exit_cash = res["ExitPrice"] * (1 - costs.slippage_rate) * (1 - costs.fee_rate - costs.sell_tax_rate)
    res["NetReturnPct"] = (exit_cash / entry_cash - 1) * 100
    res["Status"] = "closed"
    return res


def main() -> None:
    data = _add_cci(_load_data())
    sim_start = pd.Timestamp("2018-01-01")
    trades_df = _build_signal_trades(data, sim_start)
    costs = PaperCostConfig()
    res = _fill_trades(data, trades_df, costs)

    config = ReversionValidationConfig()
    dates = pd.DatetimeIndex(data["Date"].drop_duplicates().sort_values())
    labelled = _label_windows(res, dates, config)
    oos = labelled[labelled["Window"].notna()].copy()

    metrics = trade_metrics(oos, return_column="NetReturnPct", window_column="Window")

    print("=== cci-zero-cross-v1 Stage 1 재현 검증 (D-37) ===")
    print(f"완결거래 {metrics.closed_trades}건, 윈도우 {metrics.window_count}개")
    print(f"승률 {metrics.win_rate*100:.2f}%  PF {metrics.profit_factor:.3f}  "
          f"중앙값 {metrics.median_net_return:.3f}%  평균 {metrics.mean_net_return:.3f}%  "
          f"페이오프비 {metrics.payoff_ratio:.3f}  이익윈도우비율 {metrics.profitable_window_share*100:.2f}%")

    immediate_reject = metrics.mean_net_return <= 0.0
    print(f"\n[사전합의 즉시기각선: 비용후 평균수익 <= 0] "
          f"{'FAIL(기각)' if immediate_reject else 'PASS(통과, 추가 기준으로 판정 계속)'}")

    print("\n[참고 진단: ReversionGoal (D-7, 평균회귀 계열 기준 -- 최종 판정 아님)]")
    rev_goal = ReversionGoal()
    print("  " + ("PASS" if rev_goal.passes(metrics, rev_goal.minimum_long_term_trades) else "FAIL"))

    print("\n[공식 판정: TrendGoal (추세추종 계열 기준 -- CCI 손익구조에 맞는 프레임)]")
    trend_goal = TrendGoal()
    trend_pass = trend_goal.passes(metrics)
    print("  " + ("PASS" if trend_pass else "FAIL"))
    print(f"  (payoff>={trend_goal.minimum_payoff_ratio}: {metrics.payoff_ratio:.3f}, "
          f"PF>={trend_goal.minimum_profit_factor}: {metrics.profit_factor:.3f}, "
          f"expectancy>0: {metrics.expectancy:.3f}%, "
          f"window_share>={trend_goal.minimum_profitable_window_share}: "
          f"{metrics.profitable_window_share:.3f}, "
          f"trades>={trend_goal.minimum_trades}: {metrics.closed_trades})")

    print("\n=== 결론: Stage 1 승격", "PASS" if trend_pass else "FAIL, 연구 중단 (D-37)", "===")


if __name__ == "__main__":
    main()
