def make_ohlcv(
    n: int = 400,
    start: str = "2025-01-02",
    close_start: float = 10000.0,
    trend: float = 0.0003,
    seed: int = 42
) -> pd.DataFrame:
    rng   = np.random.default_rng(seed)
    dates = pd.bdate_range(start=start, periods=n, freq="B")
    close = close_start * np.cumprod(1 + trend + rng.normal(0, 0.015, n))
    # E-9a: 거래량 충분히 높여 min_turnover 확정 통과
    volume = rng.integers(500_000, 2_000_000, n).astype(int)
    # V3 픽스처: 거래량·고가를 일수익률 부호에 연동(상승일 거래량 ×1.5·고가 종가 근접, 하락일 거래량 ×0.5).
    #   순수 난수 픽스처에서는 ⑩~⑬(PVR≤·EVR≥·CS≥) 이 전부 거짓 → 체결 0건 → 시스템 테스트 12건 skip.
    #   눌림(저거래량) → 회복(고거래량·강한 봉) 구조만 부여. 난수 소비 순서·열 구성·E-9a 하한 유지, 전략 코드 무변경.
    up     = np.diff(close, prepend=close[0]) > 0
    volume = np.where(up, volume * 1.5, volume * 0.5).astype(int)

    df = pd.DataFrame({
        "Open":   close * rng.uniform(0.99, 1.01, n),
        "High":   close * np.where(up, rng.uniform(1.00, 1.01, n), rng.uniform(1.00, 1.03, n)),
        "Low":    close * rng.uniform(0.97, 1.00, n),
        "Close":  close,
        "Volume": volume,
    }, index=dates)
    df["High"]     = df[["Open","High","Close"]].max(axis=1)
    df["Low"]      = df[["Open","Low","Close"]].min(axis=1)
    df["Turnover"] = df["Close"] * df["Volume"]
    return df


def make_trade(
    ticker: str = "TEST",
    entry_price: float = 10000.0,
    exit_price: float  = 10500.0,
    stop_price: float  = 9500.0,
    shares: int = 100,
    holding_days: int = 5,
    exit_reason: str = "trailing",        # V1-4: ema_cross 청산 없음
    entry_date: str = "2025-03-01",
    exit_date: str  = "2025-03-08"
) -> pd.Series:
    return pd.Series({
        "ticker":       ticker,
        "sector":       "Unknown",
        "entry_date":   pd.Timestamp(entry_date),
        "exit_date":    pd.Timestamp(exit_date),
        "entry_price":  entry_price,
        "exit_price":   exit_price,
        "stop_price":   stop_price,
        "shares":       shares,
        "entry_cost":   np.nan,
        "exit_cost":    np.nan,
        "pnl_net":      np.nan,
        "r_multiple":   np.nan,
        "exit_reason":  exit_reason,
        "holding_days": holding_days,
        "score":        0.5
    })


def make_trades_with_dates(
    n_win: int = 6,
    n_loss: int = 4,
    eq_index: pd.DatetimeIndex = None
) -> pd.DataFrame:
    if eq_index is None:
        eq_index = pd.bdate_range("2024-01-01", periods=400)

    n_total = n_win + n_loss
    step    = max(len(eq_index) // (n_total + 1), 1)
    dates   = [
        eq_index[min(i * step, len(eq_index) - 1)]
        for i in range(1, n_total + 1)
    ]

    rows = []
    for i in range(n_win):
        ep = 10000.0
        rows.append(make_trade(
            ticker=f"WIN{i:02d}",
            entry_price=ep, exit_price=ep * 1.08,
            stop_price=ep * 0.93, shares=100,
            holding_days=5, exit_reason="trailing",      # V1-4
            entry_date=str((dates[i] - pd.Timedelta(days=5)).date()),
            exit_date=str(dates[i].date())
        ))
    for i in range(n_loss):
        j  = n_win + i
        ep = 10000.0
        rows.append(make_trade(
            ticker=f"LOSS{i:02d}",
            entry_price=ep, exit_price=ep * 0.95,
            stop_price=ep * 0.93, shares=100,
            holding_days=3, exit_reason="stop",
            entry_date=str((dates[j] - pd.Timedelta(days=3)).date()),
            exit_date=str(dates[j].date())
        ))

    df = pd.DataFrame(rows, columns=TRADE_COLUMNS).reset_index(drop=True)
    return attach_costs(df)


def make_equity(
    n: int = 200,
    start: str = "2025-01-02",
    initial: float = 100_000_000.0,
    drift: float = 0.0005,
    seed: int = 42
) -> pd.DataFrame:
    rng   = np.random.default_rng(seed)
    dates = pd.bdate_range(start=start, periods=n)
    eq    = initial * np.cumprod(1 + drift + rng.normal(0, 0.008, n))
    return pd.DataFrame({
        "equity":    eq,
        "cash":      eq * 0.3,
        "positions": rng.integers(0, CONFIG["max_positions"] + 1, n)
    }, index=dates)


def make_system_ohlcv(n: int = 350, seed: int = 0) -> dict:
    """n=350: SMA50·ADX14 정상 계산 보장"""
    return {
        "AAA": make_ohlcv(n, seed=seed,   trend=0.003),
        "BBB": make_ohlcv(n, seed=seed+1, trend=0.002),
        "CCC": make_ohlcv(n, seed=seed+2, trend=0.001),
    }


def make_market(n: int = 350) -> pd.DataFrame:
    return make_ohlcv(n, close_start=2800.0, trend=0.0004, seed=99)


def _run_system(n=350, capital=100_000_000):
    ohlcv     = make_system_ohlcv(n)
    market_df = make_market(n)
    start     = list(ohlcv["AAA"].index)[0]
    end       = list(ohlcv["AAA"].index)[-1]
    return run_period(
        ohlcv, market_df, start, end, FIXED_CONFIG,
        initial_capital=capital,
        compute_pb_effectiveness=False
    )

class TestF1DelPositions:
    """
    F-1: pending_exits 적용 시 del positions[ticker] 없으면
         같은 포지션이 매일 재청산 → 현금 증식 → R3 거짓 통과.
    """

    def test_no_duplicate_entry_date_per_ticker(self):
        """
        F-1 직접 단언: (ticker, entry_date) 중복 = 같은 포지션 중복 청산.
        exit_date가 매일 달라 종전 중복 검사(entry+exit+shares)는 잡지 못함.
        validate_trade_frame에 추가된 검사가 이를 잡아야 함.
        """
        equity, trades, _ = _run_system()
        if trades.empty:
            pytest.skip("거래 없음")

        dup = trades.duplicated(subset=["ticker", "entry_date"])
        assert not dup.any(), \
            f"(ticker, entry_date) 중복 {dup.sum()}건 — F-1 del positions[ticker] 미적용"

    def test_cash_not_growing_unboundedly(self):
        """
        F-1 버그: 중복 청산 → 현금 무한 증식.
        equity 최대값이 initial_capital × 10 미만이어야 함.
        """
        capital = 100_000_000
        equity, _, _ = _run_system(capital=capital)
        max_eq = equity["equity"].max()
        assert max_eq < capital * 10, \
            f"equity 최대 {max_eq:,.0f} > {capital*10:,.0f} — 현금 증식 의심 (F-1)"

    def test_positions_not_stuck_at_max(self):
        """
        F-1 버그: positions가 제거되지 않아 slots=0 고착 → 신규 진입 불가.
        positions가 항상 max_positions에 있지 않아야 함.
        """
        equity, trades, _ = _run_system()
        if len(equity) < 20:
            pytest.skip("데이터 부족")

        pos_series = equity["positions"].iloc[:-1]
        max_pos    = CONFIG["max_positions"]
        stuck_rate = (pos_series == max_pos).mean()
        assert stuck_rate < 0.95, \
            f"positions가 {stuck_rate:.1%} 시간 max 고착 — F-1 del 미적용"

    def test_validate_catches_duplicate_entry_date(self):
        """validate_trade_frame이 (ticker, entry_date) 중복을 잡아야 함"""
        df = make_trades_with_dates(2, 0)
        # 의도적으로 같은 entry_date로 두 번째 행 변조
        df2 = df.copy()
        df2.loc[1, "entry_date"] = df2.loc[0, "entry_date"]
        df2.loc[1, "ticker"]     = df2.loc[0, "ticker"]
        # exit_date만 다르게 (기존 중복 검사 우회)
        df2.loc[1, "exit_date"]  = df2.loc[0, "exit_date"] + pd.Timedelta(days=1)

        ok, errs = validate_trade_frame(df2, raise_on_error=False)
        assert not ok
        assert any("entry_date" in e or "중복" in e for e in errs)

    def test_exit_proceeds_applied_next_day(self):
        """
        E-1·F-1: 청산 대금은 다음 반복 시작 시 반영.
        청산 결정일 equity에 t+1 시가 정보 없어야 함.
        """
        equity, trades, _ = _run_system()
        if trades.empty or len(equity) < 5:
            pytest.skip("데이터 부족")

        eq  = equity["equity"].iloc[:-1]
        pct = eq.pct_change().dropna()

        # 단일일 5% 초과 상승이 10% 이상이면 이상 (중복 청산 = 현금 급증)
        large_up = (pct > 0.05).mean()
        assert large_up < 0.10, \
            f"단일일 +5% 초과 비율 {large_up:.1%} — 청산 현금 선반영 의심"

class TestF3HoldingDays:
    """
    F-3: forced_close는 당일 종가 체결.
    봉거리 = exit_date_index - entry_date_index (익일 시가와 달리 -1).
    정상 청산: holding_days = i - entry_idx ✓ (불변)
    forced_close: holding_days = i - entry_idx - 1 ✓ (F-3)
    기간종료: holding_days = last_idx - entry_idx - 1 ✓ (F-3)
    """

    def test_forced_close_holding_days_gte_1(self):
        """
        G-3 수정: all_dates 합집합 축을 외부에서 재현할 수 없으므로
        봉거리 정확값 대신 >= 1 범위만 확인.
        (F-3 + E-3 + A-1 보장)
        """
        ohlcv     = make_system_ohlcv(350)
        market_df = make_market(350)
        start     = list(ohlcv["AAA"].index)[0]
        end       = list(ohlcv["AAA"].index)[-1]

        _, trades, _ = run_period(
            ohlcv, market_df, start, end, FIXED_CONFIG,
            compute_pb_effectiveness=False
        )

        forced = trades[trades["exit_reason"] == "forced_close"]
        if forced.empty:
            pytest.skip("forced_close 거래 없음")

        assert (forced["holding_days"] >= 1).all(), \
            f"forced_close holding_days < 1:\n{forced[['ticker','holding_days','entry_date','exit_date']]}"

    def test_normal_exit_holding_days_matches_union_index(self):
        """
        G-3: 정상 청산은 all_dates 합집합 봉거리와 일치.
        종목별 df.index 대신 전체 bdate_range 근사치 사용.
        """
        ohlcv     = make_system_ohlcv(350)
        market_df = make_market(350)
        start     = list(ohlcv["AAA"].index)[0]
        end       = list(ohlcv["AAA"].index)[-1]

        _, trades, _ = run_period(
            ohlcv, market_df, start, end, FIXED_CONFIG,

        mismatches = 0
        for date in df.index:
            row    = df.loc[date]
            scalar = _entry_signal(row, FIXED_CONFIG, apply_pullback=False)
            vec    = bool(base.get(date, False))

                mismatches += 1

            f"base 신호 벡터/스칼라 불일치 {mismatches}건"

# 6. F-7b: 역인덱스 검증
# ══════════════════════════════════════════════

class TestF7bReverseIndex:
    """F-7b: date→tickers 역인덱스 정합성"""

    def test_reverse_index_complete(self):
        """역인덱스의 (ticker, date) 집합이 entry_cache의 full 신호와 일치"""
        ohlcv    = make_system_ohlcv(350)
        prepared = {t: _compute_indicators(df) for t, df in ohlcv.items()}
        cache    = _compute_entry_signals(prepared, FIXED_CONFIG)

        # 역인덱스 직접 구성
        date_to_tickers = {}
        for ticker, sig in cache.items():
            for d in sig["full"].index[sig["full"]]:
                date_to_tickers.setdefault(d, []).append(ticker)

        # entry_cache의 (ticker, date) 집합
        expected = set()
        for ticker, sig in cache.items():
            for d in sig["full"].index[sig["full"]]:
                expected.add((ticker, d))

        # 역인덱스에서 재구성한 집합
        actual = set()
        for d, tickers in date_to_tickers.items():
            for ticker in tickers:
                actual.add((ticker, d))

        assert actual == expected, \
            f"역인덱스 누락 {len(expected - actual)}건, 초과 {len(actual - expected)}건"

class TestD1HoldingDays:

    def test_stop_fires_on_day_one_gap_scenario(self):

        dates = pd.bdate_range("2025-01-02", periods=n)
        rng   = np.random.default_rng(7)
        close = 10000.0 * np.cumprod(1 + 0.002 + rng.normal(0, 0.01, n))
        vol   = np.full(n, 1_000_000)

            "Open":    close * 0.99,
            "High":    close * 1.02,
            "Low":     close * 0.98,
            "Close":   close,
            "Volume":  vol,
            "Turnover": close * vol
        }, index=dates)
        df["High"] = df[["Open","High","Close"]].max(axis=1)
        df["Low"]  = df[["Open","Low","Close"]].min(axis=1)

        # 201봉부터 대형 갭다운 (stop_price 돌파)
        for j in range(200, 210):
            df.iloc[j, df.columns.get_loc("Open")]  = close[j] * 0.80
            df.iloc[j, df.columns.get_loc("Close")] = close[j] * 0.82
            df.iloc[j, df.columns.get_loc("Low")]   = close[j] * 0.79
            df.iloc[j, df.columns.get_loc("High")]  = close[j] * 0.83

        ohlcv     = {"GAPPER": df}
        market_df = make_market(n)
        start     = dates[0]
        end       = dates[-1]

        # RuntimeError 없이 실행 완료
        equity, trades, _ = run_period(
            ohlcv, market_df, start, end, FIXED_CONFIG,
            compute_pb_effectiveness=False
        )
        assert isinstance(trades, pd.DataFrame)
        if not trades.empty:
            assert (trades["holding_days"] >= 1).all()

def make_monotone_ohlcv(
    n: int = 300,
    start: str = "2025-01-02",
    close_start: float = 10000.0,
    step: float = 0.001,
    drop_idx=(),
) -> pd.DataFrame:
    """
    H-11: 결정적 시리즈. Close 가 매 봉 +0.1% 단조 증가, Open=0.999C, High=1.01C, Low=0.99C.
      - stop:      next_open ≥ Open[i+1] > entry_price > stop_price  → 발동 불가
      - ema_cross: 단조 증가 → Close > EMA10 항상               → 발동 불가
      - trailing:  next_open ≈ 0.999999·C_i > C_i − 2·ATR(≈0.04·C_i) → 발동 불가
      → 정상 청산은 holding_limit(15) 만 가능. 신호는 _inject_signals 로 주입한다.
    drop_idx: 삭제할 봉의 위치 인덱스(거래정지·상장폐지 시뮬레이션).
    """
    dates = pd.bdate_range(start=start, periods=n, freq="B")
    close = close_start * (1.0 + step) ** np.arange(n)
    df = pd.DataFrame({
        "Open":   close * 0.999,
        "High":   close * 1.01,
        "Low":    close * 0.99,
        "Close":  close,
        "Volume": np.full(n, 1_000_000, dtype=int),
    }, index=dates)
    df["Turnover"] = df["Close"] * df["Volume"]
    drop_idx = list(drop_idx)
    if drop_idx:
        df = df.drop(index=dates[drop_idx])
    return df


def make_monotone_market(n: int = 300) -> pd.DataFrame:
    """단조 증가 시장 시리즈 → SMA50 계산 이후 항상 Close > SMA50 (시장 필터 ON)"""
    return make_monotone_ohlcv(n, close_start=2800.0)


def _inject_signals(monkeypatch, signal_dates: dict):
    """
    H-11: run_period 가 모듈 전역으로 조회하는 _compute_entry_signals 를 결정적 가짜로 교체.
    signal_dates = {ticker: [Timestamp, ...]} 에 있는 날짜만 full/base True.
    """
    def fake(prepared, config):
        out = {}
        for t, df in prepared.items():
            s = pd.Series(False, index=df.index, dtype=bool)
            for d in signal_dates.get(t, []):
                if d in s.index:
                    s.loc[d] = True
            out[t] = {"base": s.copy(), "full": s}
        return out
    monkeypatch.setattr(strat, "_compute_entry_signals", fake)


def _union_axis(ohlcv: dict, start, end) -> list:
    """run_period 내부 all_dates(합집합 축) 재현 — _attach_warmup 은 기간 내 봉을 바꾸지 않는다"""
    return sorted(set(
        d for df in ohlcv.values() for d in df.index if start <= d <= end
    ))


def _assert_union_identity(trades: pd.DataFrame, axis: list):
    """
    H-1: 모든 청산 경로에서 holding_days == pos[exit_date] − pos[entry_date]
      정상 청산:  i − entry_idx,           exit=all_dates[i+1], entry=all_dates[entry_idx+1]
      forced_close(루프 내): i − entry_idx − 1, exit=all_dates[i]
      forced_close(기간종료): last_idx − entry_idx − 1, exit=all_dates[last_idx]
    """
    pos = {d: k for k, d in enumerate(axis)}
    for _, tr in trades.iterrows():
        assert tr["entry_date"] in pos and tr["exit_date"] in pos, \
            f"{tr['ticker']}: 진입/청산일이 합집합 축에 없음"
        expected = pos[tr["exit_date"]] - pos[tr["entry_date"]]
        assert tr["holding_days"] == expected, (
            f"{tr['ticker']} {tr['exit_reason']}: holding_days={tr['holding_days']} "
            f"≠ 축 위치차 {expected} ({tr['entry_date'].date()}→{tr['exit_date'].date()})"
        )


def _run_injected(monkeypatch, ohlcv: dict, signals: dict, n: int = 300):
    """단조 픽스처 + 주입 신호로 run_period 실행. 반환: equity, trades, axis"""
    market_df = make_monotone_market(n)
    axis_src  = pd.bdate_range(start="2025-01-02", periods=n, freq="B")
    start, end = axis_src[0], axis_src[-1]
    _inject_signals(monkeypatch, signals)
    equity, trades, _ = run_period(
        ohlcv, market_df, start, end, FIXED_CONFIG,
        compute_pb_effectiveness=False
    )
    return equity, trades, _union_axis(ohlcv, start, end)

        H-10: 모든 거래의 entry_date 직전 합집합일에 (a) 해당 종목 full 신호 True,
              (b) 시장 Close > SMA50 이어야 한다 (신호일 t → 체결 t+1).

def _cand(ticker, dist_ema10, adx, rsi, **extra):
    """run_period 가 만드는 후보 dict 와 같은 키"""
    base = {
        "ticker": ticker, "dist_ema10": dist_ema10, "adx": adx, "rsi": rsi,
        "return20": 0.0, "dist_sma50": 0.0, "vol_ratio": 1.0,
        "atr": 100.0, "next_open": 10000.0,
    }
    base.update(extra)
    return base


def _goal_trades(n_win, n_loss, win_mult, loss_mult, eq_index):
    """승·패를 날짜 축에 교대로 배치(폴드마다 섞이도록). 비용 부착 완료 프레임 반환."""
    n, step = n_win + n_loss, max(len(eq_index) // (n_win + n_loss + 1), 1)
    kinds, w, l = [], 0, 0
    while w < n_win or l < n_loss:
        if w < n_win and (l >= n_loss or w * n_loss <= l * n_win):
            kinds.append("W"); w += 1
        else:
            kinds.append("L"); l += 1
    rows = []
    for i, k in enumerate(kinds):
        d  = eq_index[min((i + 1) * step, len(eq_index) - 1)]
        ep = 10000.0
        rows.append(make_trade(
            ticker=f"{k}{i:02d}",
            entry_price=ep, exit_price=ep * (win_mult if k == "W" else loss_mult),
            stop_price=ep * 0.93, shares=100,
            holding_days=5, exit_reason="trailing" if k == "W" else "stop",
            entry_date=str((d - pd.Timedelta(days=5)).date()),
            exit_date=str(d.date()),
        ))
    return attach_costs(pd.DataFrame(rows, columns=TRADE_COLUMNS).reset_index(drop=True))

def _mk_on(n=300):
    """단조 증가 벤치마크 → SMA50 유효 이후 항상 ON"""
    return make_monotone_ohlcv(n, close_start=2800.0)


def _mk_off(n=300):
    """단조 감소 벤치마크 → Close < SMA50 항상 → OFF (SMA NaN 구간도 OFF)"""
    return make_monotone_ohlcv(n, close_start=2800.0, step=-0.001)


def _run_gated(monkeypatch, ohlcv, signals, benchmarks, market_map, n=300):
    """주입 신호 + 시장별 게이트로 run_period 실행. market_df 위치 인자는 benchmarks["KOSPI"]."""
    axis_src = pd.bdate_range(start="2025-01-02", periods=n, freq="B")
    start, end = axis_src[0], axis_src[-1]
    _inject_signals(monkeypatch, signals)
    equity, trades, _ = run_period(
        ohlcv, benchmarks["KOSPI"], start, end, FIXED_CONFIG,
        compute_pb_effectiveness=False, benchmarks=benchmarks, market_map=market_map,
    )
    return equity, trades, _union_axis(ohlcv, start, end)

def _fx_insert(path, symbol, market, dates):
    conn = sqlite3.connect(path)
    rows = []
    for k, d in enumerate(dates):
        close, volume = 10000.0 + 10.0 * k, 1000 + k
        rows.append((
            d.strftime("%Y-%m-%d"), symbol, f"NAME_{symbol}", market, "",
            close - 5.0, close + 20.0, close - 20.0, close, volume,
            close * volume * 3.0, 1.0e12, 1_000_000, "2026-09-12T00:00:00",
        ))
    conn.executemany("INSERT INTO daily_prices VALUES (" + ",".join("?" * 14) + ")", rows)
    conn.commit()
    conn.close()

def _pb_reference(df: pd.DataFrame) -> pd.DataFrame:
    """사전등록 §2.2 정의를 그대로 옮긴 O(n·k) 루프 참조 구현(벡터 구현과 독립). 픽스처 전용(Volume NaN 없음)."""
    c = df["Close"].to_numpy(dtype=float); h = df["High"].to_numpy(dtype=float)
    l = df["Low"].to_numpy(dtype=float);   v = df["Volume"].to_numpy(dtype=float)
    e = df["EMA10"].to_numpy(dtype=float)
    n = len(df)
    k_min, k_max = CONFIG["pb_streak_min"], CONFIG["pb_streak_max"]
    win, mn      = CONFIG["pb_prior_window"], CONFIG["pb_prior_min"]
    streak = np.zeros(n, dtype=int); rec = np.zeros(n, dtype=bool)
    pvr = np.full(n, np.nan); evr = np.full(n, np.nan); cs = np.full(n, np.nan)
    for i in range(n):
        k, j = 0, i - 1
        while j >= 0 and (c[j] < e[j]):          # NaN 비교 False → 정지
            k += 1; j -= 1
        streak[i] = k
        if c[i] > e[i] and k_min <= k <= k_max:
            rec[i] = True
            pb    = v[i - k:i] if i - k >= 0 else np.array([])
            prior = v[max(0, i - k - win):i - k]
            avg_pb    = pb.mean()    if len(pb) == k and k > 0 else np.nan
            avg_prior = prior.mean() if len(prior) >= mn       else np.nan
            if not np.isnan(avg_pb) and not np.isnan(avg_prior) and avg_prior > 0:
                pvr[i] = avg_pb / avg_prior
            if not np.isnan(avg_pb) and avg_pb > 0:
                evr[i] = v[i] / avg_pb
        if h[i] - l[i] > 0:
            cs[i] = (c[i] - l[i]) / (h[i] - l[i])
    return pd.DataFrame({"PB_STREAK": streak, "IS_RECOVERY": rec, "PVR": pvr,
                         "EVR": evr, "CANDLE_STRENGTH": cs}, index=df.index)


def _pb_scenario(k: int, prior_vol: float = 1000.0, pb_vol: float = 500.0, rec_vol: float = 1000.0,
                 rec_high: float = 110.0, rec_low: float = 90.0, rec_close: float = 105.0) -> pd.DataFrame:
    """EMA10 = 100 고정. 20봉 상회(거래량 prior_vol) → k봉 하회(pb_vol) → 회복봉(rec_*) → 5봉 상회.
    반환 프레임은 _compute_pullback_stats 직접 호출용(Close/High/Low/Volume/EMA10)."""
    n_pre, n_post = 20, 5
    close  = [105.0] * n_pre + [95.0] * k + [rec_close] + [105.0] * n_post
    vol    = [prior_vol] * n_pre + [pb_vol] * k + [rec_vol] + [prior_vol] * n_post
    high   = [c + 2 for c in close]; low = [c - 2 for c in close]
    i_rec  = n_pre + k
    high[i_rec], low[i_rec] = rec_high, rec_low
    idx = pd.bdate_range("2025-01-02", periods=len(close), freq="B")
    return pd.DataFrame({"Close": close, "High": high, "Low": low, "Volume": vol,
                         "EMA10": [100.0] * len(close)}, index=idx)


def _entry_row(**over) -> pd.Series:
    """base 를 확실히 통과하는 스칼라 행 + v3 열. over 로 덮어쓴다."""
    row = {
        "WEEKLY_TREND": True, "EMA5": 101.0, "EMA10": 100.0, "Close": 105.0, "RSI": 60.0, "ADX": 30.0,
        "MACD_HIST": 0.5, "AVG_TURNOVER20": CONFIG["min_turnover"] * 2, "DAILY_RET": 0.01, "ATR": 2.0,
        "EMA10_TOUCH5": True, "IS_RECOVERY": True, "PVR": 0.5, "EVR": 2.0, "CANDLE_STRENGTH": 0.75,
    }
    row.update(over)
    return pd.Series(row)


def _entry_frame(rows: list) -> pd.DataFrame:
    idx = pd.bdate_range("2025-01-02", periods=len(rows), freq="B")
    return pd.DataFrame([r.to_dict() for r in rows], index=idx)


class TestV3PullbackStats:
    """V3-2: _compute_pullback_stats — 참조 구현 대조·구성 시나리오·경계"""

    def test_vector_matches_loop_reference(self):
        """무작위 픽스처 5시드: 벡터 구현 == 루프 참조 구현(모든 열, NaN 위치 포함)"""
        n_rec_total = 0
        for seed in range(5):
            df  = _compute_indicators(make_ohlcv(300, seed=seed))
            ref = _pb_reference(df)
            assert (df["PB_STREAK"].to_numpy() == ref["PB_STREAK"].to_numpy()).all(), f"seed {seed}: PB_STREAK"
            assert (df["IS_RECOVERY"].to_numpy() == ref["IS_RECOVERY"].to_numpy()).all(), f"seed {seed}: IS_RECOVERY"
            for col in ("PVR", "EVR", "CANDLE_STRENGTH"):
                a, b = df[col].to_numpy(dtype=float), ref[col].to_numpy(dtype=float)
                assert np.array_equal(np.isnan(a), np.isnan(b)), f"seed {seed}: {col} NaN 위치 불일치"
                assert np.allclose(a[~np.isnan(a)], b[~np.isnan(b)], rtol=1e-12, atol=0), f"seed {seed}: {col} 값"
            n_rec_total += int(df["IS_RECOVERY"].sum())
        assert n_rec_total > 0, "회복 봉이 하나도 없으면 대조가 공허함"

    def test_columns_present_after_indicators(self):
        df = _compute_indicators(make_ohlcv(120))
        for col in ("PB_STREAK", "IS_RECOVERY", "PVR", "EVR", "CANDLE_STRENGTH", "EMA10_TOUCH5"):
            assert col in df.columns
        assert df["IS_RECOVERY"].dtype == bool
        assert (df["PB_STREAK"] >= 0).all()
        # 회복이 아닌 봉의 PVR·EVR 은 NaN, CANDLE_STRENGTH 는 회복 여부와 무관하게 [0,1]
        nr = ~df["IS_RECOVERY"]
        assert df.loc[nr, "PVR"].isna().all() and df.loc[nr, "EVR"].isna().all()
        cs = df["CANDLE_STRENGTH"].dropna()
        assert ((cs >= 0) & (cs <= 1)).all()

    def test_short_frame_is_safe(self):
        """짧은 프레임(픽스처·워밍업 부족)에서도 예외 없이 열이 생성되고 전부 NaN/False"""
        for n in (1, 2, 5):
            df = _compute_pullback_stats(_pb_scenario(3).iloc[:n].copy())
            assert len(df) == n and "IS_RECOVERY" in df.columns
        empty = _compute_pullback_stats(_pb_scenario(3).iloc[:0].copy())
        assert len(empty) == 0 and "PVR" in empty.columns

    @pytest.mark.parametrize("k", [1, 2, 3, 7])
    def test_scenario_values_within_bounds(self, k):
        """k봉 눌림(거래량 500) 후 회복(거래량 1000, 직전 10봉 1000): PVR=0.5, EVR=2.0, CS=(105−90)/(110−90)=0.75"""
        df = _compute_pullback_stats(_pb_scenario(k))
        i  = 20 + k
        assert df["PB_STREAK"].iloc[i] == k
        assert bool(df["IS_RECOVERY"].iloc[i]) is True
        assert df["PVR"].iloc[i] == pytest.approx(0.5)
        assert df["EVR"].iloc[i] == pytest.approx(2.0)
        assert df["CANDLE_STRENGTH"].iloc[i] == pytest.approx(0.75)
        # 회복 봉 외에는 회복 아님(눌림 중 봉·이후 상회 봉·직전 상회 봉)
        others = [j for j in range(len(df)) if j != i]
        assert not df["IS_RECOVERY"].iloc[others].any()
        assert df["PVR"].iloc[others].isna().all() and df["EVR"].iloc[others].isna().all()

    def test_streak_above_max_is_not_recovery(self):
        """k = pb_streak_max + 1 → PB_STREAK 는 기록되지만 IS_RECOVERY False, PVR·EVR NaN"""
        k  = CONFIG["pb_streak_max"] + 1
        df = _compute_pullback_stats(_pb_scenario(k))
        i  = 20 + k
        assert df["PB_STREAK"].iloc[i] == k
        assert bool(df["IS_RECOVERY"].iloc[i]) is False
        assert np.isnan(df["PVR"].iloc[i]) and np.isnan(df["EVR"].iloc[i])
        assert not df["IS_RECOVERY"].any()

    def test_zero_streak_is_not_recovery(self):
        """어제도 상회(k=0)면 회복 아님"""
        df = _compute_pullback_stats(_pb_scenario(0))
        assert not df["IS_RECOVERY"].any()

    def test_h_equals_l_gives_nan_candle_strength(self):
        df = _compute_pullback_stats(_pb_scenario(2, rec_high=105.0, rec_low=105.0))
        i  = 22
        assert bool(df["IS_RECOVERY"].iloc[i]) is True
        assert np.isnan(df["CANDLE_STRENGTH"].iloc[i])
        assert df["PVR"].iloc[i] == pytest.approx(0.5)     # 다른 열은 영향 없음

    def test_prior_window_below_min_gives_nan_pvr(self):
        """눌림 직전 기준 봉수 < pb_prior_min → PVR NaN(EVR 은 계산됨)"""
        base = _pb_scenario(2)
        mn   = CONFIG["pb_prior_min"]
        df   = _compute_pullback_stats(base.iloc[20 - (mn - 1):].copy())   # 직전 상회 봉 mn−1 개만 남김
        i    = (mn - 1) + 2
        assert bool(df["IS_RECOVERY"].iloc[i]) is True
        assert np.isnan(df["PVR"].iloc[i])
        assert df["EVR"].iloc[i] == pytest.approx(2.0)
        df2  = _compute_pullback_stats(base.iloc[20 - mn:].copy())          # 정확히 mn 개 → 계산됨
        assert df2["PVR"].iloc[mn + 2] == pytest.approx(0.5)

    def test_zero_pullback_volume_gives_nan(self):
        """눌림 봉 거래량 0 → EVR NaN(0 나눗셈 방어), PVR 0"""
        df = _compute_pullback_stats(_pb_scenario(2, pb_vol=0.0))
        i  = 22
        assert bool(df["IS_RECOVERY"].iloc[i]) is True
        assert np.isnan(df["EVR"].iloc[i])
        assert df["PVR"].iloc[i] == pytest.approx(0.0)

    def test_streak_counted_up_to_yesterday_only(self):
        """PB_STREAK 는 오늘을 제외한 직전 연속 하회 봉수: 눌림 중 봉들은 0,1,2,… 로 증가"""
        k  = 4
        df = _compute_pullback_stats(_pb_scenario(k))
        assert list(df["PB_STREAK"].iloc[20:20 + k + 1]) == list(range(k + 1))
        assert df["PB_STREAK"].iloc[20 + k + 1] == 0


class TestV3EntryRule:
    """V3-3: full = base ∧ IS_RECOVERY ∧ PVR ≤ pvr_max ∧ EVR ≥ evr_min ∧ CS ≥ cs_min — 벡터·스칼라·NaN·임계값 전파"""

    def _both(self, rows, config):
        df   = _entry_frame(rows)
        vec  = _compute_entry_signals({"T": df}, config)["T"]
        scal = [(_entry_signal(df.loc[d], config, apply_pullback=False),
                 _entry_signal(df.loc[d], config, apply_pullback=True)) for d in df.index]
        return df, vec, scal

    def test_pass_and_each_failure_mode(self):
        rows = [
            _entry_row(),                                       # 0 통과
            _entry_row(IS_RECOVERY=False),                      # 1 회복 아님
            _entry_row(IS_RECOVERY=np.nan),                     # 2 회복 NaN
            _entry_row(PVR=np.nan),                             # 3 PVR NaN
            _entry_row(EVR=np.nan),                             # 4 EVR NaN
            _entry_row(CANDLE_STRENGTH=np.nan),                 # 5 CS NaN (H==L)
            _entry_row(PVR=CONFIG["pvr_max"] + 1e-9),           # 6 PVR 초과
            _entry_row(EVR=CONFIG["evr_min"] - 1e-9),           # 7 EVR 미달
            _entry_row(CANDLE_STRENGTH=CONFIG["cs_min"] - 1e-9),# 8 CS 미달
            _entry_row(PVR=CONFIG["pvr_max"], EVR=CONFIG["evr_min"], CANDLE_STRENGTH=CONFIG["cs_min"]),  # 9 경계 = 통과
            _entry_row(RSI=75.0),                               # 10 base 실패(RSI)
            _entry_row(EMA10_TOUCH5=False),                     # 11 v2 열은 v3 판정에 무관 → 통과
        ]
        expect_full = [True, False, False, False, False, False, False, False, False, True, False, True]
        expect_base = [True] * 10 + [False, True]
        df, vec, scal = self._both(rows, FIXED_CONFIG)
        assert list(vec["base"].astype(bool)) == expect_base
        assert list(vec["full"].astype(bool)) == expect_full
        assert [s[0] for s in scal] == expect_base
        assert [s[1] for s in scal] == expect_full
        assert not (vec["full"] & ~vec["base"]).any()

    def test_thresholds_propagate_from_config(self):
        """config 의 pvr_max/evr_min/cs_min 이 벡터·스칼라 양쪽에 적용된다(선택 세트와 무관하게 두 세트 명시 비교)"""
        rows = [_entry_row(PVR=0.72, EVR=1.15, CANDLE_STRENGTH=0.52)]   # 1차 세트 3개 전부 실패·완화 세트 3개 전부 통과
        strict = {**FIXED_CONFIG, **PB_THRESHOLD_SETS["primary"]}
        _, vec_d, scal_d = self._both(rows, strict)
        assert bool(vec_d["full"].iloc[0]) is False and scal_d[0][1] is False
        loose = {**FIXED_CONFIG, **PB_THRESHOLD_SETS["relaxed"]}
        _, vec_r, scal_r = self._both(rows, loose)
        assert bool(vec_r["full"].iloc[0]) is True and scal_r[0][1] is True
        tight = {**FIXED_CONFIG, "cs_min": 0.9}
        rows2 = [_entry_row()]
        _, vec_t, scal_t = self._both(rows2, tight)
        assert bool(vec_t["full"].iloc[0]) is False and scal_t[0][1] is False

    def test_config_matches_chosen_threshold_set(self):
        assert PB_THRESHOLD_SET in PB_THRESHOLD_SETS
        assert set(PB_THRESHOLD_SETS) == {"primary", "relaxed"}
        for name, ts in PB_THRESHOLD_SETS.items():
            assert set(ts) == {"pvr_max", "evr_min", "cs_min"}, name
        p, r = PB_THRESHOLD_SETS["primary"], PB_THRESHOLD_SETS["relaxed"]
        assert p["pvr_max"] < r["pvr_max"] and p["evr_min"] > r["evr_min"] and p["cs_min"] > r["cs_min"]
        chosen = PB_THRESHOLD_SETS[PB_THRESHOLD_SET]
        for key in ("pvr_max", "evr_min", "cs_min"):
            assert CONFIG[key] == chosen[key]
            assert FIXED_CONFIG[key] == CONFIG[key]
        assert CONFIG["pb_streak_min"] == 1 and CONFIG["pb_streak_max"] == 7
        assert CONFIG["pb_prior_window"] == 10 and CONFIG["pb_prior_min"] == 3
        assert PB_PRECHECK_FLOOR == 2_000

    def test_full_subset_of_base_recovery_touch5(self):
        """실지표 픽스처: full ⊆ base, full ⊆ IS_RECOVERY, full ⊆ EMA10_TOUCH5(v2 정의) — v3 ⊆ v2 불변식"""
        for seed in range(5):
            df   = _compute_indicators(make_ohlcv(300, seed=seed))
            full = _compute_entry_signals({"T": df}, FIXED_CONFIG)["T"]["full"]
            base = _compute_entry_signals({"T": df}, FIXED_CONFIG)["T"]["base"]
            assert not (full & ~base).any()
            assert not (full & ~df["IS_RECOVERY"]).any()
            assert not (full & ~df["EMA10_TOUCH5"]).any()

    def test_e8_single_source_multi_seed(self):
        """F-7 확장: 5시드 × 300봉 벡터/스칼라 동치(base·full) — full 합계 > 0 (비공허)"""
        n_full_total = 0
        for seed in range(5):
            df    = _compute_indicators(make_ohlcv(300, seed=seed))
            cache = _compute_entry_signals({"T": df}, FIXED_CONFIG)["T"]
            n_full_total += int(cache["full"].sum())
            for d in df.index:
                row = df.loc[d]
                assert _entry_signal(row, FIXED_CONFIG, apply_pullback=False) == bool(cache["base"].loc[d]), (seed, d)
                assert _entry_signal(row, FIXED_CONFIG, apply_pullback=True)  == bool(cache["full"].loc[d]), (seed, d)
        assert n_full_total > 0, "5시드 전부 full 신호 0건이면 동치 검사가 공허함"


class TestV3Diagnostics:
    """V3-4: _pullback_filter_effectiveness 의 v3 진단 — 건수 정합·㉮ 출력 유지·주입 가짜 호환"""

    def test_counts_consistent(self, capsys):
        ohlcv    = make_system_ohlcv(350)
        prepared = {t: _compute_indicators(df) for t, df in ohlcv.items()}
        dates    = sorted(set().union(*[set(df.index) for df in prepared.values()]))
        cache    = _compute_entry_signals(prepared, FIXED_CONFIG)
        res      = _pullback_filter_effectiveness(prepared, dates, FIXED_CONFIG, cache)
        out      = capsys.readouterr().out
        assert "[㉮ 효과 검증]" in out and "기본 신호:" in out and "눌림목 통과:" in out
        assert "[V3 눌림의 질 진단]" in out and "  v3_full:" in out
        v3 = res["v3"]
        assert v3 is not None and all(isinstance(x, int) for x in v3.values())
        assert v3["full"] == res["눌림목_통과_수"]
        assert v3["recovery"] <= res["기본신호_수"]
        for key in ("recovery_pvr", "recovery_evr", "recovery_cs"):
            assert v3[key] <= v3["recovery"]
            assert v3["full"] <= v3[key]
        assert v3["touch5_v2"] >= v3["recovery"]                 # 회복 ⊆ v2 TOUCH5
        k_max = CONFIG["pb_streak_max"]
        assert sum(v3[f"streak_k{k}"] for k in range(1, k_max + 1)) == v3["recovery"]
        assert v3[f"streak_gt{k_max}"] >= 0
        # 창을 절반으로 줄이면 건수는 감소하거나 같다
        half = dates[: len(dates) // 2]
        res2 = _pullback_filter_effectiveness(prepared, half, FIXED_CONFIG, cache)
        capsys.readouterr()
        assert res2["v3"]["recovery"] <= v3["recovery"] and res2["v3"]["full"] <= v3["full"]

    def test_missing_columns_gives_none(self, capsys):
        ohlcv    = make_system_ohlcv(200)
        prepared = {t: _compute_indicators(df).drop(columns=["PVR"]) for t, df in ohlcv.items()}
        dates    = sorted(set().union(*[set(df.index) for df in prepared.values()]))
        fake     = {t: {"base": pd.Series(True, index=df.index), "full": pd.Series(False, index=df.index)}
                    for t, df in prepared.items()}
        res      = _pullback_filter_effectiveness(prepared, dates, FIXED_CONFIG, fake)
        out      = capsys.readouterr().out
        assert res["v3"] is None and "[V3 눌림의 질 진단]" not in out
        assert res["기본신호_수"] == sum(len(df) for df in prepared.values()) and res["눌림목_통과_수"] == 0
        assert _v3_pullback_diagnostics({}, {}, set(), FIXED_CONFIG) is None

    def test_injected_fake_with_real_columns(self, capsys):
        """H-11 가짜 캐시(base/full 만) + 실지표 프레임 → 진단은 가짜 base 위에서 계산, full 은 가짜 full 건수"""
        ohlcv    = make_system_ohlcv(200)
        prepared = {t: _compute_indicators(df) for t, df in ohlcv.items()}
        dates    = sorted(set().union(*[set(df.index) for df in prepared.values()]))
        t0       = next(iter(prepared))
        d_sig    = [prepared[t0].index[150], prepared[t0].index[160]]
        fake     = {}
        for t, df in prepared.items():
            s = pd.Series(False, index=df.index, dtype=bool)
            if t == t0:
                s.loc[d_sig] = True
            fake[t] = {"base": s.copy(), "full": s}
        res = _pullback_filter_effectiveness(prepared, dates, FIXED_CONFIG, fake)
        capsys.readouterr()
        assert res["기본신호_수"] == 2 and res["눌림목_통과_수"] == 2
        assert res["v3"] is not None and res["v3"]["full"] == 2 and res["v3"]["recovery"] <= 2
