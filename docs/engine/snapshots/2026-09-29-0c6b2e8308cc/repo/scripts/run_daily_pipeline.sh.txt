#!/usr/bin/env bash
# 매 평일 (1) KRX 가격 DB 갱신, (2) 후보 스캔·감사로그 적재, (3) 벤치마크 종가 스냅샷,
# (4) Pullback v2/v4 전향 관측 러너(DB 가 전진했을 때만)를 순서대로 실행한다.
# Windows 작업 스케줄러(KoreaQuantDailyPipeline)가 평일 08:20 KST 부터 15분 간격으로 1시간 동안
# (08:20/08:35/08:50/09:05/09:20) 이 스크립트를 호출한다 -- D-44:
#   wsl.exe -d Ubuntu-26.04 -- bash -lc "/home/gram/korea-quant-system/scripts/run_daily_pipeline.sh"
# T-1 봉이 이미 DB 에 있고 감사로그가 동기면 반복 호출은 no-op(skip 한 줄만 기록). 인자 `force` 는
# 이 가드를 건너뛴다(수동 실행·점검용).
#
# 라이브 전략 파라미터·슬롯·비용 설정은 건드리지 않는다 -- 지금까지 사람이 수동으로
# 치던 두 명령을 그대로 실행할 뿐이다. 실행 결과는 모두 data/pipeline_run.log에
# append하며, 어떤 단계도 조용히 실패하지 않는다. (1)·(2) 중 하나라도 실패하면 종료코드 1 을
# 돌려 스케줄러 "마지막 실행 결과"에 남긴다. (3)·(4) 는 관측 보조 단계라 실패해도 종료코드에 반영하지 않는다.
set -uo pipefail

REPO="/home/gram/korea-quant-system"
PY="$REPO/.venv/bin/python"
LOG="$REPO/data/pipeline_run.log"

cd "$REPO" || { echo "$(date -u +%FT%TZ) FATAL: cannot cd to $REPO" >>"$LOG" 2>&1; exit 1; }

# 반복 트리거 가드: T-1 봉 적재 + 감사로그 동기면 아무것도 하지 않는다. `force` 는 가드 무시.
FORCE="${1:-}"
if [ "$FORCE" != "force" ] && "$PY" scripts/pipeline_gap_check.py --already-current >/dev/null 2>&1; then
    echo "$(date -u +%FT%TZ) skip: T-1 bar already in DB and audit in sync (repeat trigger)" >>"$LOG"
    exit 0
fi

krx_rc=1
picks_rc=1
{
    echo "===== $(date -u +%FT%TZ) pipeline start =====${FORCE:+ (arg: $FORCE)}"

    # 직전 실행이 end 스탬프 없이 끊겼으면(절전·강제 종료) 재구성 불가 사실만 남긴다.
    last_start=$(grep -n "pipeline start" "$LOG" | tail -n 2 | head -n 1 | cut -d: -f1)
    last_end=$(grep -n "pipeline end" "$LOG" | tail -n 1 | cut -d: -f1)
    if [ -n "${last_start:-}" ] && [ "${last_start:-0}" -gt "${last_end:-0}" ]; then
        echo "WARNING: 직전 실행이 end 스탬프 없이 중단된 것으로 추정 (log line $last_start)"
    fi

    db_before=$("$PY" scripts/pipeline_gap_check.py --db-latest)
    echo "  DB latest before: ${db_before:-none}"

    echo "[1/4] krx_cli.py update --lookback-days 10"
    "$PY" -m korea_quant.krx_cli --db data/krx_daily.sqlite3 update --lookback-days 10
    krx_rc=$?
    echo "  exit code: $krx_rc"

    echo "[2/4] daily_picks_cli.py"
    "$PY" -m korea_quant.daily_picks_cli --krx-db data/krx_daily.sqlite3
    picks_rc=$?
    echo "  exit code: $picks_rc"

    db_after=$("$PY" scripts/pipeline_gap_check.py --db-latest)
    echo "  DB latest after: ${db_after:-none}"

    echo "[3/4] benchmark_snapshot.py (069500/229200 종가 append-only)"
    "$PY" scripts/benchmark_snapshot.py
    echo "  exit code: $?"

    # Pullback v2/v4 전향 관측(D-42/D-43, 사전등록 §9): DB 가 전진했을 때만 실행. 러너는 DB 를
    # mode=ro 로만 열고 건수·불리언·해시만 찍는다. 결과 전문은 /home/gram/pullback_v{2,4}_forward_runs.log.
    if [ -n "${db_after:-}" ] && [ "${db_after:-}" != "${db_before:-}" ]; then
        echo "[4/4] pullback forward runners (DB advanced ${db_before:-none} -> $db_after)"
        for ver in v2 v4; do
            runner="/home/gram/pullback_${ver}_forward.py"
            tmp_out=$(mktemp)
            "$PY" "$runner" >"$tmp_out" 2>&1
            runner_rc=$?
            grep -E "^(FORWARD_WINDOW|FORWARD_TRADES:|JUDGE_FORWARD|RUN_ERROR|ABORT)" "$tmp_out" | sed "s/^/  $ver /"
            if [ "$runner_rc" -ne 0 ]; then
                echo "  $ver WARNING: runner exit code $runner_rc"
                tail -n 5 "$tmp_out" | sed "s/^/  $ver | /"
            fi
            rm -f "$tmp_out"
        done
    else
        echo "[4/4] pullback forward runners skipped (DB latest unchanged: ${db_after:-none})"
    fi

    echo "--- 갭 자가진단 ---"
    "$PY" scripts/pipeline_gap_check.py

    if [ "$krx_rc" -ne 0 ] || [ "$picks_rc" -ne 0 ]; then
        echo "WARNING: 파이프라인 단계 실패 (krx_rc=$krx_rc picks_rc=$picks_rc)"
    fi

    echo "===== $(date -u +%FT%TZ) pipeline end ====="
    echo ""
} >>"$LOG" 2>&1

if [ "$krx_rc" -ne 0 ] || [ "$picks_rc" -ne 0 ]; then
    exit 1
fi
exit 0
