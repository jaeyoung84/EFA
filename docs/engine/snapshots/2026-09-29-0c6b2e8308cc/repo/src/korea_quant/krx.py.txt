def _fill_halted_trading_days(history: pd.DataFrame) -> pd.DataFrame:
    """Carry the prior close forward on trading-halt rows (OHLC reported as 0/blank).

    KRX reports zero prices on days a symbol is halted (pending delisting, merger,
    disposal trading, etc.). Feeding a literal 0 into the backtester as an order
    price crashes the simulator, so treat these as flat no-trade days instead.
    """

    price_cols = ["Open", "High", "Low", "Close"]
    halted = history[price_cols].le(0).any(axis=1)
    if not halted.any():
        return history
    history = history.sort_values(["Symbol", "Date"]).reset_index(drop=True)
    halted = history[price_cols].le(0).any(axis=1)
    history.loc[halted, price_cols] = np.nan
    prior_close = history.groupby("Symbol")["Close"].ffill()
    history.loc[halted, price_cols] = np.repeat(prior_close.loc[halted].to_numpy()[:, None], 4, axis=1)
    return history.sort_values(["Date", "Symbol"]).reset_index(drop=True)


class KrxApiError(RuntimeError):
    """Raised when KRX cannot return a valid response."""


def _dotenv_value(path: Path, key: str) -> str:
    """Read one plain KEY=value value without executing a dotenv file."""

    if not path.is_file():
        return ""
    for raw_line in path.read_text(encoding="utf-8").splitlines():
        line = raw_line.strip()
        if not line or line.startswith("#") or "=" not in line:
            continue
        candidate, value = line.split("=", 1)
        if candidate.strip() != key:
            continue
        value = value.strip()
        if len(value) >= 2 and value[0] == value[-1] and value[0] in {"'", '"'}:
            value = value[1:-1]
        return value.strip()
    return ""


@dataclass(frozen=True)
class KrxOpenApiClient:
    """Client for KRX date-based stock and ETF daily-trading endpoints."""

    auth_key: str
    base_url: str = "https://data-dbg.krx.co.kr/svc/apis"
    timeout: float = 30.0
    retries: int = 3
    transport: Transport | None = None

    @classmethod
    def from_env(cls, env_name: str = "KRX_AUTH_KEY", **kwargs: Any) -> KrxOpenApiClient:
        auth_key = os.environ.get(env_name, "").strip() or _dotenv_value(Path(".env"), env_name)
        if not auth_key:
            raise ValueError(f"Set {env_name} to the approved KRX Open API authentication key.")
        return cls(auth_key=auth_key, **kwargs)

    def fetch_market_day(self, market: Market, trading_day: date) -> list[dict[str, Any]]:
        if market not in MARKET_ENDPOINTS:
            raise ValueError(f"Unsupported market: {market}")
        query = urllib.parse.urlencode({"basDd": trading_day.strftime("%Y%m%d")})
        url = f"{self.base_url.rstrip('/')}/{MARKET_ENDPOINTS[market]}?{query}"
        payload = self._request(url)
        rows = payload.get("OutBlock_1")
        if not isinstance(rows, list):
            raise KrxApiError("KRX response is missing the OutBlock_1 list.")
        return [
            _normalize_row(row, market, trading_day)
            for row in rows
            if not _skip_unusable_etf_row(row, market)
        ]

    def _request(self, url: str) -> dict[str, Any]:
        headers = {"AUTH_KEY": self.auth_key, "Accept": "application/json"}
        if self.transport is not None:
            return self.transport(url, headers)

        last_error: Exception | None = None
        for attempt in range(self.retries + 1):
            try:
                request = urllib.request.Request(url, headers=headers, method="GET")
                with urllib.request.urlopen(request, timeout=self.timeout) as response:
                    payload = json.loads(response.read().decode("utf-8"))
                if not isinstance(payload, dict):
                    raise KrxApiError("KRX response must be a JSON object.")
                return payload
            except (urllib.error.URLError, TimeoutError, json.JSONDecodeError) as exc:
                last_error = exc
                if attempt < self.retries:
                    time.sleep(2**attempt)
        raise KrxApiError(f"KRX request failed after retries: {last_error}") from last_error


class KrxDailyStore:
    """SQLite store with idempotent daily-bar UPSERTs and collection checkpoints."""

    def __init__(self, path: str | Path = "data/krx_daily.sqlite3") -> None:
        self.path = Path(path)
        self.path.parent.mkdir(parents=True, exist_ok=True)
        self._initialize()

    def connect(self) -> sqlite3.Connection:
        connection = sqlite3.connect(self.path)
        connection.row_factory = sqlite3.Row
        return connection

    def _initialize(self) -> None:
        with self.connect() as connection:
            self._migrate_market_constraints(connection)
            connection.executescript(
                """
                PRAGMA journal_mode=WAL;
                CREATE TABLE IF NOT EXISTS daily_prices (
                    trade_date TEXT NOT NULL,
                    symbol TEXT NOT NULL,
                    name TEXT NOT NULL,
                    market TEXT NOT NULL CHECK (market IN ('KOSPI', 'KOSDAQ', 'ETF')),
                    section TEXT NOT NULL,
                    open INTEGER NOT NULL CHECK (open >= 0),
                    high INTEGER NOT NULL CHECK (high >= 0),
                    low INTEGER NOT NULL CHECK (low >= 0),
                    close INTEGER NOT NULL CHECK (close >= 0),
                    volume INTEGER NOT NULL CHECK (volume >= 0),
                    trading_value INTEGER NOT NULL CHECK (trading_value >= 0),
                    market_cap INTEGER NOT NULL CHECK (market_cap >= 0),
                    listed_shares INTEGER NOT NULL CHECK (listed_shares >= 0),
                    collected_at TEXT NOT NULL,
                    PRIMARY KEY (trade_date, symbol)
                );
                CREATE INDEX IF NOT EXISTS idx_daily_symbol_date
                    ON daily_prices(symbol, trade_date);
                CREATE TABLE IF NOT EXISTS collection_log (
                    trade_date TEXT NOT NULL,
                    market TEXT NOT NULL CHECK (market IN ('KOSPI', 'KOSDAQ', 'ETF')),
                    row_count INTEGER NOT NULL,
                    collected_at TEXT NOT NULL,
                    PRIMARY KEY (trade_date, market)
                );
                CREATE TABLE IF NOT EXISTS universe_snapshots (
                    universe TEXT NOT NULL,
                    effective_date TEXT NOT NULL,
                    available_date TEXT NOT NULL,
                    source TEXT NOT NULL,
                    member_count INTEGER NOT NULL CHECK (member_count >= 0),
                    collected_at TEXT NOT NULL,
                    PRIMARY KEY (universe, effective_date)
                );
                CREATE TABLE IF NOT EXISTS universe_members (
                    universe TEXT NOT NULL,
                    effective_date TEXT NOT NULL,
                    symbol TEXT NOT NULL,
                    PRIMARY KEY (universe, effective_date, symbol),
                    FOREIGN KEY (universe, effective_date)
                        REFERENCES universe_snapshots(universe, effective_date)
                        ON DELETE CASCADE
                );
                CREATE INDEX IF NOT EXISTS idx_universe_snapshot_dates
                    ON universe_snapshots(universe, effective_date, available_date);
                """
            )

    @staticmethod
    def _migrate_market_constraints(connection: sqlite3.Connection) -> None:
        """Expand legacy two-market tables without losing already collected bars."""
        schemas = {
            row[0]: row[1]
            for row in connection.execute(
                "SELECT name, sql FROM sqlite_master WHERE type = 'table'"
            )
        }
        if not schemas:
            return
        needs_prices = "daily_prices" in schemas and "'ETF'" not in schemas["daily_prices"]
        needs_log = "collection_log" in schemas and "'ETF'" not in schemas["collection_log"]
        if not needs_prices and not needs_log:
            return
        connection.execute("PRAGMA foreign_keys = OFF")
        if needs_prices:
            connection.executescript(
                """
                BEGIN IMMEDIATE;
                ALTER TABLE daily_prices RENAME TO daily_prices_legacy;
                CREATE TABLE daily_prices (
                    trade_date TEXT NOT NULL,
                    symbol TEXT NOT NULL,
                    name TEXT NOT NULL,
                    market TEXT NOT NULL CHECK (market IN ('KOSPI', 'KOSDAQ', 'ETF')),
                    section TEXT NOT NULL,
                    open INTEGER NOT NULL CHECK (open >= 0),
                    high INTEGER NOT NULL CHECK (high >= 0),
                    low INTEGER NOT NULL CHECK (low >= 0),
                    close INTEGER NOT NULL CHECK (close >= 0),
                    volume INTEGER NOT NULL CHECK (volume >= 0),
                    trading_value INTEGER NOT NULL CHECK (trading_value >= 0),
                    market_cap INTEGER NOT NULL CHECK (market_cap >= 0),
                    listed_shares INTEGER NOT NULL CHECK (listed_shares >= 0),
                    collected_at TEXT NOT NULL,
                    PRIMARY KEY (trade_date, symbol)
                );
                INSERT INTO daily_prices SELECT * FROM daily_prices_legacy;
                DROP TABLE daily_prices_legacy;
                CREATE INDEX idx_daily_symbol_date ON daily_prices(symbol, trade_date);
                COMMIT;
                """
            )
        if needs_log:
            connection.executescript(
                """
                BEGIN IMMEDIATE;
                ALTER TABLE collection_log RENAME TO collection_log_legacy;
                CREATE TABLE collection_log (
                    trade_date TEXT NOT NULL,
                    market TEXT NOT NULL CHECK (market IN ('KOSPI', 'KOSDAQ', 'ETF')),
                    row_count INTEGER NOT NULL,
                    collected_at TEXT NOT NULL,
                    PRIMARY KEY (trade_date, market)
                );
                INSERT INTO collection_log SELECT * FROM collection_log_legacy;
                DROP TABLE collection_log_legacy;
                COMMIT;
                """
            )

    def is_collected(self, market: Market, trading_day: date) -> bool:
        with self.connect() as connection:
            row = connection.execute(
                "SELECT 1 FROM collection_log WHERE trade_date = ? AND market = ?",
                (trading_day.isoformat(), market),
            ).fetchone()
        return row is not None

    def upsert_day(
        self, market: Market, trading_day: date, rows: Iterable[dict[str, Any]]
    ) -> int:
        materialized = list(rows)
        now = datetime.now(UTC).isoformat()
        values = [
            (
                row["trade_date"], row["symbol"], row["name"], row["market"],
                row["section"], row["open"], row["high"], row["low"], row["close"],
                row["volume"], row["trading_value"], row["market_cap"],
                row["listed_shares"], now,
            )
            for row in materialized
        ]
        with self.connect() as connection:
            connection.executemany(
                """
                INSERT INTO daily_prices VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
                ON CONFLICT(trade_date, symbol) DO UPDATE SET
                    name=excluded.name, market=excluded.market, section=excluded.section,
                    open=excluded.open, high=excluded.high, low=excluded.low,
                    close=excluded.close, volume=excluded.volume,
                    trading_value=excluded.trading_value, market_cap=excluded.market_cap,
                    listed_shares=excluded.listed_shares, collected_at=excluded.collected_at
                """,
                values,
            )
            connection.execute(
                """
                INSERT INTO collection_log VALUES (?, ?, ?, ?)
                ON CONFLICT(trade_date, market) DO UPDATE SET
                    row_count=excluded.row_count, collected_at=excluded.collected_at
                """,
                (trading_day.isoformat(), market, len(materialized), now),
            )
        return len(materialized)

    def prices(
        self,
        symbol: str,
        start: str | date | None = None,
        end: str | date | None = None,
    ) -> pd.DataFrame:
        symbol = validate_krx_symbol(symbol)
        start_date = _as_date(start, "start") if start is not None else None
        end_date = _as_date(end, "end") if end is not None else None
        if start_date and end_date and start_date > end_date:
            raise ValueError("start must be on or before end.")
        conditions = ["symbol = ?"]
        parameters: list[str] = [symbol]
        if start_date:
            conditions.append("trade_date >= ?")
            parameters.append(start_date.isoformat())
        if end_date:
            conditions.append("trade_date <= ?")
            parameters.append(end_date.isoformat())
        query = f"""
            SELECT trade_date AS Date, open AS Open, high AS High, low AS Low,
                   close AS Close, volume AS Volume
            FROM daily_prices WHERE {' AND '.join(conditions)} ORDER BY trade_date
        """
        with self.connect() as connection:
            frame = pd.read_sql_query(query, connection, params=parameters, parse_dates=["Date"])
        if frame.empty:
            return pd.DataFrame(columns=["Open", "High", "Low", "Close", "Volume"])
        frame = frame.set_index("Date")
        # KRX reports zero O/H/L on no-trade days. Preserve raw zeroes in SQLite but
        # render a flat candle at the close when consumers request chart-ready data.
        for column in ("Open", "High", "Low"):
            frame[column] = frame[column].where(frame[column] > 0, frame["Close"])
        return frame

    def benchmark_closes(
        self, symbols: dict[str, str], end: str | date | None = None
    ) -> dict[str, pd.Series]:
        """Load each market's benchmark close for regime filtering.

        No start bound is applied: the moving average needs warm-up history that
        predates the validation window, and loading from the first stored bar
        maximises it. The end bound still applies, so nothing after the tested
        period can leak in.
        """

        closes: dict[str, pd.Series] = {}
        for market, symbol in symbols.items():
            frame = self.prices(symbol, None, end)
            if frame.empty:
                raise ValueError(f"Benchmark {symbol} has no stored bars for {market}.")
            closes[market] = frame["Close"]
        return closes

    def daily_prices(
        self, symbol: str, start: str | date, end: str | date | None = None
    ) -> pd.DataFrame:
        """Implement PriceDataSource using locally collected KRX bars."""

        frame = self.prices(symbol, start, end)
        if frame.empty:
            raise ValueError(
                f"No KRX database prices for {symbol} in the requested period."
            )
        return frame

    def securities(self, query: str | None = None, limit: int = 50) -> pd.DataFrame:
        """Return the latest name/market snapshot, optionally filtered by code/name."""

        if limit < 1:
            raise ValueError("limit must be positive.")
        parameters: list[Any] = []
        where = ""
        if query:
            where = "WHERE symbol LIKE ? OR name LIKE ?"
            pattern = f"%{query.strip()}%"
            parameters.extend([pattern, pattern])
        parameters.append(limit)
        sql = f"""
            WITH ranked AS (
                SELECT symbol, name, market, trade_date,
                       ROW_NUMBER() OVER (PARTITION BY symbol ORDER BY trade_date DESC) AS rn
                FROM daily_prices {where}
            )
            SELECT symbol, name, market, trade_date AS latest_date
            FROM ranked WHERE rn = 1 ORDER BY market, symbol LIMIT ?
        """
        with self.connect() as connection:
            return pd.read_sql_query(sql, connection, params=parameters)

    def security_name(self, symbol: str) -> str | None:
        symbol = validate_krx_symbol(symbol)
        with self.connect() as connection:
            row = connection.execute(
                """
                SELECT name FROM daily_prices WHERE symbol = ?
                ORDER BY trade_date DESC LIMIT 1
                """,
                (symbol,),
            ).fetchone()
        return None if row is None else str(row[0])

    def market_history(
        self,
        start: str | date,
        end: str | date | None = None,
        markets: tuple[Market, ...] = ("KOSPI", "KOSDAQ"),
    ) -> pd.DataFrame:
        """Load historical universe rows without dropping securities that later disappear."""

        start_date = _as_date(start, "start")
        end_date = _as_date(end, "end") if end is not None else None
        if end_date and start_date > end_date:
            raise ValueError("start must be on or before end.")
        if not markets or any(market not in MARKET_ENDPOINTS for market in markets):
            raise ValueError("markets must contain KOSPI and/or KOSDAQ.")
        placeholders = ", ".join("?" for _ in markets)
        conditions = ["trade_date >= ?", f"market IN ({placeholders})"]
        parameters: list[Any] = [start_date.isoformat(), *markets]
        if end_date:
            conditions.append("trade_date <= ?")
            parameters.append(end_date.isoformat())
        sql = f"""
            SELECT trade_date AS Date, symbol AS Symbol, name AS Name, market AS Market,
                   open AS Open, high AS High, low AS Low, close AS Close, volume AS Volume,
                   trading_value AS TradingValue, market_cap AS MarketCap
            FROM daily_prices WHERE {' AND '.join(conditions)}
            ORDER BY trade_date, symbol
        """
        with self.connect() as connection:
            history = pd.read_sql_query(sql, connection, params=parameters, parse_dates=["Date"])
        return _fill_halted_trading_days(history)

    def upsert_universe_snapshot(
        self,
        universe: str,
        effective_date: str | date,
        symbols: Iterable[str],
        *,
        source: str,
        available_date: str | date | None = None,
    ) -> int:
        """Atomically replace one dated universe snapshot."""

        name = universe.strip().upper()
        provenance = source.strip()
        if not name:
            raise ValueError("universe must not be blank.")
        if not provenance:
            raise ValueError("source must not be blank.")
        effective = _as_date(effective_date, "effective_date")
        available = (
            _as_date(available_date, "available_date")
            if available_date is not None
            else effective
        )
        members = sorted(
            {validate_krx_symbol(str(symbol).strip().upper()) for symbol in symbols}
        )
        now = datetime.now(UTC).isoformat()
        with self.connect() as connection:
            connection.execute("PRAGMA foreign_keys = ON")
            connection.execute(
                """
                INSERT INTO universe_snapshots VALUES (?, ?, ?, ?, ?, ?)
                ON CONFLICT(universe, effective_date) DO UPDATE SET
                    available_date=excluded.available_date, source=excluded.source,
                    member_count=excluded.member_count, collected_at=excluded.collected_at
                """,
                (
                    name, effective.isoformat(), available.isoformat(), provenance,
                    len(members), now,
                ),
            )
            connection.execute(
                "DELETE FROM universe_members WHERE universe = ? AND effective_date = ?",
                (name, effective.isoformat()),
            )
            connection.executemany(
                "INSERT INTO universe_members VALUES (?, ?, ?)",
                ((name, effective.isoformat(), symbol) for symbol in members),
            )
        return len(members)

    def universe_members(
        self,
        universe: str,
        as_of: str | date,
        *,
        known_on: str | date | None = None,
    ) -> pd.DataFrame:
        """Return the latest effective snapshot that was knowable by ``known_on``."""

        name = universe.strip().upper()
        if not name:
            raise ValueError("universe must not be blank.")
        decision_date = _as_date(as_of, "as_of")
        knowledge_date = (
            _as_date(known_on, "known_on") if known_on is not None else decision_date
        )
        sql = """
            WITH chosen AS (
                SELECT effective_date, available_date, source
                FROM universe_snapshots
                WHERE universe = ? AND effective_date <= ? AND available_date <= ?
                ORDER BY effective_date DESC, available_date DESC LIMIT 1
            )
            SELECT m.symbol, s.effective_date, s.available_date, s.source
            FROM chosen s
            JOIN universe_members m
              ON m.universe = ? AND m.effective_date = s.effective_date
            ORDER BY m.symbol
        """
        with self.connect() as connection:
            return pd.read_sql_query(
                sql,
                connection,
                params=(
                    name, decision_date.isoformat(), knowledge_date.isoformat(), name,
                ),
            )

    def universe_membership_panel(
        self,
        universe: str,
        dates: Iterable[str | date | pd.Timestamp],
        symbols: Iterable[str],
    ) -> pd.DataFrame:
        """Build a causal date-by-symbol membership mask from stored snapshots."""

        name = universe.strip().upper()
        if not name:
            raise ValueError("universe must not be blank.")
        index = pd.DatetimeIndex(pd.to_datetime(list(dates))).normalize()
        if index.has_duplicates:
            raise ValueError("dates must not contain duplicates.")
        columns = pd.Index(
            [validate_krx_symbol(str(symbol).strip().upper()) for symbol in symbols],
            dtype="object",
        )
        if columns.has_duplicates:
            raise ValueError("symbols must not contain duplicates.")
        panel = pd.DataFrame(False, index=index, columns=columns, dtype=bool)
        if index.empty:
            return panel

        with self.connect() as connection:
            snapshots = pd.read_sql_query(
                """
                SELECT effective_date, available_date
                FROM universe_snapshots
                WHERE universe = ? AND effective_date <= ? AND available_date <= ?
                ORDER BY effective_date, available_date
                """,
                connection,
                params=(name, index.max().date().isoformat(), index.max().date().isoformat()),
                parse_dates=["effective_date", "available_date"],
            )
            members = pd.read_sql_query(
                """
                SELECT effective_date, symbol FROM universe_members
                WHERE universe = ? AND effective_date <= ?
                """,
                connection,
                params=(name, index.max().date().isoformat()),
                parse_dates=["effective_date"],
            )
        if snapshots.empty or columns.empty:
            return panel

        members_by_date = {
            effective_date: set(group["symbol"]) & set(columns)
            for effective_date, group in members.groupby("effective_date")
        }
        for trading_day in index:
            eligible = snapshots[
                (snapshots["effective_date"] <= trading_day)
                & (snapshots["available_date"] <= trading_day)
            ]
            if eligible.empty:
                continue
            effective_date = eligible.iloc[-1]["effective_date"]
            current_members = members_by_date.get(effective_date, set())
            if current_members:
                panel.loc[trading_day, list(current_members)] = True
        return panel

    def status(self) -> dict[str, Any]:
        with self.connect() as connection:
            row = connection.execute(
                """
                SELECT MIN(trade_date), MAX(trade_date), COUNT(*), COUNT(DISTINCT symbol)
                FROM daily_prices
                """
            ).fetchone()
            calls = connection.execute("SELECT COUNT(*) FROM collection_log").fetchone()[0]
        return {
            "first_date": row[0], "last_date": row[1], "rows": row[2],
            "symbols": row[3], "completed_market_days": calls,
        }


@dataclass
class KrxCollector:
    client: KrxOpenApiClient
    store: DailyStore
    delay_seconds: float = 0.1

    def collect_range(
        self,
        start: date,
        end: date,
        *,
        refresh: bool = False,
        markets: tuple[Market, ...] = tuple(MARKET_ENDPOINTS),
        progress: Callable[[date, Market, int], None] | None = None,
    ) -> dict[str, int]:
        if start > end:
            raise ValueError("start must be on or before end.")
        if not markets or any(market not in MARKET_ENDPOINTS for market in markets):
            raise ValueError("markets must contain supported KRX market values.")
        totals = {"requests": 0, "rows": 0, "skipped": 0}
        for trading_day in weekdays(start, end):
            for market in markets:
                if not refresh and self.store.is_collected(market, trading_day):
                    totals["skipped"] += 1
                    continue
                rows = self.client.fetch_market_day(market, trading_day)
                count = self.store.upsert_day(market, trading_day, rows)
                totals["requests"] += 1
                totals["rows"] += count
                if progress:
                    progress(trading_day, market, count)
                if self.delay_seconds:
                    time.sleep(self.delay_seconds)
        return totals

    def update_recent(
        self,
        end: date | None = None,
        lookback_days: int = 10,
        markets: tuple[Market, ...] = tuple(MARKET_ENDPOINTS),
    ) -> dict[str, int]:
        if lookback_days < 1:
            raise ValueError("lookback_days must be positive.")
        end = end or datetime.now(UTC).date()
        return self.collect_range(
            end - timedelta(days=lookback_days - 1), end, refresh=True, markets=markets
        )


def weekdays(start: date, end: date) -> Iterable[date]:
    current = start
    while current <= end:
        if current.weekday() < 5:
            yield current
        current += timedelta(days=1)


def _normalize_row(row: Any, market: Market, requested_day: date) -> dict[str, Any]:
    if not isinstance(row, dict):
        raise KrxApiError("KRX row must be a JSON object.")
    symbol = str(row.get("ISU_CD", "")).strip().upper()
    if len(symbol) != 6 or not symbol.isascii() or not symbol.isalnum():
        raise KrxApiError(f"Invalid KRX symbol in response: {symbol!r}")
    raw_date = str(row.get("BAS_DD", "")).strip()
    try:
        response_day = date.fromisoformat(
            f"{raw_date[0:4]}-{raw_date[4:6]}-{raw_date[6:8]}"
        )
    except ValueError as exc:
        raise KrxApiError(f"Invalid KRX date in response: {raw_date!r}") from exc
    if response_day != requested_day:
        raise KrxApiError(f"KRX returned {response_day} for requested date {requested_day}.")

    normalized: dict[str, Any] = {
        "trade_date": response_day.isoformat(),
        "symbol": symbol,
        "name": str(row.get("ISU_NM", "")).strip(),
        "market": market,
        "section": str(row.get("SECT_TP_NM", "")).strip(),
    }
    for source, target in NUMERIC_FIELDS.items():
        value = str(row.get(source, "")).replace(",", "").strip()
        if not value and source in ZERO_WHEN_BLANK_FIELDS:
            value = "0"
        try:
            normalized[target] = int(value)
        except ValueError as exc:
            raise KrxApiError(f"Invalid integer in {source}: {value!r}") from exc
        if normalized[target] < 0:
            raise KrxApiError(f"Negative value in {source}: {value!r}")
    return normalized


def _skip_unusable_etf_row(row: Any, market: Market) -> bool:
    """Exclude ETF response rows without a close, which cannot form a daily bar."""
    return (
        market == "ETF"
        and isinstance(row, dict)
        and not str(row.get("TDD_CLSPRC", "")).replace(",", "").strip()
    )
