"""D-36 `index-tsmom-v1` locked-window data collection/normalization.

Merges and normalizes the three locked-window source types (raw KRX daily price
CSVs, 분배금 xls, CD91 ECOS series) into aligned datasets. Computes no returns,
signals, or performance figures — that remains Stage 5C and requires separate
authorization. Running this module against the real locked-window source files
(2002-10-14~2016-08-31) is Stage 5A itself and also requires separate
authorization; see PREREGISTRATION_D36_index_tsmom_v1.md and its hash-lock
record for the governing protocol.
"""

from __future__ import annotations

import bisect
from datetime import date
from pathlib import Path

import pandas as pd

PRICE_DATE_COL = "일자"
DIST_CODE_COL_HINT = "코드"
DIST_DATE_COL_HINT = "지급기준일"
DIST_AMOUNT_COL_HINT = "분배금"

# 합의17: forward-fill CD91 through short gaps; a gap longer than this many
# consecutive trading days without a quote is a data-quality failure, not
# something to paper over with carry-forward or a substitute rule.
CD91_MAX_GAP_BUSINESS_DAYS = 10


class DataQualityError(RuntimeError):
    """Raised when collected/merged data fails a D-36 completeness requirement."""


def merge_price_files(paths: list[Path], *, encoding: str = "cp949") -> pd.DataFrame:
    """Merge KRX daily price CSVs into one date-sorted, deduplicated frame.

    Each file must contain the KRX Korean-header 일자 column. Overlapping dates
    across files are allowed only if every shared column agrees; disagreement
    raises DataQualityError rather than silently picking one side.
    """

    frames = []
    for path in paths:
        df = pd.read_csv(path, encoding=encoding)
        if PRICE_DATE_COL not in df.columns:
            raise DataQualityError(f"{path.name}: missing required column '{PRICE_DATE_COL}'.")
        df = df.copy()
        df[PRICE_DATE_COL] = pd.to_datetime(df[PRICE_DATE_COL], format="%Y/%m/%d")
        df["__source"] = path.name
        frames.append(df)

    if not frames:
        raise DataQualityError("No price files were provided to merge.")

    merged = pd.concat(frames, ignore_index=True).sort_values(PRICE_DATE_COL)

    dup_mask = merged.duplicated(subset=[PRICE_DATE_COL], keep=False)
    if dup_mask.any():
        compare_cols = [c for c in merged.columns if c not in (PRICE_DATE_COL, "__source")]
        conflicting = [
            str(day.date())
            for day, group in merged.loc[dup_mask].groupby(PRICE_DATE_COL)
            if group[compare_cols].nunique().gt(1).any()
        ]
        if conflicting:
            raise DataQualityError(
                "Overlapping price files disagree on: " + ", ".join(conflicting)
            )
        merged = merged.drop_duplicates(subset=[PRICE_DATE_COL], keep="first")

    return merged.drop(columns="__source").sort_values(PRICE_DATE_COL).reset_index(drop=True)


def check_price_continuity(
    dates: pd.Series, *, max_calendar_gap_days: int = 10
) -> list[tuple[date, date]]:
    """Flag calendar-day gaps larger than max_calendar_gap_days for holiday review.

    This only surfaces candidate gaps; judging whether each is a legitimate
    market-holiday run against an official trading calendar is Stage 5B.
    """

    sorted_dates = pd.to_datetime(pd.Series(dates)).sort_values().reset_index(drop=True)
    diffs = sorted_dates.diff().dt.days
    return [
        (sorted_dates.iloc[idx - 1].date(), sorted_dates.iloc[idx].date())
        for idx in sorted_dates.index[diffs > max_calendar_gap_days]
    ]


def approximate_ex_dividend_date(
    record_date: date, trading_days: pd.DatetimeIndex | list[date]
) -> date:
    """Approximate the ex-dividend date as (record date - 1 trading day).

    Confirmed rule (2026-09-09, PREREGISTRATION_D36_index_tsmom_v1.md §4.2):
    Korea's T+2 settlement means a purchase must settle on or before
    record_date to be holder-of-record, so the last trading day to buy
    cum-dividend is the trading day before the trading day on/before
    record_date; that trading day is the ex-dividend date. `trading_days`
    must be the actual observed trading-day set (e.g. from the merged price
    series), not a generic calendar library, so Korean market holidays are
    handled correctly without an extra dependency.
    """

    days = sorted({pd.Timestamp(d).date() for d in trading_days})
    if not days:
        raise DataQualityError("trading_days is empty; cannot approximate an ex-dividend date.")

    idx = bisect.bisect_right(days, record_date) - 1
    if idx <= 0:
        raise DataQualityError(
            f"No trading day before record date {record_date} within the provided "
            "trading-day set."
        )
    return days[idx - 1]


def normalize_distributions(
    path: Path,
    *,
    symbol: str,
    trading_days: pd.DatetimeIndex | list[date],
) -> pd.DataFrame:
    """Normalize the 분배금 xls for one symbol into (record_date, ex_dividend_date, amount).

    Matches cross_check_069500.py's proven parsing quirks: header row 2 (0-indexed),
    code column zero-padded to 6 digits, 지급기준일 as YYYYMMDD text. Rows whose
    record_date falls outside [min(trading_days), max(trading_days)] are dropped
    before ex-dividend-date approximation is attempted, since a raw distribution
    export can span far more history than the caller's window and computing an
    ex-dividend date against a trading-day set that doesn't cover the event would
    silently produce a meaningless result.
    """

    raw = pd.read_excel(path, engine="xlrd", header=2)
    code_col = next((c for c in raw.columns if DIST_CODE_COL_HINT in str(c)), None)
    date_col = next((c for c in raw.columns if DIST_DATE_COL_HINT in str(c)), None)
    amount_col = next(
        (
            c
            for c in raw.columns
            if DIST_AMOUNT_COL_HINT in str(c) and DIST_DATE_COL_HINT not in str(c)
        ),
        None,
    )
    if code_col is None or date_col is None or amount_col is None:
        raise DataQualityError(
            "Distribution file missing an expected 코드/지급기준일/분배금 column; "
            f"found columns: {list(raw.columns)}"
        )

    df = raw.copy()
    normalized_code = (
        df[code_col].astype(str).str.strip().str.replace(r"\.0$", "", regex=True).str.zfill(6)
    )
    df = df[normalized_code == symbol].copy()
    df[date_col] = pd.to_datetime(df[date_col].astype(str), format="%Y%m%d", errors="coerce")
    if df[date_col].isna().any():
        raise DataQualityError(
            f"Distribution rows for {symbol} contain an unparseable {date_col} value."
        )

    df["record_date"] = df[date_col].dt.date

    trading_day_set = sorted({pd.Timestamp(d).date() for d in trading_days})
    if not trading_day_set:
        raise DataQualityError("trading_days is empty; cannot normalize distributions.")
    window_start, window_end = trading_day_set[0], trading_day_set[-1]
    df = df[(df["record_date"] >= window_start) & (df["record_date"] <= window_end)].copy()

    df["ex_dividend_date"] = df["record_date"].apply(
        lambda d: approximate_ex_dividend_date(d, trading_days)
    )
    df["amount"] = pd.to_numeric(df[amount_col], errors="coerce")
    if df["amount"].isna().any():
        raise DataQualityError(f"Distribution rows for {symbol} contain a non-numeric {amount_col} value.")

    return (
        df[["record_date", "ex_dividend_date", "amount"]]
        .sort_values("record_date")
        .reset_index(drop=True)
    )


def attach_cd91_with_carry_forward(
    price_dates: pd.Series,
    cd91_frame: pd.DataFrame,
    *,
    max_gap_business_days: int = CD91_MAX_GAP_BUSINESS_DAYS,
) -> pd.DataFrame:
    """Align CD91 onto trading days, forward-filling gaps up to the 합의17 threshold.

    cd91_frame must have `date` and `value` columns (as returned by
    korea_quant.ecos.fetch_cd91). Raises DataQualityError if the series has no
    quote before the first trading day (nothing to carry forward from), or if
    any internal gap exceeds max_gap_business_days consecutive trading days —
    per D-36's '미확보 시 처리': do not substitute another rule when the cash
    rate is unavailable, wait instead.
    """

    trading_days = pd.DatetimeIndex(sorted(pd.to_datetime(pd.Series(price_dates)).unique()))
    if trading_days.empty:
        raise DataQualityError("price_dates is empty; cannot attach CD91.")

    cd91 = cd91_frame.set_index(pd.to_datetime(cd91_frame["date"]))["value"].sort_index()
    aligned = cd91.reindex(trading_days)
    is_na = aligned.isna().tolist()

    leading_gap = 0
    for missing in is_na:
        if missing:
            leading_gap += 1
        else:
            break
    if leading_gap:
        raise DataQualityError(
            f"CD91 series has no quote for the first {leading_gap} trading day(s); "
            "nothing to forward-fill from at the start of the window."
        )

    gap_lengths: list[int] = []
    current = 0
    for missing in is_na:
        if missing:
            current += 1
        else:
            if current:
                gap_lengths.append(current)
            current = 0
    if current:
        gap_lengths.append(current)

    max_gap = max(gap_lengths, default=0)
    if max_gap > max_gap_business_days:
        raise DataQualityError(
            f"CD91 series has a {max_gap}-trading-day gap, exceeding the "
            f"{max_gap_business_days}-day quality threshold (합의17)."
        )

    filled = aligned.ffill()
    return pd.DataFrame({"date": trading_days, "cd91_value": filled.to_numpy()})
