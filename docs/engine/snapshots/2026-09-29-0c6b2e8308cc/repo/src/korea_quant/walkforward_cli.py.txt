"""CLI for fixed-parameter rolling out-of-sample validation."""

from __future__ import annotations

import argparse
from pathlib import Path

from korea_quant.backtest import BacktestConfig
from korea_quant.krx import KrxDailyStore
from korea_quant.strategy import STRATEGIES
from korea_quant.walkforward import WalkForwardConfig, run_walk_forward, write_walk_forward_reports


def main() -> None:
    parser = argparse.ArgumentParser(description="Run rolling out-of-sample validation on a KRX symbol.")
    parser.add_argument("--krx-db", type=Path, required=True)
    parser.add_argument("--symbol", required=True)
    parser.add_argument("--start", required=True)
    parser.add_argument("--end", default=None)
    parser.add_argument("--strategy", choices=STRATEGIES, default="ema30-resistance")
    parser.add_argument("--training-sessions", type=int, default=252)
    parser.add_argument("--testing-sessions", type=int, default=63)
    parser.add_argument("--market-filter", default=None)
    parser.add_argument("--market-window", type=int, default=200)
    parser.add_argument("--require-golden-cross", action="store_true")
    parser.add_argument("--golden-fast-window", type=int, default=50)
    parser.add_argument("--golden-slow-window", type=int, default=200)
    parser.add_argument("--cash", type=float, default=10_000_000)
    parser.add_argument("--fee", type=float, default=0.00015)
    parser.add_argument("--slippage", type=float, default=0.0005)
    parser.add_argument("--report", type=Path, default=None)
    args = parser.parse_args()

    store = KrxDailyStore(args.krx_db)
    prices = store.daily_prices(args.symbol, args.start, args.end)
    report = run_walk_forward(
        prices,
        BacktestConfig(
            strategy=args.strategy, initial_cash=args.cash,
            fee_rate=args.fee, slippage_rate=args.slippage,
            market_filter_window=args.market_window,
            require_golden_cross=args.require_golden_cross,
            golden_fast_window=args.golden_fast_window,
            golden_slow_window=args.golden_slow_window,
        ),
        WalkForwardConfig(args.training_sessions, args.testing_sessions),
        market_close=(
            store.daily_prices(args.market_filter, args.start, args.end)["Close"]
            if args.market_filter else None
        ),
    )
    html_output = args.report or Path("reports") / f"{args.symbol}_{args.strategy}_walkforward.html"
    html, csv = write_walk_forward_reports(
        report, html_output, html_output.with_suffix(".csv"),
        title=f"{args.symbol} {store.security_name(args.symbol) or ''} — {args.strategy} Walk-forward",
    )
    print(f"Walk-forward: {html.resolve()} | {csv.resolve()}")


if __name__ == "__main__":
    main()
