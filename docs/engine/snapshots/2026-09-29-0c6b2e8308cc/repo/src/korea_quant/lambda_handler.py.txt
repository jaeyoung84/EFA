"""Scheduled AWS Lambda entry point for recent KRX daily-bar refreshes."""

from __future__ import annotations

import json
import logging
import os
from datetime import datetime
from typing import Any
from zoneinfo import ZoneInfo

from korea_quant.aws_storage import S3ParquetDailyStore
from korea_quant.krx import KrxCollector, KrxOpenApiClient

LOGGER = logging.getLogger(__name__)
LOGGER.setLevel(logging.INFO)
SEOUL = ZoneInfo("Asia/Seoul")


def handler(event: dict[str, Any], context: Any) -> dict[str, Any]:
    """Refresh recent partitions; repeated scheduler delivery is idempotent."""

    import boto3
    from botocore.config import Config

    del context
    config = Config(
        retries={"total_max_attempts": 4, "mode": "adaptive"},
        connect_timeout=5,
        read_timeout=30,
    )
    ssm = boto3.client("ssm", config=config)
    s3 = boto3.client("s3", config=config)
    parameter = ssm.get_parameter(
        Name=_required_env("KRX_AUTH_KEY_PARAMETER"), WithDecryption=True
    )
    auth_key = parameter["Parameter"]["Value"]
    end = _event_date(event)
    store = S3ParquetDailyStore(
        bucket=_required_env("DATA_BUCKET"),
        s3_client=s3,
        prefix=os.environ.get("DATA_PREFIX", "krx/daily"),
    )
    totals = KrxCollector(
        KrxOpenApiClient(auth_key=auth_key), store, delay_seconds=0.1
    ).update_recent(end=end, lookback_days=int(os.environ.get("LOOKBACK_DAYS", "10")))
    LOGGER.info("KRX refresh completed: %s", json.dumps(totals))
    return {"end": end.isoformat(), **totals}


def _event_date(event: dict[str, Any]):
    if event.get("time"):
        parsed = datetime.fromisoformat(str(event["time"]))
        return parsed.astimezone(SEOUL).date()
    return datetime.now(SEOUL).date()


def _required_env(name: str) -> str:
    value = os.environ.get(name, "").strip()
    if not value:
        raise ValueError(f"Required environment variable is missing: {name}")
    return value
