"""Historical validation for the standalone KRX mean-reversion strategies.

This harness used to backtest each out-of-sample window on its own 63-session
slice and keep only the trades that closed inside it. Both strategies here exit
purely on recovery -- no stop loss, no holding cap -- so the positions still open
at a window edge are exactly the ones that had not recovered, and deleting them
deleted only from the losing tail: 30% of bollinger-rsi's positions and 96% of
rsi-obv's, taking its measured mean from +31% to below zero. trend_validation
already ran one continuous backtest per symbol for the same reason; this module
now does too, labelling each trade by the window its entry fell in. Positions
still open at the end of the history are kept and reported rather than dropped,
because for this family an unclosed position is an untaken loss, not a
non-event.
"""

# The live scanner (daily_picks_cli.py) pools swing and long_term candidates
# into one tracked_symbols set and one shared daily-pick cap before a name is
# ever excluded as "already held" -- a symbol entered by one strategy blocks
# the other from re-entering it the same way it blocks itself. A summary row
# scoped to a single strategy can never see that: it hands deployment_metrics
# a trade list that never contains the cross-strategy collision, so the same
# symbol can appear "held" in two different one-strategy books at once. This
# label marks the additional row that pools every strategy present in a scope
# before running deployment_metrics, so dedupe and the cap are exercised the
# way one account actually running both strategies would experience them.
COMBINED_STRATEGY = "combined"


@dataclass(frozen=True)
class ReversionValidationConfig:
    training_sessions: int = 252
    testing_sessions: int = 63
    market_cap_tiers: tuple[tuple[str, int], ...] = DEFAULT_MARKET_CAP_TIERS
    liquidity_window: int = 20
    minimum_average_trading_value: float = 5_000_000_000
    minimum_history_sessions: int = 260
    outlier_gap_threshold: float = 0.50
    outlier_cooldown_sessions: int = 60
    fee_rate: float = 0.00015
    sell_tax_rate: float = 0.0018
    slippage_rate: float = 0.0005
    goal: ReversionGoal = field(default_factory=ReversionGoal)
    checkpoint_dir: Path | None = None
    checkpoint_symbols_per_file: int = 50
    strategies: tuple[str, ...] = (SWING_STRATEGY, LONG_TERM_STRATEGY)
    # Zero disables the benchmark regime gate, keeping the unfiltered baseline
    # reproducible. This family buys weakness, so the gate is inverted by
    # default: it wants the benchmark BELOW its own average, the opposite of
    # what trend_validation asks for.
    market_filter_window: int = 0
    market_filter_inverted: bool = True
    # How the trade list would actually have been deployed. The per-trade
    # metrics score signals; this scores the money that could have followed
    # them, which is a different question and, for this family, a harsher one.
    deployment: DeploymentConfig = field(default_factory=DeploymentConfig)


# Naming the columns keeps an all-zero summary shaped like a populated one. A
# regime gate strict enough to suppress every trade is a legitimate result, not
# a crash, and the report still has to render.
SUMMARY_COLUMNS = (
    "Segment", "Market", "Strategy", "Profile", *METRIC_COLUMNS,
    "Minimum Trade Count", "Passes Goal", *CAPITAL_COLUMNS, *SELECTION_COLUMNS,
)


@dataclass(frozen=True)
class ReversionValidationResult:
    trades: pd.DataFrame
    summary: pd.DataFrame


def run_reversion_validation(
    history: pd.DataFrame,
    config: ReversionValidationConfig | None = None,
    benchmark_close: dict[str, pd.Series] | None = None,
) -> ReversionValidationResult:
    """Run one continuous backtest per symbol; label out-of-sample windows by entry date."""

    config = config or ReversionValidationConfig()
    _validate_config(config)
    data = validate_market_history(history)
    dates = pd.DatetimeIndex(data["Date"].drop_duplicates().sort_values())
    if len(dates) <= config.training_sessions:
        raise ValueError("History must contain sessions after the training window.")

    # entry_eligibility is causal at every step -- a per-date cross-sectional
    # rank, backward rolling windows, an expanding count, then a shift -- so
    # computing it once over the whole history gives each date the same verdict
    # the old per-window truncation gave it, with no look-ahead introduced.
    ranked = entry_eligibility(
        data,
        market_cap_tiers=config.market_cap_tiers,
        liquidity_window=config.liquidity_window,
        minimum_average_trading_value=config.minimum_average_trading_value,
        minimum_history_sessions=config.minimum_history_sessions,
        outlier_gap_threshold=config.outlier_gap_threshold,
        outlier_cooldown_sessions=config.outlier_cooldown_sessions,
    )
    groups = [
        group for _, group in ranked.groupby(["Market", "Symbol"], sort=True)
        if group["EntryAllowed"].any()
    ]
    regimes = market_regime_allowances(
        benchmark_close,
        data["Market"].unique(),
        config.market_filter_window,
        inverted=config.market_filter_inverted,
    )
    checkpoint_dir = _prepare_checkpoint_dir(data, config, benchmark_close)
    trades = _run_all(groups, config, checkpoint_dir, regimes)
    trades = _attach_live_selection_keys(trades, ranked)
    trades = _label_windows(trades, dates, config)
    # One segment, not two. This harness scores fixed, shipped parameters and
    # fits nothing, so there is no in-sample region for a train/test split to
    # separate -- and entry_eligibility already withholds every symbol until it
    # has minimum_history_sessions (260) of history, past training_sessions
    # (252), so the old "full_sample" rows were a byte-for-byte copy of the
    # out-of-sample rows in every report ever produced. Emitting both invited
    # readers to treat one as corroborating the other. The walk-forward windows
    # remain and now carry the stability criterion instead.
    labelled = trades[trades["Window"].notna()].assign(Segment="out_of_sample")
    labelled = _attach_deployment_valuations(labelled, data, config)
    return ReversionValidationResult(
        labelled,
        summarize_reversion_trades(labelled, config.goal, config.deployment),
    )


def summarize_reversion_trades(
    trades: pd.DataFrame,
    goal: ReversionGoal | None = None,
    deployment: DeploymentConfig | None = None,
) -> pd.DataFrame:
    """Calculate costed closed-trade acceptance results by market and strategy."""

    goal = goal or ReversionGoal()
    validate_reversion_goal(goal)
    deployment = deployment or DeploymentConfig()
    validate_deployment_config(deployment)
    required = {"Segment", "Market", "Strategy", "NetReturnPct"}
    missing = required - set(trades.columns)
    if missing:
        raise ValueError(f"Trades are missing: {', '.join(sorted(missing))}")
    rows = []
    for market in ("KOSPI", "KOSDAQ", "ALL"):
        scoped = trades if market == "ALL" else trades[trades["Market"] == market]
        for segment, segment_group in scoped.groupby("Segment", sort=True):
            for strategy, group in segment_group.groupby("Strategy", sort=True):
                metrics = trade_metrics(group)
                minimum = goal.minimum_trades_for(strategy, SWING_STRATEGY)
                rows.append(
                    {
                        "Segment": segment, "Market": market, "Strategy": strategy,
                        "Profile": profile_label(strategy), **metrics.as_row(),
                        "Minimum Trade Count": minimum,
                        # Passes Goal deliberately still reads only the per-trade
                        # metrics. The capital columns beside it are reported, not
                        # enforced: picking the CAGR a strategy must clear is a
                        # documented decision, not a default a module gets to invent.
                        "Passes Goal": goal.passes(metrics, minimum),
                        **_capital_row(group, deployment),
                    }
                )
            # ReversionGoal has no threshold defined for a pooled book, so this
            # row leaves Minimum Trade Count / Passes Goal unevaluated (NaN)
            # rather than inventing one -- only the capital columns, which are
            # already documented as reported-not-enforced, carry a verdict here.
            if segment_group["Strategy"].nunique() > 1:
                rows.append(
                    {
                        "Segment": segment, "Market": market, "Strategy": COMBINED_STRATEGY,
                        "Profile": "스윙+장기 합산", **trade_metrics(segment_group).as_row(),
                        "Minimum Trade Count": float("nan"),
                        "Passes Goal": float("nan"),
                        **_capital_row(segment_group, deployment),
                    }
                )
    return (
        pd.DataFrame(rows, columns=list(SUMMARY_COLUMNS))
        .sort_values(["Segment", "Strategy", "Market"])
        .reset_index(drop=True)
    )


def _capital_row(
    group: pd.DataFrame, deployment: DeploymentConfig
) -> dict[str, float | int]:
    """Capital-weighted results for one summary row's trades.

    Each row simulates a book dedicated to that one market-and-strategy scope,
    because that is the only reading the summary's grouping supports. A reader
    running both strategies from one account holds a different portfolio than
    any single row describes -- the slots are shared and the daily cap is
    shared -- so the rows bound the outcome rather than predict it.
    """

    if not can_measure_deployment(group):
        # Callers that only ever wanted the per-trade summary need not carry
        # entry and exit dates, and this layer arriving is no reason to start
        # failing them. The columns report blanks rather than a fabricated zero.
        return {
            **unmeasured_metrics().as_row(),
            "Greedy Hindsight Low CAGR": float("nan"),
            "Greedy Hindsight High CAGR": float("nan"),
        }
    capital_group = group.copy()
    if {"Deployment Exit Timestamp", "Deployment NetReturnPct"} <= set(group.columns):
        # Raw trade returns remain the strategy evidence. The separate columns
        # are the portfolio-only holding-cap and terminal-valuation path.
        capital_group["Exit Timestamp"] = group["Deployment Exit Timestamp"]
        capital_group["NetReturnPct"] = group["Deployment NetReturnPct"]
        capital_group["Status"] = "closed"
    metrics = deployment_metrics(capital_group, deployment)
    floor, ceiling = selection_bounds(capital_group, deployment)
    return {
        **metrics.as_row(),
        "Greedy Hindsight Low CAGR": floor,
        "Greedy Hindsight High CAGR": ceiling,
    }


def _attach_deployment_valuations(
    trades: pd.DataFrame, history: pd.DataFrame, config: ReversionValidationConfig
) -> pd.DataFrame:
    """Attach costed holding-cap exits without changing raw strategy evidence."""

    if trades.empty:
        return trades.copy()

    panels: dict[str, pd.DataFrame] = {}
    price_limit_ratio = BacktestConfig().price_limit_ratio
    for symbol, raw in history.groupby("Symbol", sort=False):
        panel = raw.sort_values("Date").set_index("Date")
        cant_sell = build_trade_block_masks(panel, price_limit_ratio)[1]
        panels[str(symbol)] = panel.assign(CantSell=cant_sell)

    def net_return(row: pd.Series, price: float) -> float:
        entry_cash = float(row["Size"]) * float(row["Avg Entry Price"]) + float(row["Entry Fees"])
        if entry_cash <= 0:
            raise ValueError("Deployment valuation requires a positive entry cash amount.")
        exit_cash = (
            float(row["Size"])
            * price
            * (1.0 - config.slippage_rate)
            * (1.0 - config.fee_rate - config.sell_tax_rate)
        )
        return exit_cash / entry_cash - 1.0

    def terminal_mark(row: pd.Series) -> tuple[pd.Timestamp, float]:
        panel = panels.get(str(row["Symbol"]))
        if panel is None:
            raise ValueError(f"Deployment valuation has no history for {row['Symbol']}.")
        end = pd.Timestamp(row["Exit Timestamp"])
        if pd.isna(end):
            end = pd.Timestamp(panel.index.max())
        eligible = panel.loc[panel.index <= end]
        eligible = eligible[pd.to_numeric(eligible["Close"], errors="coerce").gt(0)]
        if eligible.empty:
            raise ValueError(f"Deployment valuation has no terminal mark for {row['Symbol']}.")
        marked = eligible.iloc[-1]
        return pd.Timestamp(marked.name), net_return(row, float(marked["Close"]))

    def forced_exit(row: pd.Series, cap_date: pd.Timestamp) -> tuple[pd.Timestamp, float]:
        panel = panels.get(str(row["Symbol"]))
        if panel is None:
            raise ValueError(f"Deployment valuation has no history for {row['Symbol']}.")
        eligible = panel.loc[panel.index >= cap_date]
        eligible = eligible[
            pd.to_numeric(eligible["Open"], errors="coerce").gt(0)
            & ~eligible["CantSell"].astype(bool)
        ]
        if not eligible.empty:
            filled = eligible.iloc[0]
            return pd.Timestamp(filled.name), net_return(row, float(filled["Open"]))
        return terminal_mark(row)

    valued = trades.copy()
    valued["Deployment Exit Timestamp"] = pd.to_datetime(valued["Exit Timestamp"])
    valued["Deployment NetReturnPct"] = pd.to_numeric(valued["NetReturnPct"], errors="coerce")
    open_positions = valued["Status"].astype(str).str.lower().eq("open")
    for index in valued.index[open_positions]:
        timestamp, marked_return = terminal_mark(valued.loc[index])
        valued.at[index, "Deployment Exit Timestamp"] = timestamp
        valued.at[index, "Deployment NetReturnPct"] = marked_return

    capped_input = valued.copy()
    capped_input["Exit Timestamp"] = capped_input["Deployment Exit Timestamp"]
    capped_input["NetReturnPct"] = capped_input["Deployment NetReturnPct"]
    capped = truncate_long_holds(
        capped_input, config.deployment.max_holding_days, forced_exit
    )
    valued["Deployment Exit Timestamp"] = capped["Exit Timestamp"]
    valued["Deployment NetReturnPct"] = capped["NetReturnPct"]
    return valued


def write_reversion_validation_reports(
    result: ReversionValidationResult,
    html_output: Path,
    csv_output: Path,
    deployment: DeploymentConfig | None = None,
) -> tuple[Path, Path]:
    """Write an HTML acceptance table and a reproducible CSV.

    ``deployment`` only labels the capital columns with the rule that produced
    them. A report that states a CAGR without stating the slot count and daily
    cap behind it is not reproducible, and the numbers move a lot with both.
    """

    deployment = deployment or DeploymentConfig()
    validate_deployment_config(deployment)
    daily_cap = deployment.daily_entry_cap
    slots = deployment.position_slots
    samples = deployment.ordering_samples
    html_output.parent.mkdir(parents=True, exist_ok=True)
    csv_output.parent.mkdir(parents=True, exist_ok=True)
    # Reports are a public metric surface: discard stale/cache-only columns
    # rather than silently re-publishing a retired cumulative-return measure.
    summary = result.summary.reindex(columns=SUMMARY_COLUMNS).copy()
    trades = result.trades.drop(columns=["Cumulative Return (KRW)"], errors="ignore")
    display = summary.copy()
    for column in display.columns:
        display[column] = [format_metric(column, value) for value in display[column]]
    display["Passes Goal"] = summary["Passes Goal"].map(
        lambda value: "N/A" if pd.isna(value) else ("PASS" if value else "FAIL")
    )
    headers = "".join(f"<th>{escape(str(column))}</th>" for column in display.columns)
    rows = [
        "<tr>" + "".join(f"<td>{escape(str(value))}</td>" for value in row) + "</tr>"
        for row in display.itertuples(index=False, name=None)
    ]
    html_output.write_text(
        f"""<!doctype html><html lang="ko"><head><meta charset="utf-8">
<title>Reversion validation</title><style>
body{{font-family:system-ui;margin:32px;max-width:1400px}}table{{border-collapse:collapse;width:100%}}
th,td{{padding:8px;border:1px solid #ddd;text-align:right}}th:nth-child(-n+3),td:nth-child(-n+3){{text-align:left}}
th{{background:#f5f5f5}}</style></head><body><h1>Reversion strategy validation</h1>
<p>PASS requires the configured closed-trade count, costed win rate, profit factor, a positive median,
and the profitable-window share. Mean net return is reported but is not a criterion: it has no time
denominator, so it scores a rule that ties capital up for 233 sessions above one that turns over in 18.
Profitable Window Share is the fraction of out-of-sample windows whose closed trades sum to a profit --
the measure that separates a rule earning steadily from one carried by a few windows. One continuous
backtest per symbol, so no position is truncated at a window boundary; windows are assigned by each
closed trade's entry date. Open Trade Count is positions still unclosed at the end of the history: they
carry no realised return and are excluded from the goal, so a large count means the columns beside it
describe only part of the sample. Research only.</p>
<p>The capital columns answer a different question: not "is each trade good?" but "what would the money
have done?". They replay the shipped deployment rule -- at most {daily_cap} entries a session, {slots}
position slots, never a name already held -- and report the resulting CAGR, so a row can carry a profit
factor above 1.7 and still trail an index fund once signal bursts and idle capital are counted. Skipped
No Slot versus Skipped Daily Cap says which limit actually bound. <strong>These columns do not gate
PASS</strong>; the CAGR a strategy must clear is a decision for DECISIONS.md, not a default. Three
cautions, all of which make the figures optimistic rather than pessimistic. Realised Max Drawdown moves
only at exits, because validation trades carry no daily mark -- a position 30% underwater mid-hold is
invisible until it closes, so read it as a floor. CAGR is the median of {samples} random same-day
orderings, with Low/High the 5th and 95th percentiles; Selection Floor and Ceiling are what perverse and
perfect hindsight would have produced, and since the live tie-break is market capitalisation rather than
a quality ranking, the width between them is what depends on choosing among a session's candidates
rather than on the signal. And each row is a book dedicated to that one market-and-strategy scope:
running several from one account shares the slots and the cap, so the rows bound the outcome rather than
predict it.</p>
<table><thead><tr>{headers}</tr></thead><tbody>{''.join(rows)}</tbody></table></body></html>""",

        [summary.assign(RecordType="summary"), trades.assign(RecordType="trade")],
        ignore_index=True, sort=False,
    ).to_csv(csv_output, index=False)
    return html_output, csv_output


def _label_windows(
    trades: pd.DataFrame, dates: pd.DatetimeIndex, config: ReversionValidationConfig
) -> pd.DataFrame:
    trades = trades.copy()
    if trades.empty:
        trades["Window"] = pd.array([], dtype="Int64")
        return trades
    entry_dates = pd.to_datetime(trades["Entry Timestamp"])
    positions = dates.get_indexer(entry_dates)
    unmatched = positions < 0
    if unmatched.any():
        positions = np.where(unmatched, dates.get_indexer(entry_dates, method="pad"), positions)
    in_oos = positions >= config.training_sessions
    window_number = np.where(
        in_oos, (positions - config.training_sessions) // config.testing_sessions + 1, -1
    )
    trades["Window"] = pd.array(np.where(in_oos, window_number, pd.NA), dtype="Int64")
    return trades


def _attach_live_selection_keys(trades: pd.DataFrame, ranked: pd.DataFrame) -> pd.DataFrame:
    """Attach the scanner order known at the previous close, never entry-day data."""

    if trades.empty:
        return trades.copy()
    required = {"Entry Timestamp", "Market", "Symbol", "Strategy"}
    if required - set(trades.columns):
        raise ValueError("Trades are missing live selection inputs.")
    prior = ranked[["Date", "Market", "Symbol", "MarketCapRank"]].copy()
    prior["Symbol"] = prior["Symbol"].astype(str)
    prior["Entry Timestamp"] = prior.groupby(["Market", "Symbol"], sort=False)["Date"].shift(-1)
    prior = prior.drop(columns="Date")

    working = trades.copy()
    working["Entry Timestamp"] = pd.to_datetime(working["Entry Timestamp"])
    working["Symbol"] = working["Symbol"].astype(str)
    attached = working.merge(prior, on=["Entry Timestamp", "Market", "Symbol"], how="left", validate="many_to_one", sort=False)
    if attached["MarketCapRank"].isna().any():
        raise ValueError("Every backtest trade must have a prior-close market-cap rank.")
    signal_types = {SWING_STRATEGY: "swing", LONG_TERM_STRATEGY: "long_term"}
    signal_type = attached["Strategy"].map(signal_types)
    if signal_type.isna().any():
        raise ValueError("Live selection is only defined for standalone reversion strategies.")
    attached[LIVE_SELECTION_KEY] = [
        live_selection_key(rank, symbol, kind)
        for rank, symbol, kind in zip(attached["MarketCapRank"], attached["Symbol"], signal_type, strict=True)
    ]
    return attached.drop(columns="MarketCapRank")


def _run_all(
    groups: list[pd.DataFrame],
    config: ReversionValidationConfig,
    checkpoint_dir: Path | None,
    regimes: dict[str, pd.Series] | None = None,
) -> pd.DataFrame:
    if not groups:
        return _empty_trades()
    saved: list[pd.DataFrame] = []
    chunk_size = config.checkpoint_symbols_per_file
    for chunk_number, first in enumerate(range(0, len(groups), chunk_size), start=1):
        checkpoint = _checkpoint_path(checkpoint_dir, chunk_number)
        if checkpoint is not None and checkpoint.exists():
            saved.append(pd.read_csv(checkpoint))
            continue
        trades = _run_group_chunk(groups[first:first + chunk_size], config, regimes)
        if checkpoint is not None:
            temporary = checkpoint.with_suffix(".tmp")
            trades.to_csv(temporary, index=False)
            temporary.replace(checkpoint)
        saved.append(trades)
    return pd.concat(saved, ignore_index=True) if saved else _empty_trades()


def _run_group_chunk(
    groups: list[pd.DataFrame],
    config: ReversionValidationConfig,
    regimes: dict[str, pd.Series] | None = None,
) -> pd.DataFrame:
    rows = []
    for raw_group in groups:
        group = raw_group.sort_values("Date").set_index("Date")
        market = str(group["Market"].iloc[-1])
        symbol = str(group["Symbol"].iloc[-1])
        prices = group[["Open", "High", "Low", "Close", "Volume", "TradingValue"]]
        allowed = group["EntryAllowed"].astype(bool)
        if regimes:
            # A market regime is one series shared by every symbol, so it is
            # reindexed onto this symbol's calendar rather than recomputed, and
            # only entries are gated -- exits must stay free to cut risk.
            allowed = allowed & regimes[market].reindex(group.index, fill_value=False)
        for strategy in config.strategies:
            settings = _backtest_config(strategy, config)
            signals = generate_signals(prices, settings).copy()
            signals["entries"] = signals["entries"].astype(bool) & allowed
            portfolio = run_signals(prices, signals, settings)
            records = portfolio.trades.records_readable.copy()
            if records.empty:
                continue
            if "Return" not in records:
                raise ValueError("Backtest trade records must include Return.")
            records = records.rename(columns={"Return": "NetReturnPct"})
            # A position open at the end of the history is carried through with
            # its mark-to-market return so the summary can count it. Filtering it
            # out here, once per 63-session window, is the bug this harness fixes.
            if "Status" not in records:
                records["Status"] = "closed"
            records["Status"] = records["Status"].astype(str).str.lower()
            records["Market"] = market
            records["Symbol"] = symbol
            records["Strategy"] = strategy
            rows.append(records)
    return pd.concat(rows, ignore_index=True) if rows else _empty_trades()


def _prepare_checkpoint_dir(
    history: pd.DataFrame,
    config: ReversionValidationConfig,
    benchmark_close: dict[str, pd.Series] | None = None,
) -> Path | None:
    if config.checkpoint_dir is None:
        return None
    directory = config.checkpoint_dir
    directory.mkdir(parents=True, exist_ok=True)
    payload = asdict(config)
    payload["checkpoint_dir"] = None
    payload["history_start"] = str(history["Date"].min())
    payload["history_end"] = str(history["Date"].max())
    payload["history_rows"] = len(history)
    payload["benchmark"] = benchmark_fingerprint(benchmark_close)
    signature = sha256(json.dumps(payload, sort_keys=True, default=str).encode()).hexdigest()
    manifest = directory / "manifest.json"
    if manifest.exists():
        recorded = json.loads(manifest.read_text(encoding="utf-8"))
        if recorded.get("signature") != signature:
            raise ValueError(
                "Checkpoint directory belongs to different validation inputs. "
                "Use a new directory or remove its prior checkpoints."
            )
    else:
        manifest.write_text(json.dumps({"signature": signature}, indent=2), encoding="utf-8")
    return directory


def _checkpoint_path(directory: Path | None, chunk_number: int) -> Path | None:
    if directory is None:
        return None
    return directory / f"chunk_{chunk_number:04d}.csv"


def _backtest_config(strategy: str, config: ReversionValidationConfig) -> BacktestConfig:
    return BacktestConfig(
        strategy=strategy, fee_rate=config.fee_rate, sell_tax_rate=config.sell_tax_rate,
        slippage_rate=config.slippage_rate, minimum_average_trading_value=0,
        minimum_signal_trading_value=0, outlier_gap_threshold=0,
    )


def _empty_trades() -> pd.DataFrame:
    return pd.DataFrame(
        columns=["Market", "Symbol", "Strategy", "Status", "NetReturnPct", "Entry Timestamp"]
    )


def _validate_config(config: ReversionValidationConfig) -> None:
    if config.training_sessions < 1 or config.testing_sessions < 2:
        raise ValueError("training_sessions must be positive and testing_sessions at least two.")
    validate_market_cap_tiers(config.market_cap_tiers)
    if config.liquidity_window < 2:
        raise ValueError("liquidity_window must be at least two.")
    if config.minimum_history_sessions < 2:
        raise ValueError("minimum_history_sessions must be at least two.")
    if config.minimum_average_trading_value < 0:
        raise ValueError("minimum_average_trading_value cannot be negative.")
    if config.outlier_gap_threshold <= 0 or config.outlier_cooldown_sessions < 1:
        raise ValueError("Outlier threshold and cooldown must be positive.")
    if config.checkpoint_symbols_per_file < 1:
        raise ValueError("checkpoint_symbols_per_file must be positive.")
    if config.market_filter_window < 0 or config.market_filter_window == 1:
        raise ValueError("market_filter_window must be zero (disabled) or at least two.")
    if not config.strategies or set(config.strategies) - set(STRATEGIES):
        raise ValueError("strategies must contain supported strategy names.")
    validate_reversion_goal(config.goal)
